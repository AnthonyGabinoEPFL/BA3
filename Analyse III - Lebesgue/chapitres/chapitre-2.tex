\section{Notions de base}

\subsection{Décomposition en boîte rectangulaire}

\begin{definition}[Boîte rectangulaire] \label{def: Boîte rectangulaire}
	Soit $a_i \leqslant b_i \in \R$. On appelle \textbf{boîte rectangulaire} tout ensemble
	\[
	B := \{ \bm x \in \Rn \mid a_j \leqslant x_j \leqslant b_j\}
	\]
	De plus on pose
	\[
	|B| = \Vol B = \prod_{j=1}^n (b_j - a_j).
	\]
	En particulier on a $\Vol B = 0$ si et seulement il existe $k$ tel que $a_k = b_k$.
\end{definition}

Cette définition est compatible avec les partitions finies de boîtes rectangulaires. On appelle $\mathcal B = \{B_j\}_{j=1}^N$, où les $B_j$ sont des boîtes rectangulaires, une \textbf{partition} d'une autre boîte rectangulaire $B$, pourvu que
\[
\forall i\neq j, \quad \mathring B_i \cap \mathring B_j = \emptyset \et \bigcup_{j=1}^N B_j = B
\]
\begin{proposition} \label{prop: partition de boîte}
	Soit $\mathcal B$ une partition finie de la boîte rectangulaire $B$. Alors on a
	\[
	\Vol B = \sum_{B_j \in \mathcal B} \Vol B_j
	\]
	De plus si $\{B_j\}$ est un recouvrement fini de $B$ par boîtes rectangulaires, alors on a
	\[
	\Vol B \leqslant \sum_j \Vol B_j
	\]
\end{proposition}

\prf{La première partie est un résultat démontrée en analyse avancée II dans le chapitre consacré à l'intégrale de Riemann. Pour l'inégalité finale, on se réduit à la première partie.
	
Ainsi on choisit une partition finit $\{\tilde B_j\}$ de $\bigcup_j B_j$ en boîtes rectangulaires, de sorte que chaque $B_j$, ainsi que $B$, est une réunion de membre de cette partition
\[
B_j = \bigcup_{i \in A_j} \tilde B_i \et B = \bigcup_{i \in A_*} \tilde B_i
\]
Ainsi par hypothèse, on a que $i \in A_*$ implique qu'il existe $j$ avec $i \in A_j$. Il suit que
\[
\Vol B = \sum_{i \in A_*} \Vol \tilde B_i \leqslant \sum_j \sum_{i \in A_j} \Vol \tilde B_i = \sum_j \Vol B_j
\]
où on a utiliser la première partie pour la dernière égalité.
}

\begin{proposition}[Décomposition d'ouverts en cubes] \label{prop: Décomposition d'ouverts en cubes}
	Soit $U \subset \Rn$ un ouvert. Alors il existe une collection de cube $\{Q_j\}_{j=1}^\infty$ deux à deux presque disjoints telle que
	\[
	U = \bigcup_{j=1}^\infty Q_j
	\]
\end{proposition}

\begin{definition}[Mesure extérieure] \label{def: Mesure extérieure}
	Soit $E \subset \Rn$. Alors nous définissons la \textbf{mesure extérieure} de $E$ par
	\[
	m^*(E) := \inf_{\bigcup Q_j \supset E} \sum_j \Vol Q_j \in [0, \infty]
	\]
	où $Q_j$ sont des \textbf{cubes fermés}, c'est-à-dire des \hyperref[def: Boîte rectangulaire]{boîtes rectangulaires} dont tous les côtés sont de longueurs identiques.
\end{definition}

\begin{lemme} \label{lem: Mesure des boîtes rectangulaires}
	Soit $R \subset \Rn$ une boîte rectangulaire. Alors on a 
	\[
	m^*(R) = \Vol R
	\]
\end{lemme}

\prf{On va traiter uniquement le cas où $R$ est non-dégénérée. Ainsi par définition de l'infimum, pour $\varepsilon > 0$ il existe un recouvrement $\{Q_j\}$ de $R$ tel que 
	\[
	\sum_j \Vol Q_j \leqslant m^*(R) + \varepsilon
	\]
	Ensuite on dilate les cubes $Q_j$ par un facteur $(1 + \varepsilon)$ de sorte que l'intérieur des nouveaux cubes $Q_j^\varepsilon$ forment un recouvrement de l'ensemble compact $R$. Et donc par Heine-Borel, il existe un sous-recouvrement fini tel que 
	\[
	R \subset \bigcup_{k=1}^N \mathring Q_{j_k}^\varepsilon \subset \bigcup_{k=1}^N Q_{j_k}^\varepsilon
	\]
	Alors la \autoref{prop: partition de boîte} implique que
	\[
	\Vol R \leqslant (1 + \varepsilon)^n\sum_{k=1}^N \Vol Q_{j_k} \leqslant (1 + \varepsilon)^n \sum_j \Vol Q_j.
	\]
	En laissant $\varepsilon \to 0$, on obtient bien que $\Vol R \leqslant m^*(R)$.
	
	De l'autre sens, pour $0 < \delta \ll b_j - a_j$, on construit les partitions
	\[
	[a_j, b_j] = [a_j, a_j + \delta] \cup [a_j + \delta, a_j + 2 \delta] + \ldots + [a_j + N_j \delta,  b_j]
	\] où $N_j = \left \lfloor \frac{b_j - a_j}{\delta} \right \rfloor$. On a alors un recouvrement de $R$ par des cubes, et donc par la \autoref{prop: partition de boîte} on a
	\[
	m^*(R) \leqslant \Bigg( \prod_{j=1}^n (N_j + 1) \Bigg) \delta^n.
	\]
	En laissant tendre $\delta$ vers $0$, on obtient
	\[
	\Vol R = \lim_{\delta \to 0} \Bigg( \prod_{j=1}^n (N_j + 1) \Bigg) \delta^n \geqslant m^*(R) 
	\]
}


\section{Propriétés de le mesure extérieure et ensemble mesurable}

\subsection{Propriétés de la mesure extérieure}
Pour commencer, il est facile de voir que si $E_1 \subset E_2 \subset \Rn$ alors
\[
\om{E_1} \leqslant \om{E_2}
\]
En effet, parce que si $\{Q_j\}$ est un recouvrement de $E_2$ alors a fortiori c'est aussi un recouvrement de $E_1$, et donc 
\[
\om{E_2} = \inf_{\bigcup Q_j \supset E_2} \sum_j \Vol Q_j \geqslant \inf_{\bigcup Q_j \supset E_1} \sum_j \Vol Q_j = \om{E_1}
\]

\begin{proposition}[Sous-additivité dénombrable] \label{prop: Sous-additivité dénombrable}
	Soit $E = \bigcup_{i=1}^\infty E_j$, alors
	\[
	\om E \leqslant \sum_{j=1}^\infty \om{E_j}
	\]
\end{proposition}

\prf{On peut supposer que pour chaque $E_j$ on ait $\om{E_j} < \infty$. Ainsi soit $\varepsilon >0$, par caractérisation de l'infimum, pour tout $j \geqslant 1$, il existe une collection de cubes $\{Q_{jl}\}_{l = 1}^\infty$ telle que 
	\[
	\sum_{l \geqslant 1} \Vol Q_{jl} \leqslant \om{E_j} + \frac\varepsilon{2^j} \et E_j \subset \bigcup_{l \geqslant 1} Q_{jl}.
	\]
	Ensuite, la collection de cubes ${Q_{jl}}_{j,l}$ est dénombrable, et $E = \cup_{j \geqslant 1} E_j \cup_{j,l \geqslant 1} Q_{jl}$, et donc
	\[
	\om E \leqslant \sum_{j,l \geqslant 1} \Vol Q_{jl} = \sum_{j \geqslant 1} \sum_{l \geqslant 1} \Vol Q_{jl} \leqslant \sum_{j \geqslant 1} \bigg( \om{E_j} + \frac\varepsilon{2^j}\bigg) = \sum_{j \geqslant 1} \om{E_j} + \varepsilon
	\]
	Mais comme $\varepsilon > 0$ est arbitraire, alors le résultat suit.
}

\begin{theoreme}[Additivité 1] \label{thm: Additivité 1}
	Soit $E_1, E_2 \subset \Rn$ deux ensembles non vides tel que $\dist(E_1, E_2) > 0$, alors
	\[
	\om{E_1\cup E_2} = \om{E_1} + \om{E_2}
	\]
\end{theoreme}

\prf{On pose $E = E_1 \cup E_2$, alors par la \autoref{prop: Sous-additivité dénombrable} on sait que
	\[
	\om E \leqslant \om{E_1} + \om{E_2}
	\]
	Il suffit donc de montrer l'inégalité inverse
	\[
	\om E \geqslant \om{E_1} + \om{E_2}
	\]
	Pour ceci, soit $\varepsilon > 0$, et soit $\{Q_j\}_{j=1}^\infty$ une collection de cube \footnote{Encore une fois, on peut supposer que $\om E < \infty$.}
	\[
	\sum_j \Vol Q_j \leqslant \om E + \varepsilon \et E \subset \bigcup_j Q_j
	\]
	On choisit $\delta >  0$ avec $\delta < \dist(E_1, E_2)$. Désormais, on décompose chaque cube $Q_j$ de diamètre $\diam Q_j > \frac\delta2$ où
	\[
	\diam Q_j = \sup_{\bm x, \bm y \in Q_j} \norme{\bm x - \bm y}
	\]
	en des cubes plus petits de diamètres $< \frac\delta2$, et en rejetant les cubes qui ont une intersection vide avec $E$, ce qui nous donne alors une collection $\{\tilde Q_j\}_{j = 1}^\infty$. Or on a toujours
	\[
	\sum_j \Vol \tilde Q_j \leqslant \sum_j \Vol Q_j \leqslant \om E + \varepsilon
	\]
	Mais par construction, on a pour tout cube $\tilde Q_j$ soit
	\[
	\tilde Q_j \subset E_1 \et \tilde Q_j \cap E_2 = \emptyset
	\]
	soit
	\[
	\tilde Q_j \subset E_2 \et \tilde Q_j \cap E_1 = \emptyset
	\]
	On peut alors partitionner la collection $\{\tilde Q_j\} = \{\tilde Q_j^1\}_j \cup \{\tilde Q_j^2\}_j$, où la première famille de cube $\{\tilde Q_j^1\}_j$ est un recouvrement de $E_1$, et la deuxième un recouvrement de $E_2$. Il suit que
	\[
	\om E + \varepsilon \geqslant \sum_j \Vol \tilde Q_j = \sum_j \Vol \tilde Q_j^1 + \sum_j \Vol \tilde Q_j^2 \geqslant \om{E_1} + \om{E_2}.
	\]
	Comme $\varepsilon > 0$ est arbitraire, l'inégalité désirée suit.
}

\lecture{8 octobre}
\begin{theoreme}[Additivité 2] \label{thm: Additivité 2}
	Soit $E = \bigcup_{j=1}^\infty Q_j$ un réunion de cube presque disjoints, c'est-à-dire d'intérieurs disjoints. Il suit que
	\[
	\om E = \sum_j \Vol Q_j = \sum_j \om{Q_j}
	\]
\end{theoreme}

\prf{La deuxième égalité est une conséquence du \autoref{lem: Mesure des boîtes rectangulaires}. Pour la première, par sous-additivité, on a 
	\[
	\om E \leqslant \sum_j \Vol Q_j
	\]
	et donc il suffit de montrer l'inégalité inverse. Pour ceci, on considère la collection $\{Q_j^\varepsilon\}$ des cubes $Q_j$ dilaté d'un facteur $(1- \varepsilon)$ où $\varepsilon > 0$. Ces cubes ont alors une distance positive et donc en utilisant le \autoref{thm: Additivité 1}, on conclut que pour tout $N \geqslant 1$, on a
	\[
	m^* \left(\bigcup_{j=1}^N Q_j^\varepsilon \right) = \sum_{j=1}^N \om{Q_j^\varepsilon}
	\]
	Mais par monotonie, et par le fait que 
	\[
	\bigcup_{j=1}^N Q_j^\varepsilon \bigcup_{j=1}^N Q_j \subset E
	\]
	on a aussi
	\[
	\om E \geqslant 	m^* \left(\bigcup_{j=1}^N Q_j^\varepsilon \right) = \sum_{j=1}^N \om{Q_j^\varepsilon} = (1-\varepsilon)^n \sum_{j=1}^N \Vol Q_j
	\]
	En laissant tendre $N \to + \infty$, on obtient
	\[
	\om E \geqslant (1- \varepsilon)^n \sum_{j \geqslant 1} \Vol Q_j
	\]
	et comme $\varepsilon > 0$ est arbitraire, le théorème suit.
}

\begin{corollaire}
	Soit $U \subset \Rn$ un ensemble ouvert, et soit $U \subset \bigcup_{j=1}^\infty Q_j$ une décomposition presque disjointe en cube fermés. Alors
	\[
	\om U = \sum_j \om{Q_j}
	\]
	En particulier, la somme à droite est indépendante de la décomposition en cube choisie. On peut encore généraliser en disant que si on considère une collection dénombre d'ouverts $U_i$ et disjoints, alors
	\[
	m^* \Bigg(\bigcup_{i=1}^\infty U_j\Bigg) = \sum_{j=1}^\infty \om{U_j}
	\]
\end{corollaire}

\prf{Conséquence directe de la \autoref{prop: Décomposition d'ouverts en cubes} et du \autoref{thm: Additivité 2}.}

\subsection{Mesure de Lebesgue}
\begin{definition}[Mesure de Lebesgue] \label{def: Mesure de Lebesgue}
	On dit que $E \subset \Rn$ est \textbf{mesure au sens de Lebesgue} pourvu que, pour tout $\varepsilon > 0$, il existe un ouvert $U \subset \Rn$ contenant $E$ tel que
	\[
	\om{U \backslash E} < \varepsilon
	\]
	Si $E$ est mesurable au sens de Lebesgue, on définit sa mesure au sens de Lebesgue par
	\[
	m(E) := \om E.
	\]
\end{definition}

\begin{proposition} \label{prop: Mesurabilité d'un ensemble de mesure nulle}
	Soit $E \subset \Rn$ avec $\om E = 0$. Alors $E$ est mesurable au sens de Lebesgue, et $m(E) = 0$.
\end{proposition}

\prf{Soit $E \subset \Rn$ avec $\om E = 0$. Alors par définition de la \nameref{def: Mesure extérieure}, pour tout $\varepsilon > 0$, il existe une collection au plus dénombrable $\{Q_j\}$ de cube tel que
	\[
	\sum_j \Vol Q_j < \frac\varepsilon2 \et E \subset \bigcup_j Q_j
	\]
	Si $\tilde Q_j$ sont des cubes concentriques avec deux fois le volume des $Q_j$, alors $E \subset \bigcup_j \mathring{\tilde Q_j} =: U$, et on a
	\[
	\om{U \backslash E} \leqslant \om U \leqslant \sum_j \Vol \tilde Q_j < \varepsilon 
	\]
}

\begin{proposition}
	Soit $U \subset \Rn$ un ouvert. Alors est mesurable.
\end{proposition}

\prf{Pour tout $\varepsilon > 0$, il suffit de prendre $E = U$, et on a bien
	\[
	\om{U \backslash E} = \om{U \backslash U} = \om \emptyset = 0 < \varepsilon
	\]	
	Ainsi $U$ est bien mesurable au sens de Lebesgue.
}


\begin{definition}[$\sigma$-algèbre] \label{def: sigma algèbre}
	Soit $\mathcal A \subset \mathcal P(\Rn)$ une collection de sous-ensemble de $\Rn$. On dit que $\mathcal A$ est une $\sigma$\textbf{-algèbre} pour vu que
	\begin{enumerate}
		\item $\Rn, \emptyset \in \mathcal A$
		\item Si $A \in \mathcal A$ alors $A^c \in \mathcal A$
		\item Et finalement, si $\{A_i\}_{i=1}^\infty$ est une collection dénombrable d'ensemble dans $\mathcal A$, alors
		\[
		\bigcup_{i=1}^\infty A_i \in \mathcal A
		\]
	\end{enumerate}
\end{definition}


\begin{proposition}[Mesurabilité d'une union] \label{prop: Mesurabilité d'une union}
	Soit $E_j$ une collection dénombrable de sous-ensemble de $\Rn$ qui sont mesurable au sens de Lebesgue. Alors
	\[
	\bigcup_{j=1}^\infty E_j 
	\]
	est mesurable au sens de Lebesgue.
\end{proposition}
\prf{Soit $\varepsilon > 0$. Pour chaque $j \geqslant 1$, on choisit un ouvert $E_j \subset U_j \subset \Rn$ de sorte que
	\[
	\om{U_j \backslash E_j} < \frac{\varepsilon}{2^j}
	\]
	Alors si on pose $U = \bigcup_{j=1}^\infty U_j$, alors $U$ est bien ouvert, par par \nameref{prop: Sous-additivité dénombrable},il suit que
	\[
	m^*\Bigg( U \backslash \bigcup_j E_j \Bigg) \leqslant \sum_j m^* \Bigg( U_j \backslash \bigcup_l E_j \Bigg) \leqslant \sum_j \om{U_j \backslash E_j} < \sum_j \frac\varepsilon{2^j} = \varepsilon
	\]
	Comme $\varepsilon > 0$ est arbitraire, la mesurabilité de $\bigcup_j E_j$ suit.
}
\lecture{13 octobre}
\begin{lemme}[Mesurabilité d'un fermé] \label{lem: Mesurabilité d'un fermé}
	Soit $E \subset \Rn$ un ensemble fermé. Alors $E$ est mesurable au sens de Lebesgue.
\end{lemme}

\prf{On remarque qu'on écrire $E = \bigcup_{i=1}^\infty E \cap \overline{B_j(0)}$, ainsi $E$ s'écrit comme union dénombrable de compact et donc par la \autoref{prop: Mesurabilité d'une union}, il suffit de montrer le lemme pour $E$ compact. Ainsi soit $U \supset E$ un ouvert ouvert et borné contenant $E$. Alors il est clair que $U \backslash E$ est aussi ouvert et borné et donc par la \autoref{prop: Décomposition d'ouverts en cubes} on peut écrire
	\[
	U \backslash E = \bigcup_{l=1}^\infty Q_l
	\]
	où les $Q_l$ sont presque disjoints et fermés. En plus on remarque que
	\[
	\sum_{l=1}^\infty \Vol Q_j = \om{U \backslash E} < \infty
	\]
	Et donc étant donné $\varepsilon >0$, on choisit $N > 0$ tel que
	\[
	\sum_{l=N +1}^\infty \Vol Q_j < \varepsilon
	\]
	Mais comme $\bigcup_{l=1}^N Q_l$ est un sous-ensemble fermé de $U \backslash E$, il suit que
	\[
	E \subset U \backslash \bigcup_{l=1}^N Q_l 
	\]
	et que $U \backslash \bigcup_{l=1}^N Q_l$ est ouvert. En plus on a
	\[
	m^* \Bigg(  \left[ U \backslash\bigcup_{l=1}^N Q_j\right] \backslash E \Bigg) = \sum_{l=N+1}^\infty \Vol Q_j < \varepsilon.
	\]
	Comme $\varepsilon > 0$ est arbitraire, alors $E$ est bien mesurable.
	
}

\begin{proposition} \label{prop: Mesurabilité du complémentaire}
	Soit $E \subset \Rn$ mesurable au sens de Lebesgue. Alors $E^c$ l'est aussi.
\end{proposition}

\prf{Par définition  de la \nameref{def: Mesure de Lebesgue}, pour chaque entier $n \geqslant 1$ il existe $U_n \supset E$ avec $U_n$ ouvert tel que
	\[
	\om{U_n \backslash E} < \frac1n
	\]
	Par le \autoref{lem: Mesurabilité d'un fermé} $U_n^c$ est mesurable, et on a 
	\[
	\om{E^c \backslash U_n^c} = \om{U_n \backslash \backslash E} < \frac1n
	\]
	Ceci implique que 
	\[
	m^* \Bigg( E^c \backslash \bigcup_m U_m^c \Bigg) \leqslant \om{E^c \backslash U_n^c} < \frac1n, \quad \forall n \in \N^*
	\]
	et donc
	\[
	m^* \Bigg( E^c \backslash \bigcup_m U_m^c \Bigg) = 0
	\]
	Et finalement si on écrit
	\[
	E^c = \Bigg( E^c \backslash \bigcup_m U_m^c \Bigg) \cup \Bigg(\bigcup_m U_m^c \Bigg),
	\]
	alors le premier ensemble est mesurable d'après la \autoref{prop: Mesurabilité d'un ensemble de mesure nulle}, et le deuxième ensemble est mesurable d'après le \autoref{lem: Mesurabilité d'un fermé} et de la \autoref{prop: Mesurabilité d'une union}.
}

\begin{corollaire} \label{cor: Caractérisation de la mesurabilité par les fermés}
	Soit $E \subset \Rn$ mesurable et $\varepsilon > 0$. Alors il existe un fermé $F \subset E$ avec 
	\[
	\om{E \backslash F} < \varepsilon
	\]
\end{corollaire}

\prf{Il suffit de choisir un ouvert $U \supset E^c$ avec $\om{U \backslash E^c	} < \varepsilon$, et de poser $F = U^c$.
}


\begin{theoreme}
	L'ensemble $\M$ des ensembles mesurables au sens de Lebesgue en $\Rn$ forment une $\sigma$-algèbre.
\end{theoreme}

\prf{Il suffit de vérifier les trois propriétés d'une \nameref{def: sigma algèbre} ; 
\begin{enumerate}
	\item Comme $\om \emptyset = 0$ alors $\emptyset$ est mesurable par la \autoref{prop: Mesurabilité d'un ensemble de mesure nulle} et donc il en découle que $\Rn$ est mesurable par la \autoref{prop: Mesurabilité du complémentaire},
	
	\item Conséquence directe de la \autoref{prop: Mesurabilité du complémentaire},
	
	\item Conséquence directe de la \autoref{prop: Mesurabilité d'une union}
	\end{enumerate}
	Ainsi $\mathcal M$ forme bien une $\sigma$-algèbre.
}

\begin{corollaire}
	Si $\{E_j\}_{j=1}^\infty$ est une collection d'ensemble mesurable au sens de Lebesgue en $\Rn$, alors
	\[
	\bigcap_{j=1}^\infty E_j
	\]
	l'est aussi.
\end{corollaire}

\prf{Il suffit de remarquer que
	\[
	\Bigg(\bigcap_{j=1}^\infty E_j\Bigg)^c = \bigcup_{i=1}^\infty E_j^c
	\]
}

\section{Propriétés fondamentales de la mesure de Lebesgue}

\subsection{Les ensembles mesurables munis de $m$ forment un espace mesuré}
\begin{definition}
	Soit $\mathcal A$ une $\sigma$-algèbre, alors on dira qu'une application $m : \mathcal A \to [0, +\infty]$ est une \textbf{mesure} pourvu que $m(\emptyset) = 0$ et 
	\[
	m \Bigg(\bigcup_{j=1}^\infty E_j\Bigg) = \sum_{j=1}^\infty m(E_j)
	\]
	pourvu que les ensembles $E_j \in \mathcal A$ soient disjoints. Si c'est le cas, on dira alors que $(\Rn, \mathcal A, m)$ est un \textbf{espace mesuré}.
\end{definition}

\begin{theoreme} \label{thm: additivité de la mesure de Lebesgue}
	L'ensemble $\mathcal M$ des ensembles mesurables au sens de Lebesgue muni de la mesure de Lebesgue $m = m^*$ forment un espace mesuré. C'est-à-dire que pour toute collection $\{E_j\}_{j=1}^\infty$ dénombrable de $\M$ deux à deux disjoints, on a
	\[
	m(E) = \sum_{j=1}^\infty m(E_j), \quad E = \bigcup_{j=1}^\infty E_j
	\]
\end{theoreme}

\prf{On commence par considérer le cas où les ensembles $E_j$ sont tous bornés. Par \nameref{prop: Sous-additivité dénombrable}, on sait que
	\[
	m(E) \leqslant \sum_{j=1}^\infty m(E_j).
	\]
	Il faut donc montrer l'inégalité inverse. Ainsi on considère $\varepsilon > 0$ arbitraire, par le \autoref{cor: Caractérisation de la mesurabilité par les fermés} pour chaque $j \geqslant 1$, il existe un ensemble fermé $F_j \subset E_j$ avec
	\[
	m(E_j \backslash F_j) < \frac\varepsilon{2^j}.
	\]
	Comme $E_j$ est borné, alors $F_j$ l'est aussi et donc est compact. De plus comme les $F_i$ sont disjoints, il suit que
	\[
	\dist(F_i, F_j) > 0, \quad i \neq j
	\]
	Ainsi, par le \autoref{thm: Additivité 1}, il en résulte que
	\[
	m(E) \geqslant m \Bigg(\bigcup_{j=1}^N F_j \Bigg) = \sum_{j=1}^N m(F_j) > \sum_{j=1}^N \Big[m(E_j) - \frac\varepsilon{2^j}\Big] > \sum_{j=1}^N m(E_j) - \varepsilon
	\]
	Comme $N \geqslant 1$ est arbitraire, on obtient par passage à la limite
	\[
	m(E) \geqslant \sum_{j=1}^\infty m(E_j) - \varepsilon
	\]
	En laissant $\varepsilon \to 0$, l'autre inégalité suit.	
	
	Il reste à traiter le cas où des ensembles $E_j$ ne sont pas bornés. On pose $Q_k = [-k, k]^n$ la suite de cube de longueur $2k$. Il est clair que $Q_k \subset Q_{k+1}$ et $\bigcup_k Q_k = \Rn$. On pose ensuite
	\[
	E_{jk} := E_j \cap (Q_{k+1} \backslash Q_k), \quad k \geqslant 1, \qquad E_{j0} = E_j \cap Q_1
	\]
	Il est alors claire que les $E_{jk}$ sont mesurable par opération préservant la mesurabilité et par mesurabilité des  $Q_k$ puisque fermés, de plus on a 
	\[
	E_j = \bigcup_{k=0}^\infty E_{jk}
	\]
	Et donc d'après la première étape de la preuve, on a
	\[
	m(E) = \sum_{jk} m(E_{jk}) \et \forall j, \quad m(E_j) = \sum_k m(E_{jk})
	\]
	Ce qui permet de conclure
	\[
	m(E) = \sum_{jk} m(E_{jk}) = \sum_j \sum_k m(E_{jk}) = \sum_j m(E_j)
	\]
}

\begin{proposition}
	On a deux propriétés analogue concernant des suites d'ensembles monotone ; 
	\begin{enumerate}
		\item Soit ${E_j}_{j=1}^\infty$ une suite d'ensemble mesurable dans $\Rn$ telle que $E_j \subset E_{j+1}$ pour tout $j$. Alors, si on pose
		\[
		E = \bigcup_{j=1}^\infty E_j
		\]
		alors $E$ est mesurable et 
		\[
		m(E) = \lim_{j \to \infty} m(E_j)
		\]
		\item Soit ${E_j}_{j=1}^\infty$ une suite d'ensemble mesurable dans $\Rn$ telle que $E_{j+1} \subset E_j$ pour tout $j$. Alors, si on pose
		\[
		E = \bigcap_{j=1}^\infty E_j
		\]
		alors $E$ est mesurable et 
		\[
		m(E) = \lim_{j \to \infty} m(E_j)
		\]
	\end{enumerate}
\end{proposition}

\prf{La mesurabilité de $E$ est dans les deux cas une conséquence du fait que les ensembles mesurables forment un $\sigma$-algèbre. Pour le reste on a
	\begin{enumerate}
		\item Si on pose 
		\[
		F_j := E_{j+1} \backslash E_j, \quad j \geqslant 1, \quad 	F_0 := E_1
		\]
		Alors on remarque 
		\[
		E = \bigcup_{j = 0}^\infty F_j
		\]
		et les ensembles $F_j$ sont mesurables et disjoints. Ainsi par le \autoref{thm: additivité de la mesure de Lebesgue} on en déduit que
		\[
		m(E) = \sum_{j=0}^\infty m(F_j) \et m(F_j) = m(E_{j+1}) - m(E_J), \quad j \geqslant 1
		\]
		Il suit que
		\[
		\sum_{j=0}^N = m(E_1) + \sum_{j=1}^N [ m(E_{j+1}) - m(E_J)] = m(E_{N+1}), \quad N \geqslant 1
		\]
		On peut donc conclure que
		\[
		m(E) = \lim_{N \to \infty} \sum_{j=0}^N m(F_j) = \lim_{N \to \infty} m(E_{N+1}) = \lim_{j \to \infty} m(E_j)
		\]
		
		\item On remarque que
		\[
		E_1 \backslash E = \bigcup_{j=1}^\infty (E_j \backslash E_{j+1})
		\]
		On a alors une union d'ensembles disjoints et donc par le \autoref{thm: additivité de la mesure de Lebesgue} on en déduit que
		\begin{align*}
			m(E_1\backslash E) &= \sum_{j=1}^\infty m(E_j \backslash E_{j+1}) = \lim_{N \to \infty} \sum_{j=1}^N  m(E_j \backslash E_{j+1}) \\
			&= \lim_{N \to \infty} \sum_{j=1}^N [ m(E_j \backslash E_{j+1})] = \lim_{N \to \infty} [m(E_1)- m(E_{N+1})]
		\end{align*}
		On a déduit que
		\[
		m(E_1) - m(E) = m(E_1\backslash E) = m(E_1) - \lim_{N \to \infty} m(E_{N+1})
		\]
		ainsi on a bien
		\[
		m(E) = \lim_{N \to \infty} m(E_{N+1}) = \lim_{j \to \infty} m(E_j)
		\]
	\end{enumerate}	
}

\begin{theoreme}[Invariance par translation] \label{thm: Invariance par translation}
	Soit $T : \Rn \to \Rn$ une translation. Alors si $E$ est mesurable, $T(E)$ l'est aussi, et
	\[
	m(T(E)) = m(E)
	\]
\end{theoreme}

\prf{Soit $Q \subset \Rn$ un cube, alors $T(Q)$ est aussi un cube, et $\Vol T(Q) = \Vol Q$. On en déduit alors facilement que
	\[
	m^*(E) = \inf_{\bigcup_j Q_j \supset E} \sum_j \Vol Q_j = \inf_{\bigcup_j T(Q_j) \supset T(E)} \sum_j \Vol Q_j = \inf_{\bigcup_j T(Q_j) \supset T(E)} \sum_j \Vol T(Q_j) = m(T(E))
	\]
	En particulier si $E$ est mesurable, étant donné $\varepsilon > 0$, il existe $U \subset \Rn$ ouvert avec $E \subset U$ et $m^*(U \backslash E) < \varepsilon$. Mais $T(E) \subset T(U)$ et donc
	\[
	m^*(T(U) \backslash T(E)) = m^*(T(U \backslash E)) = m^*(U \backslash E) < \varepsilon.
	\]
	Il suit que $T(E)$ est mesurable. Et comme $m^*(E) = m(E)$ pour $E$ mesurable, le théorème suit.
}
