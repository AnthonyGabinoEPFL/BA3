\begin{definition}[Définition de base]
	Soit $U \subset \Rn$ un \textbf{domaine}, c'est-à-dire un ouvert et connexe, on appelle
	\begin{enumerate}
		\item $f : U \to \R$, un \textbf{champ scalaire},
		\item $f : U \to \Rn$, un \textbf{champ vectoriel}.
	\end{enumerate}
	De plus si $f$ est $k$-fois continûment dérivable, on notera respectivement $f \in C^k(U, \R)$ et $\bm f \in C^k(U, \Rn)$, ou de manière générale $f \in C^k$ si le contexte est clair.
\end{definition}

\begin{definition}[Gradient] \label{def: Gradient}
	Soit $g \in C^1$ un champ scalaire. Alors on apelle
	\[
	\bm f = \nabla g
	\]
	le \textbf{gradient} de $g$. Dans ce cas, on dit que $\bm f$ dérive d'un champ scalaire $\bm f$.
\end{definition}

\section{Intégrale de chemin}
 
 \begin{definition}[chemin et reparamétrisation] \label{def: Chemin et reparamétrisation}
 	Un \textbf{chemin} est une application continue
 	\[
 	\gamma : I := [s,t] \to U
 	\]
 	$C^1$ par morceaux, on dira que $\gamma$ est un \textbf{lacet} si $\gamma(s) = \gamma(t)$. De plus si on considère un difféomorphisme $\varphi : [\tilde s, \tilde t] \to [s,t]$ tel que $\varphi' > 0$. Alors on appelle
 	\[
 	\tilde \gamma := \gamma \circ \varphi : [\tilde s, \tilde t] \to U
 	\]
 	une \textbf{reparamétrisation} de $\gamma$.
 \end{definition}
 
 \subsection{Intégrale curviligne}

 \begin{definition}[Intégrale curviligne] \label{def: Intégrale curviligne}
 	Soit $f \in C^0$ un champ scalaire, et soit $\gamma : I \to U$ un chemin, alors on définit l'\textbf{intégrale curviligne} de $f$ le long de $\gamma$ par
 	\[
 	\int_\gamma f  \, \mathrm ds = \int_I f(\gamma(t)) \, \norme{\gamma'(t)} \, \mathrm dt
 	\]
 	où $\norme{\gamma'(t)} = \sqrt{\sum_{i=1}^n \gamma_i'(t)^2}$.
 \end{definition}
 
 
 \begin{theoreme}[Invariance par reparamétrisation] \label{thm: Invariance par reparamétrisation}
 	Soit $U \subset \Rn$ un domaine, $f \in C^0$ un champ scalaire, $\gamma : [s,t] \to U$ un chemin et $\tilde \gamma : [\tilde s, \tilde s] \to U$ une reparamétrisation, alors
 	\[
 	\int_{\tilde \gamma} f \, \mathrm ds = \int_\gamma f \, \mathrm ds
 	\]
 	autrement dit $\int_\gamma f \, \mathrm ds$ est \textbf{invariant par reparamétrisation}.
\end{theoreme}
\prf{En utilisant les mêmes notations que dans la \autoref{def: Chemin et reparamétrisation}, on a
	\begin{align*}
		\int_{\tilde \gamma} f \, \mathrm ds = \int_{\gamma \circ \varphi} f \, \mathrm ds &= \int_{\tilde I} f(\gamma \circ \varphi(t)) \, \norme{(\gamma \circ \varphi)'(t)} \, \mathrm dt = \int_{\tilde I} f(\gamma (\varphi(t))) \, \norme{\varphi'(t) \gamma'(\varphi(t))} \, \mathrm dt \\
		&= \int_{\tilde I} f(\gamma(\varphi(t))) \, \norme{\gamma'(\varphi(t))} \, \varphi'(t) \, \mathrm dt = \int_{\varphi(\tilde s)}^{\varphi(\tilde t)} f(\gamma(y)) \, \norme{\gamma'(y)} \, \mathrm dy \\
		&= \int_s^t f(\gamma(y)) \, \norme{\gamma'(y)} \, \mathrm dy = \int_\gamma f \, \mathrm ds
	\end{align*}
	d'où le résultat.
}

\subsection{Circulation d'un champ vectoriel}

\begin{definition}[Circulation] \label{def: Circulation}
	Soit $U \subset \Rn$ un domaine et $\bm f \in C^0$ un champ vectoriel et $\gamma : I \to U$ un chemin, alors on définit la \textbf{circulation} du champ vectoriel le long de $\gamma$ comme
	\[
	\int_{\gamma} \bm f \cdot \mathrm d \bm s = \int_I \bm f(\gamma(t)) \cdot \bm \gamma'(t) \, \mathrm dt
	\]
	où $\cdot$ désigne le produit scalaire euclidien.
\end{definition}

\begin{definition}[Concaténation de chemin] \label{def: Concaténation de chemin}
	Soit $U \subset \Rn$ un domaine et soient $\gamma_1 : [a,b] \to U$ et $\gamma_1 : [b, c] \to U$ deux chemins tels que $\gamma_1(b) = \gamma_2(b)$, alors on définit la \textbf{concaténation} $\gamma_1 \oplus \gamma_2$ par 
	\[
	\begin{array}{cccc}
		\gamma \oplus \tilde \gamma : & [a,c] & \longrightarrow & U \\
										& t & \longmapsto & 
										\begin{cases}
											\gamma(u), & u \in [a,b] \\
											\tilde\gamma(u), & u \in [b,c]
										\end{cases}
	\end{array}
	\]
	Il est facile de voir que $\gamma \oplus \tilde \gamma$ est bien $C^1$ par morceaux et donc il s'agit bien d'un chemin.
\end{definition}

\begin{remarque}
	Il est facile de voir que
	\[
	\int_{\gamma_1 \oplus \gamma_2} \bm f \cdot \mathrm d \bm s = \int_{\gamma_1} \bm f \cdot \mathrm d \bm s + \int_{\gamma_2} \bm f \cdot \mathrm d \bm s \et \int_{-\gamma_1} \bm f \cdot \mathrm d \bm s = - \int_{\gamma_1} \bm f \cdot \mathrm d \bm s 
	\]
	où $-\gamma_1(t) := \gamma_1(a+b -t)$.
\end{remarque}

On se demande s'il existe des champs vectoriels $\bm f$ tels que $\int_\gamma \bm f \cdot \mathrm d \bm s$ ne dépend que du point initial et du point final de $\gamma$.

\begin{lemme}
	Soit $U \subset \Rn$ un domaine, $\bm f \in C^0$ un champ vectoriel. Alors $\bm f$ à la propriétés que $\int_\gamma \bm f \cdot \, \mathrm d \bm s$ ne dépend que des extrémités des chemins si et seulement si $\int_\gamma \bm f \cdot \, \mathrm d \bm s  = 0$ pour chaque lacet $\gamma : I \to U$.
\end{lemme}

\prf{
	\begin{enumerate}
		\item[$\implies$ :] 
		Supposons que $\int_\gamma \bm f \cdot \, \mathrm d \bm s$ ne dépend que des extrémités, on considère un lacet $\gamma$ arbitraire. On considère deux points arbitraire $\bm p$ et $\bm q$ du lacet, alors on sait qu'il existe $\gamma_1$ et $\gamma_2$ reliant $\bm p$ et $\bm q$ tel que $\gamma = \gamma_1 \oplus -\gamma_2$  et donc a 
		\[
		\int_\gamma \bm f \cdot \mathrm d \bm s = \int_{\gamma_1 \oplus -\gamma_2} \bm f \cdot \mathrm d \bm s = \int_{\gamma_1} \bm f \cdot \mathrm d \bm s - \int_{\gamma_2} \bm f \cdot \mathrm d \bm s = 0
		\]	
		\item[$\impliedby$ :]
		Et inversement, alors si $\int_\gamma \bm f \cdot \, \mathrm d \bm s = 0$ pour tout lacet $\gamma : I  \to U$m  et si $\gamma_1, \gamma_2$ lient $\bm p$ et $\bm q$, alors on peut introduire le lacet qui la concaténation de $\gamma_1$ et $-\gamma_2$ et donc
		\[
		\int_{\gamma_1} \bm f \cdot \, \mathrm d \bm s - \int_{\gamma_2} \bm f \cdot \, \mathrm d \bm s = \int_{\gamma_1 \oplus - \gamma_2} \bm f \cdot \, \mathrm d \bm s = 0
		\]
		et donc on a bien la propriété voulue.
	\end{enumerate}
}

\begin{theoreme} \label{thm: caractérisation de dérive}
	Soit $U \subset \Rn$ un domaine, et $\bm f \in C^k(U)$ un champ vectoriel, alors $\int_\gamma \bm f\cdot \, \mathrm d \bm s = 0$ pour tout lacet $\gamma$ si et seulement si $\bm f$ dérive d'un potentiel.
\end{theoreme}

\prf{\begin{enumerate}
		\item[$\implies$ :] 
		Supposons que $\bm f = \nabla g$, et que $\gamma : [0,1] \to U$ est un lacet, alors
		\[
		\int_\gamma \bm f \cdot \mathrm d \bm s = \int_0^1 \nabla g(\gamma(t)) \cdot \gamma'(t) \, \mathrm dt = \int_0^1 \frac{\mathrm d}{\mathrm dt} g(\gamma(t)) \, \mathrm dt = g(\gamma(1)) - g(\gamma(0))
		\]
		mais comme $\gamma$ est un lacet alors $\gamma(1) = \gamma(0)$ et donc on a bien
		\[
		\int_\gamma \bm f \cdot \mathrm d \bm s = 0
		\]	
		\item[$\impliedby$ :]
		Réciproquement supposons que $\int_\gamma \bm f \cdot \mathrm d \bm s = 0$ pour tout lacet. Par connexité de $U$, on considère $p \in U$, et on pose
		\[
		g(x) = \int_\gamma \bm f \cdot \mathrm d \bm s
		\]
		où $\gamma$ est un chemin arbitraire en $U$ qui relie $p$ à $x$. D'après  les hypothèse, alors $g$ est bien défini. Il est facile de voir que $g$ est de classe $C^0$. De plus
		\[
		\frac{\partial g}{\partial x_j} (x) = \lim_{h \to 0} \frac{g(x + h e_j) - g(x)}{h} 
		\]
		or $g(x + h e_j) = \int_{\tilde \gamma} \bm f \cdot \mathrm d\bm s$ avec $\tilde \gamma $ est un chemin arbitraire qui lie $p$ à $x + he_j$. Or par concaténation on peut écrire, $\tilde \gamma = \gamma \oplus \gamma_1$ où $\gamma_1 : [0,1] \to U, \quad \gamma_1(t) = x  + t h e_j$,
		\begin{figure}[H]
			\centering
			\includegraphics[width=0.25\textwidth]{"images/concaténation.png"}	
		\end{figure}
		ce qui est bien valable car $\int_\gamma \bm f \cdot \mathrm d \bm s$ ne dépend que des extrémités. Ainsi
		\[
		g(x +he_j) = \int_{\gamma \oplus \gamma_1} \bm f \cdot \mathrm d \bm s = \int_\gamma \bm f \cdot \mathrm d \bm s + \int_{\gamma_1} \bm f \cdot \mathrm d \bm s
		\]
		Et donc on a
		\begin{align*}
		g(x + h e_j) - g(x) &= \int_{\gamma_1} \bm f \cdot \mathrm d \bm s = \int_0^1 \bm f(x + th e_j) \cdot h e_j \, \mathrm dt = h \int_0^1 f_j(x + the_j) \, \mathrm dt  \\&= h \int_0^1 f_j(x) \, \mathrm dt + h \int_0^1 r_j(t,h,x) \, \mathrm dt
		\end{align*}
		où $r_j(t,h,x) = f_j(x + the_j) - f_j(x) \longrightarrow 0$ et donc $h \int_0^1 r_j(t,h,x) \, \mathrm dt = o(h)$ et donc 
		\[
		\frac{\partial g}{\partial x_j} (x) = \lim_{h \to 0} \frac{g(x + h e_j) - g(x)}{h} = f_j(x)
		\]
		Ainsi les dérivées partielles existent et sont continues, ainsi $g \in C^1$, de plus on a bien $\nabla g = \bm f$.
	\end{enumerate}
}

\begin{remarque}
	On voit que si $\bm f = \nabla g$ avec $g \in C^2(U)$ alors
	\[
	\frac{\partial f_j}{\partial k} = \frac{\partial^2 g}{\partial x_k \partial x_j} = \frac{\partial^2 g}{\partial x_j \partial x_k} = \frac{\partial f_k}{\partial x_j}
	\]
	Ainsi on condition nécessaire pour que $\bm f \in C^1(U, \Rn)$ dérive d'un potentiel est alors que
	\[
	\forall i,j \in \entiers, \quad \frac{\partial f_j}{\partial x_i} - \frac{\partial f_i}{\partial x_j}
	\]
	Cependant dans le cadre de ce cours, on va se restreindre au cas $n=2$ ou $n=3$, alors on a la condition nécessaire
	\[
	\bm f = \nabla g \implies \rot \bm f = 0
	\]
\end{remarque}

\subsection{Concepts topologiques}

\begin{definition}[Homotopie] \label{def: Homotopie}
	Soit $U \subset \Rn$ un domaine, et soient $\gamma_1, \gamma_2 : [0,1] \to U$ deux lacets dans $U$ possédant les mêmes points d'extrémité. Alors on dit que $\gamma_1$ est \textbf{homotope} à $\gamma_2$ s'il existe une application continue
	\[
	\Gamma: [0,1] \times [0,1] \to U
	\]
	telle que $\Gamma(0,t) =  \gamma_1(t)$ et $\Gamma(1,t) = \gamma_2(t)$ et pour chaque $s \in [0,1]$ fixé, $\Gamma(s,t)$ est un chemin qui lie les mêmes point d'extrémité que $\gamma_1$ et $\gamma_2$. La définition est analogue pour les lacets.
\end{definition}


\begin{definition}[Contractibilité et simplement connexe] \label{def: Contractibilité et simplement connexe}
	Soit $U \subset \Rn$ un domaine, et $\gamma : [0,1] \to U$ un lacet, alors on dit que $\gamma$ est \textbf{contractible} pourvu qu'il soit homotope à un lacet constant en $U$. En plus si $U \subset \Rn$ dont tous les lacets sont contractibles, alors on dira que $U$ est \textbf{simplement connexe}.
\end{definition}

\begin{definition}[Étoilé] \label{def: Étoilé}
	ON appelle un domaine $U \subset \Rn$ \textbf{étoilé}, pourvu qu'il existe un $p \in U$ tel que pour tout $x \in U$, le segment de ligne droite $[p,x]$ est inclus dans $U$. De plus on dira que $U$ est \textbf{convexe} pourvu que $U$ est étoilé par rapport à chacun de ses points.
\end{definition}

\begin{lemme}
	Si $U$ est étoilé alors $U$ est simplement connexe.
\end{lemme}

\prf{Soit $\gamma : [0,1] \to U$ un lacet. Soit $p$ dans la définition, alors on pose
	\[
	\Gamma(s,t) = sp + (1-s) \gamma(t), \quad s \in [0,1], \, t \in [0,1]
	\]
	Si $s = 1, \quad \Gamma(1,t) = p$ donc c'est un lacet constant. Et $\Gamma(s,t) \in U$ pour tout $(s,t) \in [0,1]^2$. 
}


\section{Green-Riemann et potentiels scalaires}

\subsection{Green-Riemann}
\begin{proposition}[Partition de l'unité] \label{prop: Partition de l'unité}
	Soit $K \subset \Rn$ un sous-ensemble compact, soit $U_i, V_i$ pour $i = 1, \ldots, N$ des domaines tels que $\overline U_i \subset V_i$, chaque $U_i$ est borné, et 
	\[
	K \subset \bigcup_{i=1}^N U_i
	\]
	Alors, il existe des fonctions $\phi_j \in C^\infty_0(V_j), \quad \phi_j \geqslant 0$ et telles que
	\[
	\forall x \in K, \quad \sum_{j=1}^N \phi_j(x) = 1
	\]
\end{proposition}

\lecture{14 septembre}

\prf{On construit des fonctions $g_i \in C^0_0(V_i)$ de sorte que ${g_i}_{\mid \overline U_i} = 1$ pour tout $i \in \llbracket 1, N \rrbracket$. 
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.2\textwidth]{"images/ui-vi.png"}
	\end{figure}
	Ainsi soit $\varepsilon_i = \dist(U_i, V_i^c) = \dist(\overline U_i, V_i^c) > 0$ car $\overline U_i$ et $V_i^c$ sont fermés et $\overline U_i \cap V_i^c = \emptyset$. On pose donc $\varepsilon = \min_{i \in \llbracket 1, N \rrbracket} \varepsilon_i$ et on définit $\zeta : \R_+ \to \R$ comme
	\[
	\zeta : t \mapsto \begin{cases} 1 & t \in [0, \varepsilon/2] \\ (\varepsilon/2 -t)/(\varepsilon/4) & t \in [\varepsilon/4, \varepsilon/2] \\ 0 & t \in [\varepsilon/2, \infty[ 
	\end{cases}
	\]
	et finalement on pose $g_i = \zeta(\dist(x, U_i))$. On remarque alors que $g_i \in C^0$ par composition de fonction continue. Il reste donc à régulariser $g_i$
	
	Ainsi soit $\psi \in C^\infty_0(B_1(0))$, de plus on peut supposer que $\psi \geqslant 0$ et $\int_{\Rn} \psi (x) \, \mathrm dx = 1$. Pour tout $\delta > 0$ on définit $\psi_\delta(x) = \delta^{-n} \psi(x \delta^{-1}) \in C^\infty_0(B_\delta(0))$. On considère ensuite les convolutions suivantes
	\[
	\tilde \varphi_i(x) = (g_i * \psi_\delta)(x) = \int_{\Rn} g_i(x -y) \psi_\delta(y) \, \mathrm dy =  \int_{\Rn} g_i(x) \psi_\delta(x - y) \, \mathrm dy
	\]
	les fonctions $\tilde \varphi_i$ sont bien définit et sont de classe $C^\infty$. On considère en particulier $\delta < \varepsilon/4$, alors pour tout $x \in \overline U_i$ et $y \in B_\delta(0)$, alors
	\[
	\dist(x-y, U_i) \leqslant \dist(x, U_i) + \norme y < \delta < \varepsilon/4
	\]
	ainsi pour tout $y \in B_\delta(0), \quad g_i(x -y) = 1$ si $x \in \overline U_i$, et donc
	\[
	\forall x \in \overline U_i, \quad \tilde \varphi_i (x) = \int_{\Rn} g_i(x-y) \psi_\delta(y) \, \mathrm dy = \int_{B_\delta(0)}g_i(x-y) \psi_\delta(y) \, \mathrm dy = \int_{B_\delta(0)} \psi_\delta(y) \, \mathrm dy = 1
	\]
	ainsi ${\tilde{\varphi}_i}{}_{\mid \overline{U}_i} = 1$. De la même manière si $x \in V^c$ et $y \in B_\delta(0)$ alors
	\[
	\dist(x -y, U_i) \geqslant \dist(x, U_i) - \norme y \geqslant \varepsilon_i - \varepsilon/4 \geqslant \varepsilon - \varepsilon/4 = \frac{3\varepsilon}4
	\]
	donc $\forall y \in B_\delta(0), \quad g_i(x-y)$ si $x \in V_i^c$, ainsi
	\[
	\forall x \in V_i^c, \quad \tilde \varphi_i(x) = \int_{\Rn} g_i(x-y) \psi_\delta(y) \, \mathrm dy = \int_{B_\delta(0)}g_i(x-y) \psi_\delta(y) \, \mathrm dy = 0
	\]
	ainsi ${\tilde \varphi_i}{}_{\mid V_i^c} = 0$. Ainsi les propriétés des $g_i$ sont conservées sur les $\tilde \varphi_i$, sauf que désormais $\tilde \varphi_i \in C^\infty_0(V_i)$. 
	
	Finalement, pour tout $i \in \llbracket1, N\rrbracket$, on pose
	\[
	\phi_i = (1 - \tilde \varphi_1)(1 - \tilde \varphi_2) \cdot \ldots \cdot (1- \tilde \varphi_i)
	\]
	Par récurrence immédiate on vérifie que 
	\[
	1 - \sum_{i=1}^N \phi_i = (1 - \tilde \varphi_1) \cdot \ldots \cdot (1 - \tilde \varphi_N)
	\]
	et donc si $x \in K \subset \bigcup_{i=1}^N U_i$, alors il existe $j$ tel que $x \in U_j$, donc $\tilde \varphi_j(x) = 1$ et par conséquent
	\[
	1 - \sum_{i=1}^N \phi_i(x) = 0
	\]
	d'où le résultat.
}


\begin{lemme}[Green-Riemann sur des régions de type I ou II] \label{lem: Green-Riemann sur des régions de type I ou II}
Soit $V$ un région de type $I$, c'est-à-dire qu'il existe $h_{1,2} \in C^1([a,b], \R)$ avec $h_1 < h_2$ tel que
\[
V = \{ (x,y) \in \R^2 \mid x \in [a,b], \quad h_1(x) < y < h_2(x)\}
\]
Alors la conclusion de Green-Riemann s'applique
\[
\int_\gamma \bm f \cdot \mathrm d \bm s = \iint_V \rot \bm f \, \mathrm d x \mathrm dy
\]
pour tout $\bm f \in C^1$, où $\gamma$ est une paramétrisation de $\partial V$ entourant $V$ au sens positif.
\end{lemme}

\prf{On calcule explicitement les deux cotés, ainsi on pour le coté droite
	\[
	\iint_V \partial_x f_2 - \partial y f_1 \, \mathrm d x \, \mathrm dy = \underbrace{\int_a^b \int_{h_1(x)}^{h_2(x)} \partial_x f_2 (x,y) \, \mathrm d x \,\mathrm dy}_A - \underbrace{\int_a^b \int_{h_1(x)}^{h_2(x)} \partial_y f_1 (x,y) \, \mathrm d x \, \mathrm dy}_B
	\]
	Ainsi pour $A$ on trouve
	\begin{align*} 
	A &=  \int_a^b \frac{\partial }{\partial x} \left[ \int_{h_1(x)}^{h_2(x)} f_2(x,y) \, \mathrm dy \right] \mathrm dx - \int_a^b h_2'(x) f_2(x, h_2(x)) \, \mathrm dx + \int_a^b h_1'(x) f_2(x, h_1(x)) \, \mathrm dx \\
	&=\int_{h_1(b)}^{h_2(b)} f_2(b,y) \, \mathrm dy - \int_{h_1(a)}^{h_2(a)} f_2(a,y) \, \mathrm dy - \int_a^b h_2'(x) f_2(x, h_2(x)) \, \mathrm dx + \int_a^b h_1'(x) f_2(x, h_1(x)) \, \mathrm dx 
	\end{align*}
		et pour $B$ on a plus simplement,
	\[
	B =\int_a^b \left(\int_{h_1(x)}^{h_2(x)} \partial_y f_1 (x,y) \, \mathrm d x \right)\mathrm dy = \int_a^b f_1(x, h_2(x)) \, \mathrm dx - \int_a^b f_1(x, h_1(x)) \, \mathrm dx
	\]
	Ensuite par concaténation on pose $\gamma_1 = \{(x,h_1(x)) \mid x \in [a,b]\}$, alors
	\[
	\int_{\gamma_1} \bm f \cdot \mathrm d \bm s = \int_a^b \bm f(x, h_1(x)) \cdot \begin{pmatrix}
		1 \\ h_1'(x) \end{pmatrix} \,\mathrm dx = \int_a^b f_1(x, h_1(x)) \, \mathrm dx + \int_a^b h_1'(x) f_2(x, h_1(x)) \, \mathrm dx 
	\]
	De plus on pose $\gamma_2 = \{ (b,y) \mid y \in [h_1(b), h_2(b)]\}$ ainsi
	\[
	\int_{\gamma_2} \bm f \cdot \mathrm d \bm s = \int_{h_1(b)}^{h_2(b)} \bm f(b,y) \cdot \begin{pmatrix} 0 \\ 1 \end{pmatrix} \, \mathrm dy = \int_{h_1(b)}^{h_2(b)} f_2(b,y) \, \mathrm dy 	
	\]
	En posant $-\gamma_3 = \{ (x, h_2(x)) \mid x \in [a,b] \}$ alors
	\[
	\int_{\gamma_3} \bm f \cdot \mathrm d \bm s = - \int_{- \gamma_3} \bm f \cdot \mathrm d \bm s = - \int_a^b f_1(x, h_2(x)) + h_2'(x) f_2(x, h_2(x)) \, \mathrm dx 
	\]
	Et finalement avec $\gamma_4 = \{ (a,y) \mid y \in [h_2(a), h_1(x)]\}$ alors
	\[
	\int_{\gamma_4} \bm f \cdot \mathrm d \bm s = - \int_{h_1(a)}^{h_2(a)} f_2(a, y) \, \mathrm d \bm s
	\]
	Et donc par concaténation on a bien 
	\[
	\int_\gamma \bm f \cdot \mathrm d \bm s = \int_{\gamma_1} \bm f \cdot \mathrm d \bm s + \int_{\gamma_2} \bm f \cdot \mathrm d \bm s + \int_{\gamma_3} \bm f \cdot \mathrm d \bm s + \int_{\gamma_4} \bm f \cdot \mathrm d \bm s = A - B = \iint_V \rot \bm f \, \mathrm d x \mathrm dy
	\]
	D'où le lemme. La preuve les des domaines de types II est analogue.
}

\begin{theoreme}[Green-Riemann] \label{thm: Green-Riemann}
	Soit $U \subset \Rn$ un domaine, et soit $V \subset U$ un sous-domaine borné, avec $\overline V \subset U$, borné pour un lacet $\gamma$ régulier, simple, orienté positivement. Alors on a que
	\[
	\int_\gamma \bm f \cdot \mathrm d \bm s = \iint_V \rot \bm f \mathrm d x \, \mathrm dy
	\]
\end{theoreme}

\begin{remarque}
	Un lacet $\gamma$ est dit \textbf{simple} $\gamma : [0,1] \to U$ est injective à la condition près $\gamma(0) = \gamma(1)$. Un lacet $\gamma$ est dit \textbf{régulier} si $\gamma \in C^1([0,1], U)$ avec $\gamma'(t) \neq 0$ pour tout $t \in [0,1]$ et $\gamma'(1) = \gamma'(0)$.
\end{remarque}

\prf{Soit $\gamma$ et $V$ comme donnée dans l'énoncé, et $\bm f \in C^1$, alors par régularité de $\gamma$, on a que 
	\[
	\gamma'(t) = \begin{pmatrix} \gamma_1'(t) \\ \gamma_2'(t) \end{pmatrix} \neq 0, \quad \forall t \in I
	\]
	Soit $t_\star \in I$ tel que $\gamma_1'(t_\star) \neq 0$. Ainsi par le théorème des fonctions implicite, il existe un intervalle ouvert $J$ tel que $\gamma_1(t_\star) \in J$, ainsi qu'un autre intervalle ouvert $\tilde J$ tel que $t_\star \in \tilde J$, tel que
	\[
	\forall x \in J, \, \exists ! t \in \tilde J, \quad \gamma_1(t) = x
	\]
	En plus, si on pose $t = t(x) \in C^1(J, \tilde J)$, on peut écrire $y = \gamma_2(t) = \gamma_2(t(x)) := h(x) \in C^1(J)$. Ainsi localement on a 
	\[
	\{ \gamma(x) \mid t \in \tilde J \} = \{ (x, h(x)) \mid x \in J\}
	\]
	
	Pour chaque $t \in I$ on considère le point $\gamma(t)$, et on prend un rectangle $V$ ouvert aligné avec les axes de coordonnées, tel que $V \cap \gamma $ est le graphe d'une fonction par rapport à $x$ ou à $y$. On peut aussi choisir un rectangle ouvert $U$ tel que $\overline U \subset V$ et $\gamma(t) \in U$. On fait ça pour tout $t \in I$, ainsi les $\{U\}$ forment un recouvrement de $\gamma(I)$ par des ouverts. Ainsi par compacité de $\gamma(I)$, d'après Heine-Borel, on peut choisir un nombre fini $\{U_i, V_i\}_{i=i}^N$ tels que $\overline U_i \subset V_i$ et 
	\[
	\gamma(I) \subset \bigcup_{i=1}^N U_i 
	\]
	Pour obtenir un recouvrement de tout $\overline V$. On rajoute encore une paire $U_{N+1}, V_{N+1}$ construit comme suit ; par construit les $U_i$ déjà choisies forment un recouvrement ouvert de $\gamma(I)$, on peut se convaincre qu'il existe $\delta > 0$ tel que
	\[
	\{ x \in V \mid \dist(x, \gamma(I)) < \delta \} \subset \bigcup_{i=1}^N U_i
	\]
	Ensuite on pose,
	\[
	U_{N+1} = \{x \in V \mid \dist(x, \gamma(I)) > \frac\delta2\} \et V_{N+1} = \{x \in V \mid \dist(x, \gamma(I)) > \frac\delta4\}
	\]
	Maintenant qu'on a un recouvrement, on utilise la \autoref{prop: Partition de l'unité} précédente avec $K = \overline V$, et $(U_i, V_i)$, $i \in \llbracket 1, N+1 \rrbracket$. On obtient alors des fonctions
	\[
	\varphi_j \in C^\infty_0 (V_j), \quad \varphi_j \geqslant 0, \quad \sum_{j=1}^{N+1} \varphi_j(x) = 1,\quad \forall x \in \overline V
	\]
	On a alors, pour le champ $\bm f$,
	\[
	\bm f = \sum_{j=1}^{N+1} \varphi_j \bm f
	\]
	Il s'ensuit alors que 
	\[
	\int_\gamma \bm f \cdot \mathrm d \bm s = \int_\gamma \sum_{j=1}^{N+1} \varphi_j \bm f \cdot \mathrm d \bm s = \sum_{j=1}^{N+1} \int_\gamma \varphi_j \bm f \cdot \mathrm d \bm s
	\]
	et de même
	\[
	\iint_V \rot \bm f \, \mathrm dx \, \mathrm dy = \iint_V \rot \sum_{j=1}^{N+1} \varphi_j \bm f \, \mathrm dx \, \mathrm dy = \sum_{j=1}^{N+1} \iint_V \rot(\varphi_j \bm f) \, \mathrm dx \, \mathrm dy
	\]
	Donc il suffit de montrer l'égalité terme par terme. En appliquant le \autoref{lem: Green-Riemann sur des régions de type I ou II} alors 
	\[
	\int_{\partial (V \cap V_j)} \varphi_j \bm f \cdot \mathrm d \bm s = \iint_{V \cap V_j} \rot(\varphi_j \bm f) \, \mathrm d x \, \mathrm dy 
	\]
	Mais comme $\varphi_j \bm f$ est nul en dehors de $V_j$, alors 
	\[
	\iint_V \rot(\varphi_j \bm f) \, \mathrm dx \, \mathrm dy = \iint_{V \cap V_j} \rot(\varphi_j \bm f) \, \mathrm dx \, \mathrm dy
	\]
	De plus le bord de $V \cap V_j$ est constitué de deux morceaux, un morceau de $\gamma$ et un morceau de bord de $V_j$. Or comme $\varphi_j$ est à support compact dans $V_j$, alors elle nulle sur $\partial V_j$ et donc
	\[
	\int_\gamma \varphi_j \bm f \cdot \mathrm d \bm s = \int_{\partial(V \cap V_j)}  \varphi_j \bm f \cdot \mathrm d \bm s
		\]
		On a donc montré que $\int_\gamma \varphi_j \bm f \cdot \mathrm d \bm s = \iint_V \rot(\varphi_j \bm f) \, \mathrm dx \, \mathrm dy$ pour tout $j = 1, 2, \ldots, N$. De plus si $j = N+1$, alors
		\[
		\int_{\gamma} \varphi_{N+1} \bm f \cdot \mathrm ds = 0
		\]
		parce que $\varphi_{N+1}$ n'intersecte pas le lacet $\gamma$. De plus on a
		\[
		\iint_V \rot(\varphi_{N+1} \bm f) \, \mathrm dx \, \mathrm dy = \iint_{\R^2} \rot(\varphi_{N+1} \bm f) \, \mathrm dx \, \mathrm dy = 0
		\]
		car le support de $\varphi_{N+1}$ est un sous-ensemble de $V$. Et donc on a bien l'égalité souhaité
		\[
		\forall j \in \llbracket 1, N+1 \rrbracket, \quad \int_\gamma \varphi_j \bm f \cdot \mathrm d \bm s = \iint_V \rot(\varphi_j \bm f) \, \mathrm dx \, \mathrm dy
		\]
		Ce qui conclut la preuve.	
}

\subsection{Potentiel scalaire en $\R^2$}
\begin{proposition}\label{prop: Condition suffisante de dérive d'un potentiel scalaire}
	Soit $U \subset \Rn$ un domaine et $\bm f \in C^1(U, \Rn)$. Supposons que $\rot \bm f = 0$.
	\begin{enumerate}
		\item Soient $\gamma_1, \gamma_2 : [0,1] \to U$ deux chemines en $U$ avec les mêmes extrémités qui sont homotopes, alors
		\[
		\int_{\gamma_1} \bm f \cdot \mathrm d \bm s = \int_{\gamma_2} \bm f \cdot \mathrm d \bm s
		\]
		\item Soit $\gamma : [0,1] \to U$ un lacet contractible, alors
		\[
		\int_\gamma \bm f \cdot \mathrm d \bm s = 0
		\]
		\item Si $U$ est simplement connexe, alors $\bm f$ dérive d'un potentiel, c'est-à-dire
		\[
		\bm f = \nabla g
		\]
	\end{enumerate}
\end{proposition}

\lecture{24 septembre}

\prf{On va montrer que le point 3. car c'est le plus important. On traite déjà le cas où $U$ est étoilé, ainsi par hypothèse il existe $\bm a \in U$ tel que pour tout $\bm x \in U$, le segment $[\bm a, \bm x]$ est contenu dans $U$. Ainsi il est naturel de définir la fonction
	\[
	g(\bm x) = \int_{\gamma_{\bm a \to \bm x}} \bm f \cdot \mathrm d \bm s 
	\]
	où $\gamma_{\bm a \to \bm x}$ désigne le chemin reliant $\bm a$ et $\bm x$ par un segment. Soit maintenant $\bm x \in U$ et $|h| > 0$ suffisamment petit tel que $\bm x + h \bm e_i \in U$. Alors
	\begin{align*}
		g(\bm x + h \bm e_i) - g (\bm x) &= \int_{[\bm a, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s -\int_{[\bm a, \bm x]} \bm f \cdot \mathrm d \bm s \\
			=&  \left(\int_{[\bm a, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s - \int_{[\bm a, \bm x]} \bm f \cdot \mathrm d \bm s  -\int_{[\bm x, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s  \right) + \int_{[\bm x, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s
			=& \int_{\partial \Delta_{\bm a, \bm x, \bm x + h \bm e_i}} \bm f \cdot \mathrm d \bm s + \int_{[\bm x, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s
	\end{align*} 
	Où $\Delta_{\bm a, \bm x, \bm x + h \bm e_i}$ désigne le triangle de sommets $\bm a, \bm x$ et $\bm x + h \bm e_i$.
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.15\textwidth]{"images/triangle.png"}
	\end{figure}
	Ainsi par le \autoref{thm: Green-Riemann} on obtient que l'intégral est nulle le long de $\partial \Delta$ et donc
	\[
	g(\bm x + h \bm e_i) - g (\bm x) = \int_{[\bm x, \bm x + h \bm e_i]} \bm f \cdot \mathrm d \bm s = \int_0^h \bm f(\bm x + t \bm e_i) \cdot \bm e_i \, \mathrm dt = \int_0^h f_i(\bm x + t \bm e_i) \, \mathrm dt
	\]
	Et donc
	\[
	\frac{\partial g}{\partial x_i}(\bm x) = \lim_{h \to 0} \frac{	g(\bm x + h \bm e_i) - g (\bm x)}{h} = \lim_{h \to 0} \int_0^h f_i(\bm x + t\bm e_i) \, \mathrm dt = f_i(\bm x)
	\]
	Il suit alors que $\nabla g (\bm x) = \bm f(x)$. Et donc $\bm f$ dérive bien d'un potentiel.	
	
	
	Maintenant il nous reste à traiter le cas où $U$ est simplement connexe.
}
\section{Théorème de Kelvin-Stokes et potentiel scalaire en $\R^3$}
\subsection{Intégrale de surface et flux d'un champ à travers une surface}
On pense d'une surface $S \subset \R^2$ comme une généralisation d'un lacet : 
\begin{definition}[Surface paramétrisée] \label{def: Surface paramétrisée}
	Si $U \subset \R^2$ est un domaine et $\varphi : U \to \R^3$ de régularité $C^1$, on dit que $ S = \varphi(U) \subset R^3$ est une \textbf{surface paramétrisée régulière}, pourvu que
	\begin{enumerate}
		\item $\varphi$ est injective,
		\item $\varphi_u \times \varphi_v \neq \bm 0$, où $(u,v)$ sont les coordonnées de $\R^2$, et $\varphi_u = \frac{\partial \varphi}{\partial u}$ et $\varphi_v = \frac{\partial \varphi}{\partial v}$.
	\end{enumerate}
	De plus, soit $\varphi : U \subset \R^2 \to \R^3$ une surface paramétrisée régulière en $\R^3$, alors on dira qu'une fonction $f : S = \varphi(U) \to \R$ est \emph{continue} si $f \circ \varphi : U \to \R$ est continue. 
\end{definition}

\begin{definition}
	Soit $f : S \to \R$ une fonction continue au sens définit précédemment, si le support de $f$ est compact, alors on définit
	\[
	\iint_S f  \, \mathrm d s = \iint_U f \circ \varphi \, \norme{\varphi_u \times \varphi_v} \, \mathrm du \, \mathrm dv
	\]
	De plus, soit $(S, \varphi)$ une surface paramétrisée et soit $V \subset \R^3$ un domaine, si $\psi : V \to U$ est une difféomorphisme $C^1$, alors on appelle la paramétrisation $\varphi \circ \psi : V \to S$ une \textbf{reparamétrisation} de $(S, \varphi)$.
\end{definition}

\begin{lemme}
	L'intégrale est invariante sous reparamétrisation. Alors si on pose $\tilde \varphi := \varphi \circ \psi$, on a 
	\[
	\iint_U f \circ \varphi \, \norme{\varphi_u \times \varphi_v } \, \mathrm du \, \mathrm dv =  \iint_V f \circ \tilde \varphi \, \norme{\tilde \varphi_{\tilde u} \times \tilde \varphi_{\tilde v} } \, \mathrm d \tilde u \, \mathrm d \tilde v
	\]
\end{lemme}

\prf{Si on note $\psi = (\psi^1, \psi^2)$, alors on remarque que
\[
\tilde \varphi_{\tilde u} = (\varphi \circ \psi)_{\tilde u} = \psi_{\tilde u}^1\varphi_u + \psi_{\tilde u}^2 \varphi_v \et \tilde \varphi_{\tilde v} = (\varphi \circ \psi)_{\tilde v} = \psi_{\tilde v}^1\varphi_u + \psi_{\tilde v}^2 \varphi_v
\]
et donc il s'ensuit que
\[
\tilde \varphi_{\tilde u} \times \tilde \varphi_{\tilde v} = \varphi_u \times \varphi_v \, (\psi_{\tilde u}^1 \psi_{\tilde v}^2 - \psi_{\tilde u}^2 \varphi_{\tilde v}^1)
\]
Alors
\[
\norme{\tilde \varphi_{\tilde u} \times \tilde \varphi_{\tilde v}} = \norme{\varphi_u \times \varphi_v} \cdot |D \psi(\tilde u, \tilde v) |
\]
Ce qui permet de conclure
\begin{align*}
	\iint_V f \circ \tilde \varphi \, \norme{\tilde \varphi_{\tilde u} \times \tilde \varphi_{\tilde v} } \, \mathrm d \tilde u \, \mathrm d \tilde v &=   \iint_V f \circ \tilde \varphi \, \norme{(\varphi_u \times \varphi_v)(\psi(\tilde u, \tilde v))} \cdot |D \psi(\tilde u, \tilde v)| \, \mathrm d \tilde u \, \mathrm d \tilde v \\
	&= \iint_U f \circ \varphi \, \norme{\varphi_u \times \varphi_v } \, \mathrm du \, \mathrm dv
\end{align*}
}

Soit $(S, \varphi)$ une surface paramétrisée en $\R^3$, on appelle \emph{orientation} un choix de vecteur $\bm n$ normal et	unitaire en chaque $p \in S$ qui dépend continument de $p$. On peut exprimer un choix de $\bm n$ en terme de paramétrisation
\[
\bm n : = \frac{\varphi_u \times v}{\norme{ \varphi_u \times \varphi_v}} \ou \bm n : = - \frac{\varphi_u \times v}{\norme{ \varphi_u \times \varphi_v}}
\]
On a donc la définition naturelle suivante
\begin{definition}[Flux] \label{def: Flux}
	Soit $(S, \varphi)$, une surface paramétrisée régulière en $\R^3$ munie d'une orientation $\bm n$. Alors si $\bm f \in C^0(S, \R^3)$ est un champ vectoriel continue sur $S$, on appelle l'intégrale
	\[
	\iint_S \bm f \cdot \bm n \, \mathrm ds
	\]
	le \textbf{flux} de $\bm f$ à travers $(S,\bm n)$.
\end{definition}

\lecture{28 septembre}
\begin{theoreme}[Théorème de Kelvin-Stokes] \label{thm: Théorème de Kelvin-Stokes}
	Soit $U \subset \R^2$ un domaine tel que $\overline U$ est compact et $\partial U$ est un lacet simple et régulier muni de l'orientation positive. Soit $\varphi \in C^1(\overline U, \R^3)$ une paramétrisation régulière de la surface $S = \varphi(\overline U)$ avec $\partial S = \varphi(\partial U)$. Ainsi si $\bm f \in C^1(\R^3, \R^3)$, alors on a 
	\[
	\int_{\partial S} \bm f \cdot \mathrm d \bm s = \iint_S \rot \bm f \cdot \bm n \, \mathrm ds
	\]
\end{theoreme}

\prf{On observe que 
	\begin{align*}
		\int_{\partial S} \bm f \cdot \mathrm d \bm s &= \int_0^1 \bm f(\varphi(\gamma(t))) \cdot  \frac{\mathrm d}{\mathrm dt} (\varphi \circ \gamma) \, \mathrm dt \\
		&= \int_0^1 \bm f(\varphi(\gamma(t))) \cdot [ \gamma_1' \varphi_u + \gamma_2' \varphi_v] \, \mathrm dt	\\
		&= \int_0^1 \bm f(\varphi(\gamma(t))) \cdot \gamma_1' \varphi_u + \bm f (\varphi(\gamma(t))) \cdot \gamma_2' \varphi_v \, \mathrm dt
	\end{align*}
	Ainsi on posant le champ vectoriel $\bm g = \binom{g_1}{g_2}$, donné par $g_1 = \bm f(\varphi) \cdot \varphi_u$ et $g_2 = \bm f(\varphi) \cdot \varphi_v$, alors on a 
	\[
	\int_{\partial S} \bm f \cdot \mathrm d \bm s = \int_{\partial U} \bm g \cdot \mathrm d \bm s
	\]
	On applique alors le \autoref{thm: Green-Riemann} pour le champ vectoriel $\bm g$
	\[
	\int_{\partial U} \bm g \cdot \mathrm d \bm s = \iint_U \rot \bm g \, \mathrm du \, \mathrm dv
	\]
	Or on a 
	\[
	\rot \bm g = \partial_u g_2 - \partial_v g_1 = \partial_u[\bm f (\varphi) \cdot \varphi_v] - \partial_v [\bm f(\varphi) \cdot \varphi_u] = \partial_u[\bm f(\varphi)] \cdot \varphi_v - \partial_v [\bm f(\varphi)] \cdot \varphi_u
	\]
	Or par la règle de dérivation des composition on a
	\[
	\partial_u[\bm f(\varphi)] = D \bm f \,\varphi_u \et \partial_v[\bm f(\varphi)] = D \bm f \,\varphi_v
	\]
	et donc
	\[
	\rot \bm g =  (D \bm f \, \varphi_u) \cdot \varphi_v - (D \bm f \, \varphi_v) \cdot \varphi_v = \varphi_u^\top D \bm f^\top \varphi_v - \varphi_v^\top D \bm f^\top \varphi_u = \varphi_u^\top (D \bm f^\top - D \bm f) \varphi_v
	\]
	Or on remarque que
	\[
	\bm u^\top (D \bm f^\top - D \bm f) \bm v = (\bm u \times \bm v) \cdot \rot \bm f
	\]
	Et donc
	\[
	\int_{\partial S} \bm f \cdot \mathrm d \bm s = \int_{\partial U} \bm g \cdot \mathrm d \bm s = \iint_U \rot \bm g \, \mathrm d u \, \mathrm dv = \iint_U \varphi_u \times \varphi_v \cdot \rot \bm f \, \mathrm d u \, \mathrm dv = \iint_U  \bm n \cdot \rot \bm f \, \mathrm ds
	\]
}

\subsection{Potentiel scalaire en $\R^3$}
\begin{proposition}
	Si $U$ est un ensemble simplement connexe et $\bm f \in C^1(U, \R^3)$ avec $\rot \bm f = 0$, alors $\bm f$ dérive d'un potentiel scalaire.
\end{proposition}

\prf{On a une preuve presque analogue à celle en $\R^2$ dans la \autoref{prop: Condition suffisante de dérive d'un potentiel scalaire} que l'on peut résumer par
	\begin{enumerate}
		\item Supposons $U \subset \R^3$ un étoilé, alors par le \autoref{thm: Théorème de Kelvin-Stokes} nous avons pour des triangles bien choisis
		\[
		\iint_\Delta \rot \bm f \cdot \bm n \, \mathrm d \sigma = \int_{\partial \Delta} \bm f \cdot \mathrm d \bm s
		\]
		Car $\rot \bm f = 0$, nous savons que $\int_{\partial \Delta} \bm f \cdot \mathrm d \sigma = 0$ ce qui permet de montrer que $\bm f$ dérive d'un potentiel scalaire.
		\item Ensuite par le \autoref{thm: caractérisation de dérive} il suffit de montrer que $\int_\gamma \bm f \cdot \mathrm d \bm s = 0$ pour tout lacet $\gamma$ dans $U$.
		\item et finalement on procède de manière identique à la preuve en dimension deux pour conclure.
	\end{enumerate}
}	

\section{Théorème de la Divergence / Gauss-Ostrogradsky}
Avant d'introduire le théorème de Gauss-Ostrogradsky on a besoin de définir plusieurs notions au préalable. Ainsi soit $U \subset \R^3$ un domaine borné par une surface $S$ compacte, fermée et régulière $(C^1)$. Alors on a les équivalences suivantes : 
\begin{enumerate}
	\item[$\iff$] $S$ est localement une surface paramétrée régulière de classe $C^1$.
	\item[$\iff$] $\forall p \in S, \, \exists \delta > 0$ tel que $B_\delta (p) \cap S$ admet une paramétrisation $C^1$ régulière.
\end{enumerate}
Par le théorème des fonctions implicites, on peut supposer que $B_\delta (p) \cap S$ est le graphe d'une fonction, soit de $(x,y)$, soit de $(x,z)$ ou soit de $(y,z)$.

\begin{theoreme}[Théorème de Gauss-Ostrogradsky] \label{thm: Gauss-Ostrogradsky}
	Soit $U \subset \R^2$, un domaine borné par $S$, une surface compacte fermé et régulière. Alors, si $\bm n$ est l'orientation de $S$ qui pointe vers l'extérieur et $\bm f \in C^1(\overline U, \R^3)$, alors on a 
	\[
	\iint_S \bm f \cdot \bm n \, \mathrm d \sigma = \iiint_U \dive \bm f \, \mathrm dx \, \mathrm dy \, \mathrm dz
	\]
	où $\dive \bm f = \nabla \cdot \bm f$.
\end{theoreme}

\prf{La preuve est similaire au \autoref{thm: Green-Riemann}, sauf qu'ici on considère 3 types de régions
	\begin{enumerate}
		\item[(I)] \label{def: I}$S\cap V$ est donné par $z = h(x,y)$,
		\item[(II)] \label{def: II}$S\cap V$ est donné par $y = h(x,z)$,
		\item[(III)] \label{def: III}$S\cap V$ est donné par $x = h(y,z)$.
	\end{enumerate}
	Puis comme $S$ est compact, on peut extraire un sous-recouvrement fini,
	\[
	S \subset \bigcup_{i=1}^L V_j
	\]
	Et donc par la \autoref{prop: Partition de l'unité} on a 
	\[
	\bm f = \sum_{j=1}^{L+1} \varphi_j \bm f 
	\]
	où $\varphi \in C^\infty_0(V_j)$, $\varphi_j \geqslant 0$ et $\sum_{j=1}^{L+1} \varphi_j (x) = 1$. Et donc par linéarité de l'intégrale, on a 
	\[
	\iint_S \bm f \cdot \bm n \, \mathrm d \sigma = \sum_{j=1}^{L+1} \iint_S \varphi_j \bm f \cdot \bm n \, \mathrm d \sigma
	\]
	Ensuite on montre que
	\[
	\iint_S \varphi_j \bm f \cdot n \, \mathrm d \sigma = \iiint_U \dive \varphi_j \bm f \, \mathrm dV
	\]
	Ce qui ce fait naturellement par calcul explicite car
	\[
	\iint_S \varphi_j \bm f \cdot n \, \mathrm d \sigma = \iint_{S \cap V_j} \varphi_j \bm f \cdot n \, \mathrm d \sigma 
	\]
	et on sait que $S \cap V_j$ est soit de type \hyperref[def: I]{(I)}, soit de type \hyperref[def: II]{(II)} ou soit de type \hyperref[def: III]{(III)}. Ainsi on conclut bien que,
	\[
	\iint_S \bm f \cdot \bm n \, \mathrm d \sigma = \iiint_U \dive \bm f \, \mathrm dx \, \mathrm dy \, \mathrm dz
	\]
}
