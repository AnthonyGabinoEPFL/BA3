\section{Topologie et base}

\begin{definition}[Espace topologique] \label{def: Espace topologique}
	Soit $X$ un ensemble, une \textbf{topologie} sur $X$ est un sous-ensemble $\T \subset \mathcal P(X)$ tel que
	\begin{enumerate}
		\item[(T1)] \label{def: T1} $\emptyset, X \in \T$
		\item[(T2)] \label{def: T2} $\forall \mathcal A \subset \T, \quad \bigcup_{U \in \mathcal A} U \in \T$
		\item[(T3)] \label{def: T3} $\forall U_1, \ldots, U_n \in \T, \quad \bigcap_{i=1}^n U_i \in \T$
	\end{enumerate}
	le couple $(X, \T)$ est alors appelé un \textbf{espace topologique} et les éléments de $\T$ sont appelés des \textbf{ouverts} de $(X, \T)$.
\end{definition}

\begin{exemple}
	Tout ensemble $X$ possède les topologies suivantes
	\begin{enumerate}
		\item La \textbf{topologie grossière} définit par
		\[
		\Tgr = \{ \emptyset, X\}
		\]
		\item La \textbf{topologie discrète} définit par
		\[
		\Tdis = \mathcal P(X)
		\]
		\
	
	 \item Mais il existe un topologie moins intuitive qu'on appelle \textbf{topologie du complément fini} que l'on définit par
	\[
	\Tcomp = \{ U \in \mathcal P(X) \mid \# U^c < \infty\} \cup \{\emptyset\}
	\]
	En effet on a 
	\begin{enumerate}
		\item[\TU : ] Par définition $\emptyset \in \Tcomp$. De plus $X \in \Tcomp$ car $\# X^c = \# \emptyset = 0 < \infty$
		
		\item[\TD : ] De plus pour $ \mathcal A \subset \Tcomp$ alors
		\[
		\bigg(\bigcup_{U \in \mathcal A} U \bigg)^c = \bigcap_{U \in \mathcal A} U^c
		\]
		donc
		\[
		\bigg(\bigcup_{U \in \mathcal A} U \bigg)^c = \#\bigcap_{U \in \mathcal A} U^c \leqslant \# V^c < \infty
		\]
		où $V \in \mathcal A$ quelconque. 
		
		\item[\TT: ] Et finalement pour $U_1, \ldots, U_n \in \Tcomp$ alors
		\[
		\bigg(\bigcap_{i=1}^n U_i \bigg)^c = \bigcup_{i=1}^n U_i^c
		\]
		ainsi
		\[
		\#\bigg(\bigcap_{i=1}^n U_i \bigg)^c = \#\bigcup_{i=1}^n U_i^c \leqslant \sum_{i=1}^n \# U_i^c < \infty
		\]
		Ainsi $(X, \Tcomp)$ est bien une topologie.
		
	\end{enumerate}
\end{enumerate}

\end{exemple}

\begin{definition}
	Soit $\T, \T'$ des topologies sir $X$. Si $\T \subset \T'$, alors on dit que $\T'$ est plus \textbf{fine} que $\T$. Mais aucune des topologies n'est incluses dans l'autre alors on dira qu'elles sont \textbf{incomparable}.
\end{definition}

\subsection{Les bases topologiques}

\begin{definition}[Base topologique] \label{def: Base topologique}
	Soit $X$ un ensemble. Une \textbf{base de topologie} est un sous-ensemble $\mathcal B \subset \mathcal P(X)$ qui vérifie
	\begin{enumerate}
		\item[(B1)] \label{def: B1} On a $X = \bigcup_{B \in \mathcal B} B$,
		\item[(B2)] \label{def: B2} Pour tout $B_1, B_2 \in \mathcal B$, on a que pour tout $x \in B_1 \cap B_2$, il existe un $B_3 \in \mathcal B$ tel que 
		\[
		x \in B_3 \subset B_1 \cap B_2
		\]
		Il est facile de voir qu'une topologie $\T$ sur $X$ est aussi une base de topologie. Par contre la réciproque est généralement fausse.
	\end{enumerate}
\end{definition}

\begin{definition} \label{def: Topologie engendrée}
	Soit $\B$ une base de topologie sur $X$, alors on définit la \textbf{topologie engendrée} par $\B$ comme
	\[
	\T_\B = \left\{ \bigcup_{\mathcal V \in \mathcal A} V \mid A \subset \B \right\}
	\]
\end{definition}

\begin{lemme}
	$\T_\B$ est bien une topologie sur $X$.
\end{lemme}

\prf{On doit vérifié que $\TB$ vérifie les trois conditions requises pour être une topologie.
	\begin{enumerate}
		\item[\TU :] Par définition d'une base on a en particulier pour $\mathcal A = \mathcal B$
		\[
		X = \bigcup_{V \in \mathcal B} V \in \T_\B
		\]
		et de même pour $\mathcal A = \emptyset$ on a bien $\emptyset \in \T_\B$.
		
		\item[\TD :] Soit $\{U_\alpha \mid \alpha \in I\} \subset \T_\B$, alors par définition de $\T_\B$ pour tout $\alpha \in I$ il existe $\mathcal A_\alpha \subset \B$ tel que $U_\alpha = \bigcup_{V \in \mathcal A_\alpha} V$, d'où
		\[
		\bigcup_{\alpha \in I} U_\alpha = \bigcup_{\alpha \in I} \bigcup_{V \in \mathcal A_\alpha} V = \bigcup_{V \subset \bigcup_{\alpha \in I} \mathcal A_\alpha} V \in \T_\B
		\]
		
		\item[\TT :] Finalement, soit $\{U_1, \ldots, U_n\} \subset \T_\B$, alors par définition de $\T_\B$ pour tout $i$ il existe $\mathcal A_i \subset \B$ tel que $U_i = \bigcup_{V \in \mathcal A_i} V$ ainsi
		\[
		\bigcap_{i=1}^n U_i = \bigcap_{i=1}^n \bigcup_{V \in \mathcal A_i} V
		\]
		Pour conclure, il suffit de considérer le cas $n = 2$ car on peut ensuite conclure par récurrence, car si on sait que
		\[
		U_1, U_2 \in \T_\B \implies U_1 \cap U_2 \in \T_\B
		\]
		alors
		\[
		U_1 \cap U_2 \cap U_3 = (U_1 \cap U_2) \cap U_3 \in \T_\B
		\]
		Or
		\[
		U_1 \cap U_2 = \bigcup_{V \in A_1} V \cap \bigcup_{V' \in A_2} V' = \bigcup_{V \in \mathcal A_1, \, V \in \mathcal A_2} (V \cap V')
		\]
		or par définition de la base on a
		\[
		V \cap V' = \bigcap_{x \in V \cap V'} V_x
		\]
		d'où 
		\[
		U_1 \cap U_2 = \bigcup_{V \in \mathcal A_1, \, V \in \mathcal A_2}\bigcap_{x \in V \cap V'} V_x \in \T_\B
		\]
	\end{enumerate}
	Ainsi la topologie engendrée $\TB$ est bien une topologie.
}

\begin{theoreme}[Caractérisation de la topologie engendrée]
	Soit $\B$ une base de topologie sur un ensemble $X$, alors
	\[
	\T_\B = \{U \in \mathcal P(X) \mid \forall x \in U, \, \exists B \in \B, \, x \in B \subset U\}
	\]
\end{theoreme}

\prf{Soit $U = \bigcup_{V\in \mathcal A} V \in \T_\B$, alors pour tout $x \in U$ il existe $v \in \mathcal A \subset B$ tel que $x \in V$. Ainsi $\T_\B \subset \T'$.
	
	Réciproquement, soit $U \in \T'$, alors pour tout $x \in U$, il existe $V_x$ tel que $x \in V_x \subset U$ et donc par conséquent $U = \bigcup_{x \in U} V_x \in \T_\B$.
}

\begin{lemme}
	Soit $(X,d)$ un espace métrique, on définit les boules ouvertes
	\[
	B_d(x,r) = \{y \in X \mid d(x,y) < r\}
	\]
	pour tout $x \in X$ et $r > 0$. Alors
	\[
	\mathcal B_d = \{ B_d(x,r) \mid x \in X, ~ r > 0\}
	\]
	est une base de topologie sur $X$.
\end{lemme}

\prf{
\begin{enumerate}
	\item[\BU :] On sait que pour tout $x \in X$, on a $x \in B_d(x,r)$ pour $r > 0$, ce qui implique que
	\[
	X = \bigcup_{x \in X} B_d(x,r)
	\]
	\item[\BD :] De plus soit $B_d(x_1, r_1)$ et $B_d(x_2, r_2)$ deux boules, alors si on considère
	\[
	r = \min\{ r_1 - d(x, x_1), r_2 - d(x, x_2)\}
	\]
	alors $B_d(x,r) \subset B_d(x_1, r_1) \cap B_d(x_2, r_2)$ car si $y \in B_d(x,r)$ alors $d(x,y) < r$ et donc
	\[
	d(x_i, y) \leqslant d(x_1, x) + d(x,y)  < d(x_i, x) + (r_i - d(x_i, x)) < r
	\]
	pour $i =1,2$. 
\end{enumerate}
Ainsi on pose $\mathcal T_d = \mathcal T_{\mathcal B_d}$ la \textbf{topologie métrique} sur $X$ induite par la métrique $d$. Par la suite si $X = \Rn$ et $d = d_{euc}$, on écrit $\T_{st} = \T_{d_{euc}}$ la \textbf{topologie standard}.
}

\begin{lemme}[Comparaison de bases] \label{lem: Comparaison de bases}
	Soit $X$ un ensemble, soient $\B_1, \B_2$ des bases topologies sur $X$. Alors
	\[
	\T_{\B_1} \subset \T_{\B_2} \iff \forall B_1 \in \B_1, \, \forall x \in B_1, \, \exists B_2 \in \mathcal B_2, \quad x \in B_2 \subset B_1
	\]
\end{lemme}

\prf{
	\begin{enumerate}
		\item[$\implies$ :] Puisque $\mathcal B_1 \subset \T_{\B_1} \subset \T_{\B_2}$, alors pour tout $B_1 \in \B_1$, on a que pour tout $x \in B_1$, il existe $B_2 \in \B_2$ tel que $x \in B_2 \subset B_1$.
		
		\item[$\impliedby$ :] Réciproquement, alors par hypothèse on a $B_1 \in \T_{\B_2}$ pour tout $B_1 \in \B_1$. Autrement dit, $\B_1 \subset \T_{\B_2}$. Soit $U \in \T_{\B_1}$, alors par la \autoref{def: Topologie engendrée}, il existe une collection $\{B_i\} \subset \B_1$ telle que $U = \bigcup_{i \in I} B_i$. Or on sait que $B_i \in \T_{\B_2}$ pour tout $i \in I$. Puisque $\T_{B_2}$ est une topologie, alors
		\[
		U = \bigcup_{i \in I} B_i \in \T_{\B_2}
		\]
	\end{enumerate}
}

\begin{lemme}[Test de base] \label{lem: Test de base}
Soit $(X, \T)$ un espace topologique, soit $\mathcal C \subset \T$, alors $\mathcal C$ est une base de topologie et $\T_{\mathcal C} = \T$ si et seulement si
\[
\forall U \in \T, \, \forall x \in U, \, \exists C \in \mathcal C, \quad x \in C \subset U
\]
\end{lemme}

\prf{
	\begin{enumerate}
		\item[$\implies$ :] Soit $\mathcal C$ une base de topologie, et $\T_{\mathcal C} = \T = \T_\T$, et donc en particulier $\T_\T \subset \T_\C$, d'où par le \autoref{lem: Comparaison de bases},
		\[
		\forall U \in \T, \, \forall x \in U, \, \exists C \in \C, \quad x \in C \subset U
		\]
		
		\item[$\impliedby$ :] On doit montrer que $\C$ engendre $\T$, il faut donc vérifier les deux axiomes
		\begin{enumerate}
			\item[\BU :] par hypothèse sur $\C$, en particulier pour $X \in \T$, 
			\[
			\forall x \in X, \, \exists C \in \C, \quad x \in C \subset X
			\]
			d'où le recouvrement.
			\item[\BD :] Soit $C_1, C_2 \in \C$ et soit $x \in C_1 \cap C_2$, mais comme $C_1$ et $C_2$ sont des ouverts, alors $C_1 \cap C_2$ est aussi un ouvert, et donc par hypothèse on a pour $x \in C_1 \cap C_2$, et on en déduit qu'il existe $C_3 \in \C$ tel que
			\[
			x \in C_3 \subset C_1 \cap C_2
			\]
			Il nous à montrer que $\T \subset \T_\C$. Les hypothèses sur $\C$ implique, par le \autoref{lem: Comparaison de bases} que $\T = \T_\T \subset \T_\C$.
		\end{enumerate}
	\end{enumerate}
}

\section{Les sous-bases de la topologie}

\begin{definition}
	Soit $X$ un ensemble. Une \textbf{sous-base de topologie} $\SB$ est un sous-ensemble de $\mathcal P(X)$ tel que
	\begin{enumerate}
		\item[(S1)] \label{def: S1} $X = \bigcup_{S \in \SB} S$
	\end{enumerate}	
\end{definition}

\begin{lemme}
	Étant donné une sous-base de topologie sur $X$, alors la collection
	\[
	\B_\SB = \{ S_1 \cap \ldots  \cap S_n \mid S_i \in \SB, \, n \geqslant 1\}
	\]
	est une base de topologie. 
\end{lemme}

\prf{On doit montrer que $\B_\SB$ vérifie bien les axiomes d'une base
	\begin{enumerate}
		\item[\BU :] Comme $\SB$ est une sous-base, elle vérifie \SU = \BU. Comme $\SB \subset \BS$, on peut conclure que $\BS$ vérigie \BU.
		
		\item[\BD :] Soient $B = \bigcap_{i=1}^m S_i$ et $\tilde B = \bigcap_{j=1}^n \tilde S_j$ deux éléments de $\BS$. Alors leur intersection
		\[
		B \cap \tilde B = \left(\bigcap_{i=1}^m S_i\right) \cap \left(\bigcap_{j=1}^n \tilde S_j\right) \in \BS
		\]
	\end{enumerate}
	Ainsi $\BS$ est bien une base.
}

\section{Autour de la notion de fermé}

\begin{definition}[Ensemble fermé] \label{def: Ensemble fermé}
	Soit $(X, \T)$ un espace topologique. Soit $A \subset X$, alors $A$ est dit \textbf{fermé par rapport à la topologie} $\T$ si $A^c \in \T$.
\end{definition}

\begin{exemple}
	On a
	\begin{enumerate}
	 \item Par si on considère la topologie grossière $(X, \T_{gr})$ alors les seuls fermés sont $X$ et $\emptyset$ donc $\emptyset$ et $X$ sont à la fois ouvert et fermé. 
	
	Mais plus généralement, pour tout espace topologique $\T$ sur $X$ alors $\emptyset$ et $X$ sont ouverts et fermés.
	
	 \item Maintenant si on considère la topologie discrète $\T_{disc}$ alors tout $A \subset X$ est fermé et ouvert.
	\end{enumerate}
\end{exemple}

\begin{proposition}
	Soit $(X, \T)$ un espace topologique, alors
	\begin{enumerate}
		\item $\emptyset, \, X$ sont fermés par rapport à $\T$
		\item Toute intersection de fermés est fermé.
		\item Toute réunion finie de fermés est fermé.
	\end{enumerate}
\end{proposition}

\prf{
Soit $\{A_\alpha \mid \alpha \in I\}$ une collection de fermés de $X$ par rapport à $\T$, alors
\[
\left(\bigcap_{\alpha \in I} A_\alpha\right)^c = \bigcup_{\alpha \in I} A_\alpha^c \in \T
\]
donc $\bigcap A_\alpha$ est bien fermé.

De plus, soient $A_1, \ldots, A_n$ des fermés par rapport à $\T$, alors
\[
\left(\bigcup_{i = 1}^n A_i\right)^c = \bigcap_{i=1}^n A_i^c \in \T
\]
Donc $\bigcup A_i$ est bien fermé
}

\begin{definition}
	Soit $(X, \T)$ un espace topologique, et $A \subset X$, alors on définit
	\begin{enumerate}
		\item L'\textbf{intérieur} de $A$ comme
		\[
		\Int A := \mathring A := \bigcup_{\{U \mid U \in \T, \, U \subset A\}} U
		\]
		\item L'\textbf{adhérence} de $A$ comme
		\[
		\overline A = \bigcap_{ \{C \mid C \text{ fermé}, \, A \subset C\}} C 
		\]
	\end{enumerate}
\end{definition}

\begin{remarque}
	On a toujours les inclusions
	\[
	\Int A \subset A \subset \overline A 
	\]
	avec des égalités si et seulement $A$ est respectivement ouvert ou fermé.
\end{remarque}


\begin{theoreme}
	Soient $(X, \T)$ un espace topologique et $\B$ une base de $\T$. Alors pour tout $A \subset X$,
	\[
	\overline A = \{x \in X \mid \forall B \in \B, \, x \in B, \quad A \cap B \neq \emptyset\}
	\]
\end{theoreme}

\prf{Soit $x \in \overline A$, alors pour tout fermé $C$ tel que $A \subset C$ alors $x \in C$. De plus soit $V \in \B$ tel que $x \in \B$, si on suppose que $A \cap B = \emptyset$, alors on aurait
	\[
	(A \cap B)^c = \emptyset^c = X
	\]
	or $(A \cap B)^c = A^c \cup B^c$. Mais comme $A \subset X$ et $A \cap A^c = \emptyset$, on en déduit que $A \subset B^c$ et donc on devrait avoir $x \in B^c$ ce qui contradictoire. Par conséquent, on a bien $A \cap B \neq \emptyset$.
}

\section{Les applications continues}

\subsection{Définitions et propriétés}

\begin{definition}[Continuité globale] \label{def: Continuité globale}
	Soit $(X, \T)$ et $(X', \T')$ des espaces topologiques. Une application
	\[
	f : X \to X
	\]
	est \textbf{continue par rapport} à $\T$ et $\T'$ si pour tout $U' \in \T$, on a $f^{-1}(U') \in \T'$. On écrira alors
	\[
	\text{"$f :(X, \T) \to (X', \T')$ est continue"}
	\]
\end{definition}
Tout comme la continuité standard, il est possible d'énoncer une version locale de cette définition

\begin{definition}[Continuité locale] \label{def: Continuité locale}
	Soit $(X, \T)$ et $(X', \T')$ des espaces topologiques. Une application $f : X \to X$ est \textbf{continue en $x$ par rapport} à $\T$ et $\T'$ si pour tout $U	' \in \T'$ tel que $f(x) \in U'$, il existe $U \in \T$ tel que $x \in U \subset f^{-1}(U')$.
\end{definition}

\begin{lemme}
	Sous les conditions des \autoref{def: Continuité globale} et \autoref{def: Continuité locale}, une application $f : (X, \T) \to (X', \T')$ est continue si et seulement si $f$ est continue en tout $x \in X$.
\end{lemme}

\prf{
	\begin{enumerate}
		\item[$"\implies :"$] Soient $x \in X$ et $U' \in \T'$ tel que $f(x) \in U'$. Comme $f : (X, \T) \to (X', \T')$ est continue par hypothèse, on a $f^{-1} \in \T$, et donc on peut poser $U = f^{-1}(U')$ et on aura $x \in U = f^{-1}(U')$, ce qui montre la continuité locale.
		
		\item[$"\impliedby :"$] Soit $U' \in \T'$, alors pour $x \in f^{-1}(U')$ on a par hypothèse, il existe $U_x \in \T$ tel que $x \in U_x \subset f^{-1}(U')$. Par conséquent, $f^{-1}(U') = \bigcup_{x \in f^{-1}(U')} U_x \in \T$ par \TD.
	\end{enumerate}
}

\subsection{Caractérisation de la continuité par les sous-bases}
\begin{lemme} \label{lem: Caractérisation de la continuité par les sous-bases}
	Soient $(X, \T)$ et $(X', \T')$ des espaces topologiques. Soit $\mathcal S'$ une sous-bases de $\T'$. Alors une application $f : X \to X'$ est continue par rapport à $\T$ et $\T'$ si et seulement si $f^{-1}(S') \in \T$ pour tout $S' \in \mathcal S'$.
\end{lemme}

\prf{
	\begin{enumerate}
		\item[$"\implies :"$] On a toujours $\mathcal S' \subset \mathcal B_{\mathcal S'} \subset \T_{\mathcal B_{\mathcal S'}}$ donc si l'application $f$ est continue, on a nécessairement $f^{-1}(S') \in \T$ pour tout $S' \in \mathcal S'$.
		
		\item[$"\impliedby :"$] Soit $U' \in \T'$, on se rappelle que $\T' = \T_{\mathcal B_{\mathcal S'}}$ et que
		\[
		\mathcal B_{\mathcal S'} = \Bigg\{\bigcap_{i=1}^n S_i' \in \mathcal S' \mid S_i' \in \mathcal S'\Bigg\}
		\]
		et 
		\[
		\T_{\mathcal B_{\mathcal S'}} = \Bigg\{ \bigcup_{i \in I} B_i' \mid B_i \in \mathcal B_{\mathcal S'}\Bigg\}
		\]
		Comme $U' \in \T' = \T_{\mathcal B_{\mathcal S}}$ il existe une famille $\{B_i\} \subset \mathcal B$ telle que
		\[
		U' = \bigcup_{i \in I} B_i
		\]
		et pour chaque $i \in I$, il existe une collection $\{S_{i_1}', \ldots, S_{i_n}'\} \subset \mathcal S'$ telle que
		\[
		B_i = \bigcap_{j=1}^n S_{i_j}'
		\]
		Ainsi
		\[
		f^{-1}(U') = \bigcup_{i \in I} f^{-1}(B_i) = \bigcup_{i \in I} \bigcap_{j=1}^n f^{-1} (S_{i_j}') \in \T
		\]
		par hypothèse, on a donc montré que $f^{-1}(U') \in \T$ pour tout $U' \in \T'$.
	\end{enumerate}
}

\begin{lemme}
	Soient $(X, d)$ et $(X', d')$ des espaces métriques. Une application $f : (X, d) \to (X', d')$ est continue dans le sens de la métrique si et seulement si l'application $f : (X, \T_d) \to (X, \T_{d'})$ est continue dans le sens topologique.
\end{lemme}

\subsection{Caractérisation de la continuité par les fermés}
\begin{theoreme} \label{thm: Caractérisation par les fermés}
	Soient $(X, \T)$ et $(X', \T')$ des espaces topologiques. Soit $f : X \to X'$ une application. Alors les affirmations suivantes sont équivalentes : 
	\begin{enumerate}
		\item $f : (X, \T) \to (X', \T')$ est continue,
		\item $f(\overline A) \subset \overline{f(A)}$ pour tout $A \subset X$,
		\item Pour tout fermé $C' \subset X'$ par rapport à $\T'$, sa préimage $f^{-1} (C')$ est fermé par rapport à $\T$.
	\end{enumerate}
\end{theoreme}
\prf{On va raisonner cycliquement, 
	\begin{enumerate}
		\item[$1. \implies 2.$ : ] Supposons que $f$ soient continue, soit $y \in f(\overline A)$. Alors il existe $x \in \overline A$ tel que $y = f(x)$. Par caractérisation de l'adhérence, pour tout $U \in \T$ tel que $x \in U$, on a
		\begin{equation} \label{eq: adhérence}
		U \cap A \neq \emptyset
		\end{equation}
		Or par continuité de $f$, si $U' \in \T'$, alors $f^{-1}(U') \in \T$. De plus comme $f(x) = y$, on sait que $x \in f^{-1} (u')$ pour un certain $U' \in \T$. Mais par \eqref{eq: adhérence}, on a $A \cap f^{-1}(U') \neq \emptyset$, et donc en appliquant $f$ on a
		\[
		\emptyset \neq f(A \cap f^{-1}(U')) = f(A) \cap f(f^{-1}(U')) \subset f(A) \cap U'
		\]
		d'où $y \in \overline{f(A)}$. Ainsi on a montré $f(\overline A) \subset \overline{f(A)}$.
		
		\item[$2. \implies 3.$ : ] Soit $A' \subset X'$ un sous-ensemble fermé par rapport à $\T'$. Alors
		\[
		f^{-1}(A') \subset \overline{f^{-1}(A')} \subset f^{-1} \Big(f \Big( \overline{f^{-1}(A')}\Big)\Big) \subset f^{-1} \Bigg(\underbrace{ \overline{f(f^{-1}(A'))}}_{\subset A'} \Bigg) \subset f^{-1}(\overline{A'}) = f^{-1}(A')
		\]
		Donc $f^{-1}(A') \subset \overline{f^{-1}(A')}$, donc $f^{-1}(A')$ est fermé par rapport à $\T$.
		
		\item[$3. \implies 1.$ : ] Soit $U'$ un ouvert. On pose $C' =  X' \backslash U'$, il donc fermé par hypothèse par rapport à $\T'$. De plus on sait que $f^{-1}(C')$ est fermé par rapport à $\T$, et donc
		\[
		f^{-1}(C') = f^{-1}(X' \backslash U') = X \backslash f^{-1}(U') \implies f^{-1}(U') =  X \backslash f^{-1}(C')
		\]
		dont $f^{-1}(U') \in \T$ puisque $f^{-1}(C')$ est fermé.
	\end{enumerate}
	Ainsi les 3 affirmations sont bien équivalentes.
}

\begin{definition}
	Soit $(X, \T)$ un espace topologique et $Y \subset X$  une partie de $X$. Alors on définit la \textbf{topologie induite} $\T_Y$ sur $Y$ comme
	\[
	\T_Y = \{ U \cap Y \mid U \in \T\}
	\]
\end{definition}
\begin{lemme}
	Soit $f(X, \T) \to (X', \T')$ une application continue entre espaces topologiques. Alors pour des sous-ensembles $Y \subset X$ et $Z \subset X'$ tel que $\Ima f \subset Z$, on a
	\begin{enumerate}
		\item La restriction $f_{\mid_Y} : (Y, \T_Y) \to (X', \T')$
		\item La co-restriction $f_{\mid^Z} : (X, \T) \to (Z, \T_{Z}')$
	\end{enumerate}
	sont tout deux continues.
\end{lemme}

	\hfill

			& {(Z, \mathcal T_Z')}
			\arrow["f", from=1-1, to=1-3]
			\arrow["{f_{\mid^Z}}"', from=1-1, to=2-2]

\prf{
	\begin{enumerate}
		\item On sait que $f_{\mid_Y} = f \circ \imath$, mais puisque $f$ et $\imath$ sont tout deux continues, leur composée l'est aussi. En effet,
		\[
		\forall U \in \T, \quad \imath^{-1}(U) = U \cap Y \in \T_Y
		\]
		dont $\imath$ est bien continue.
		\item On sait que $\mathcal T_Z' = \{U' \cap Z \mid U' \in \T'\}$. Comme $f$ est continue, on a toujours $f^{-1} (U') \in \T$. Mais comme $\Ima f \subset Z$, on a $f^{-1}(U'\cap Z) = f^{-1}(U')$. Ainsi
		\[
		f_{\mid^Z}^{-1} (U'\cap Z) = f^{-1} (U'\cap Z) = f^{-1}(U') \in \T
		\]
		ce qui montre bien que le co-restriction $f_{\mid^Z}$ est continue.		
	\end{enumerate}

\begin{theoreme}[Théorème de recollement]
	Soient $(X, \T)$ et $(X', \T)$ des espaces topologiques et $f : X \to X'$ une application. S'il existe une famille d'ouvert $\{U_i \mid i \in I\} \subset \T$ qui recouvre $X$ et que chaque restriction
	\[
	f_{\mid_{U_i}} : (U_i, \T_{U_i}) \to (X', \T')
	\]
	est continue pour tout $i \in I$, alors $f : (X, \T) \to (X', \T')$ est continue.	
\end{theoreme}
\prf{Comme les $U_i$ forment un recouvrement, on a pour $U' \in \T'$
\[
f^{-1}(U') = \bigcup_{i \in I} f^{-1}_{\mid_{U_i}} (U')
\]
or par hypothèse de continuité, pour tout $i \in I, \quad f^{-1}_{\mid_{U_i}}(U') \in \T_{U_i}$. Donc il existe $V_i \in \T$ tel que $f^{-1}_{\mid_{U_i}}(U') = V_i \cap U_i$. Ainsi on conclut que
\[
f^{-1} (U')= \bigcup_{i \in I} f^{-1}_{\mid_{U_i}} (U') = \bigcup_{i \in I} (V_i \cap U_i) \in \T
\]
ce qui montre bien que $f$ est continue.
}

\begin{proposition}[Théorème de recollement avec des fermés]
	Soient $(X, \T)$ et $(X', \T)$ des espaces topologiques et $f : X \to X'$ une application. S'il existe une famille de fermé $\{C_1, \ldots, C_n\}$ qui recouvre $X$ et que chaque restriction
	\[
	f_{\mid_{C_i}} : (C_i, \T_{C_i}) \to (X', \T')
	\]
	est continue pour tout $i \in \entiers$, alors $f : (X, \T) \to (X', \T')$ est continue.	
\end{proposition}

\prf{On observe que
	\[
	f^{-1}(C')  = \bigcup_{i=1}^n f^{-1}_{\mid_{C_i}}(C')
	\]
	est fermé puisque $f^{-1}_{\mid_{C_i}}(C')$ le sont de par le \autoref{thm: Caractérisation par les fermés}, et donc par le même \autoref{thm: Caractérisation par les fermés} on conclut bien que $f$ est continue.
}

\section{La compacité}

\begin{definition}
	Soit $(X, \T)$ un espace topologique et $Y \subset X$, alors on dit que $(Y, \T_Y)$ est \textbf{compact} si et seulement si
	\[
	\forall \mathcal U \subset \T, ~ Y \subset \bigcup_{U \in \mathcal U}, ~\exists U_1, \ldots, U_n \in \mathcal U, \quad Y \subset \bigcup_{i=1}^n U_i
	\]
\end{definition}

\begin{theoreme}
	Soient $(X, \T)$ et $(X', \T')$ des espaces topologiques et $Y \subset X$. Si $(X, \T)$ est compact alors
	\begin{enumerate}
		\item Si $Y$ est fermé par rapport à $\T$, alors $(Y, \T_Y)$ est aussi compact,
		\item Si $f : (X, \T) \to (X', \T')$ est continue, alors $(\Ima f, \T_{\Ima f})$ est aussi compact.
		\item 
	\end{enumerate}
\end{theoreme}	

\prf{On a dans l'ordre
\begin{enumerate}
	\item Soit $(X, \T)$ un espace topologique compact, et soit $Y \subset X$ un fermé. Puisque $Y$ est fermé, alors $Y^c \in \T$, par ailleurs
	\[
	X = Y \backslash Y^c = \bigcup_{U \in \mathcal U} U \cup Y^c
	\]
	dont $\mathcal U \cup Y^c$ est un recouvrement d'ouvert de $X$, ainsi par compacité de $(X, \T)$, il existe $U_1, \ldots, U_n \in \mathcal U$ tel que
	\[
	X = \bigcup_{i=1}^n U_i \cup Y^c
	\]
	Puisque $Y \subset X$ et $Y \cap Y^c = \emptyset$, on sait alors que $Y \subset \bigcup_{i=1}^n U_i$, d'où la compacité de $(Y, \T_Y)$.
	
	\item Soit $f :  (X, \T) \to (X', \T')$ une application continue où $(X, \T)$ un compact. Ainsi soit $\mathcal U' \subset \T'$ tel que 
	\[
	\Ima f \subset \bigcup_{U' \in \mathcal U'} U'
	\]
	On pose $\mathcal U = \{ f^{-1}(U') \mid U' \in \mathcal U'\}$, puisque $f$ est continue, et $U' \in \T'$, on sait que $\mathcal U \subset \T$. Par ailleurs
	\[
	X = \bigcup_{U \in \mathcal U} U
	\]
	et donc par compacité de $(X, \T)$, il existe $U_1 = f^{-1} (U_1'), \ldots, U_n = f^{U_n'}$ tel que $X = \bigcup_{i=1}^n U_i$ donc comme $f(f^{-1}(U_i')) \subset U_i'$ on obtient bien que
	\[
	\Ima f = \bigcup_{i=1}^n f(U_i) \subset \bigcup_{i=1}^n U_i'
	\]
	Donc $(\Ima f, \T_{\Ima f})$ est bien compact
	\end{enumerate}
}

\begin{definition}[Propriété d'intersection finie] \label{def: PIF}
	Soient $X$ un ensemble et $\mathcal C \subset \mathcal P(X)$. La collection $\mathcal C$ vérifie la \textbf{propriété d'intersection finie (PIF)} si pour tout $\{C_1, \ldots, C_n \} \subset \mathcal C$ et pour tout $n \in \N$, on a 
	\[
	\bigcap_{i=1}^n C_i \neq \emptyset
	\]
\end{definition}

\begin{theoreme}
	Soit $(X, \T)$ un espace topologique, alors $(X, \T)$ est compact si et seulement si, si $\mathcal C \subset \mathcal P(X)$ vérifie la PIF et $C$ fermée par rapport à $\T$ pour tout $C \in \mathcal C$, alors
	\[
	\bigcap_{C \in \mathcal C} C \neq \emptyset
	\]
\end{theoreme}

\prf{\begin{enumerate}
		\item[$"\implies :"$] Soit $(X, \T)$ compact, et soit $\mathcal C$ une collection de fermés  qui vérifie la PIF, supposons par l'absurde que
		\[
		\bigcap_{C \in \mathcal C} C = \emptyset \implies X = \left(\bigcap_{C \in \mathcal C} C\right)^c = \bigcup_{C \in \mathcal C} C^c
		\]
		donc $\{ C^c \mid C \in \mathcal C\}$ est un recouvrement ouvert de $(X, \T)$. Et donc par compacité de $(X, \T)$, il existe $C_1, \ldots, C_n \in \mathcal C$ tel que
		\[
		X = \bigcup_{i=1}^n C_i^c
		\]
		d'où
		\[
		\emptyset = X^c = \left(\bigcup_{i=1}^n C_i^c\right)^c = \bigcap_{i=1}^n C_i
		\]
		ce qui contredit la PIF.
		
		\item[$"\impliedby :"$] Soit $\mathcal U \subset \T$ tel que $X = \bigcup_{U \in \mathcal U} U$, alors on considère la collection de fermés
		\[
		\mathcal C = \{ U^c \mid U \in \mathcal U\}
		\]
		alors on remarque que
		\[
		\bigcap_{C \in \mathcal C} C = \bigcap_{U \in \mathcal U} U^c = \left(\bigcup_{U \in \mathcal U} U \right)^c = \emptyset
		\]
		Par conséquent, la collection $\mathcal C$ doit vérifier la PIF, donc il existes $U_1, \ldots, U_n$ tel que $\bigcap_{i=1}^n U_i^c = \emptyset$. D'où
		\[
		X = \emptyset^c = \left(\bigcap_{i=1}^n U_i^c\right)^c = \bigcup_{i=1}^n U_i
		\]
		donc $(X, \T)$ est bien un compact.
	\end{enumerate}
}

\section{Produit finis d'espaces topologiques}

\begin{definition}
	Soient $(X_1, \T_1)$ et $(X_2, \T_2)$ des espaces topologiques. La \textbf{topologie produit} sur $X_1 \times X_2$ est la plus petite topologie $\T$ telle que les projections
	\[
	\pr_i : (X_1 \times X_2, \T) \to (X_i, \T_i)
	\]
	soient continues, $\forall i = 1,2$. Et on la note $\T_1 * \T_2$.
\end{definition}


\begin{lemme} \label{lem: Sous base du l'espace produit}
	Soient $(X_1, \T_1)$ et $(X_2, \T_2)$ des espaces topologiques. Alors $\T_1 * \T_2$ est la topologie engendré par la sous-base
	\[
	\mathcal S_* = \{ U_1 \times X_2 \mid U_1 \in \T_1\} \cup \{X_1 \times U_2 \mid U_2 \in \T_2\}
	\]
\end{lemme}

\prf{Puisque $X_i \in \T_i$ pour $i=1,2$, alors $X_1 \times X_2 \in \mathcal S_*$, ainsi on a bien une sous-base.
	
	De plus on sait que les projections 
	\[
	\pr_i : (X_1 \times X_2, \T) \to (X_i, \T_i)
	\]
	sont continues si et seulement si
	\[
	\forall U_i \in \T_i, \quad \pr_i^{-1}(U_i) \in \T
	\]
	Or
	\[
	\pr_1^{-1} (U_1) = U_1 \times X_2 \et \pr_2^{-1}(U_2) = X_1 \times U_2
	\]
	Donc $\mathcal S_* \subset \T$. En particulier $\mathcal S_* \subset \T_1 * \T_2$, d'où $\T_{\mathcal S_*} \subset \T_1 * \T_2$. De plus comme les $\pr_i$ sont continues par rapport à $\T_{\mathcal S_*}$ et $\T_1 * \T_2$, alors par minimalité de $\T_1 * \T_2$ on a nécessairement
	\[
	\T_1 * \T_2 \subset \T_{\mathcal S_*}
	\]
	D'où $\T_{\mathcal S_*} = \T_1 * \T_2$.
}

\begin{proposition}
	Soient $(X_1, \T_1)$ et $(X_2, \T_2)$ des espaces topologiques. Alors une base de $\T_1 * \T_*$ est donné par
	\[
	\mathcal B_* = \{U_1 \times U_2 \mid U_1 \in \T_1, \, U_2 \in \T_2\}
	\]
\end{proposition}

\prf{Comme on connait une sous-base $\mathcal S_*$ de la topologie produit $\T_1*\T_2$, alors pour obtenir une base il suffit de prendre toutes intersections de tous ces éléments. Ainsi soit
	\[
	U_1 \times X_2, U_2 \times X_2, \ldots U_k \times X_2 \et X_1 \times V_1, X_1 \times V_2, \ldots , X_1 \times V_l
	\]
	où $U_i \in \T_1$ et $V_j \in \T_2$. Comme
	\begin{align*}
		\bigcap_{i=1}^{k} (U_i \times X_2) \cap \bigcap_{j=1}^{\ell} (X_1 \times V_j)
		&= \left( \left( \bigcap_{i=1}^{k} U_i \right) \times X_2 \right) \cap \left( X_1 \times \left( \bigcap_{j=1}^{\ell} V_j \right) \right) \\
		&= \left( \bigcap_{i=1}^{k} U_i \right) \times \left( \bigcap_{j=1}^{\ell} V_j \right)
	\end{align*}
	on en déduit que tout élément de la base $\mathcal B_*$ engendrée par la sous-base $\mathcal S_*$ est de la forme
	\[
	U \times V \quad \text{où} \quad U \in \T_1, \, V \in \T_2
	\]
	Par ailleurs pour $U \in \T_1$ et $V \in \T_2$, on a
	\[
	(U \times X_2) \cap (X_1 \times V) = U \times V
	\]
	Ainsi tout produit $U \times V$ appartient à la base $\mathcal B_*$ engendrée par la sous-base $\mathcal S_*$. Et donc le corollaire suit.
}

\section{Produits cartésiens quelconques}


\begin{definition}
	Soient $X_1,\ldots, X_n$ des ensembles. On pose le \textbf{produit cartésien} 
	\[
	\Pi (X_1, \ldots, X_n) = \Big\{ f : \{1, \ldots, n\} \to \bigcup_{i=1}^n X_i \mid f(i) \in X_i \Big\}
	\]
	et les \textbf{évaluations}
	\[
	\begin{array}{cccc}
		\ev_i : &\Pi (X_1, \ldots, X_n) & \to & X_i \\
				& f & \mapsto & f(i)
	\end{array}
	\]
\end{definition}


\begin{lemme}
	Il existe une bijection
	\[
	\varphi : X_1 \times \ldots \times X_n \to \Pi(X_1, \ldots, X_n)
	\]
	telle que $\ev_i \circ \varphi = \pr_i$ pour tout $i \in \entiers$.
\end{lemme}
En d'autre terme on a le diagramme commutatif suivant ; 
\begin{figure}[H]
	\centering
	\begin{tikzcd}
		{X_1 \times \ldots \times X_n} && {\Pi(X_1, \ldots, X_n)} \\
		& {X_i}
		\arrow["\varphi", from=1-1, to=1-3]
		\arrow["{\mathrm{pr}_i}"', from=1-1, to=2-2]
		\arrow["{\mathrm{ev}_i}", from=1-3, to=2-2]
	\end{tikzcd}
\end{figure}

\prf{Comme on veut que $\ev_i \circ \varphi = \pr_i$, on doit définit l'application $\varphi$ telle que pour tout $(x_1, \ldots, x_n) \in X_1 \times \ldots \times X_n$ on ait
	\[
	\ev_i \circ \varphi (x_1, \ldots, x_n) = \pr_i(x_1, \ldots, x_n) = x_i
	\]
	Donc on a nécessairement
	\[
	\begin{array}{cccc}
		\varphi(x_1, \ldots, x_n) :  & \entiers & \to & \bigcup_{i=1}^n X_i  \\ & j & \mapsto & x_j
	\end{array}
	\] 
	Pour voir que c'est une bijection, on observe que l'application
	\[
	\begin{array}{cccc} \psi :& \Pi(X_1, \ldots, X_n) & \to & X_1 \times \ldots X_n \\ & f & \mapsto & (f(1), \ldots, f(n)) \end{array}
	\]
	est un inverse de $\varphi$.
}

\begin{definition}
	Soit $\mathcal X = \{X_i \mid i \in I\}$ une collection d'ensemble. Alors le \textbf{produit cartésien} de la collection $\mathcal X$ est
	\[
	\prod_{i \in I} X_i = \Big\{f : I \to \bigcup_{i \in I}X_i ~\big | ~f(i) \in X_i \Big\}
	\]
	Et les \textbf{projections} sont
	\[
	\begin{array}{cccc}
		\pr_i : & \prod_{j=1} X_j & \to & X_i \\
				&	f & \mapsto & f(i)
	\end{array}
	\]
	pour tout $i \in I$.
\end{definition}
