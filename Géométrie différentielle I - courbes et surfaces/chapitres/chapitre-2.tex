\chapter{Courbes dans l'espace et le plan euclidien}

\section{Notions fondamentales}

\begin{definition}
	Une \textbf{courbe paramétrée} dans $\Rn$ est une application continue $\alpha : I \to \Rn$,
	\[
	\alpha : u \mapsto (\alpha_1(u), \alpha_2(u), \ldots, \alpha_n(u)) \in \Rn, \quad u \in I
	\]
	où $I \subset \Rn$ est un intervalle appellé l'\textbf{intervalle de paramétrisation} de la courbe. De plus l'image de $I$ par $\alpha$ définit par
	\[
	\alpha(I) = \{ \alpha(u) \mid u \in I\} \subset \Rn
	\]
	s'appelle la \textbf{trace} ou le \textbf{support} de la courbe paramétrisée $\alpha$.
	
	On dit que la courbe $\alpha$ est \textbf{différentiable} en $u_0 \in I$ si la limite
	\[
	\d \alpha (u_0) = \lim_{u \to u_0} \frac{\alpha(u) - \alpha(u_0)}{u - u_0}
	\]
	existe. Cette limite s'appelle alors le \textbf{vecteur vitesse} de la courbe $\alpha$ en $u_0$ et on le note $\dot \alpha(u_0)$. De là on définit la \textbf{vitesse} de $\alpha$ et $u_0$ comme la norme du vecteur vitesse que l'on note
	\[
	V_\alpha(u_0) = \norme{\dot \alpha(u_0)}
	\]
\end{definition}

\begin{lemme}
	La courbe $\alpha(u) = (\alpha_1(u), \ldots, \alpha_n(u))$ est différentiable si et seulement si les fonctions $\alpha_i(u)$ sont dérivable en $u_0$. De plus
	\[
	\dot \alpha (u_0) = \Bigg( \d{\alpha_1}(u_0), \ldots, \d{\alpha_n}(u_0)\Bigg).
	\]
\end{lemme}

\prf{Supposons que les $\alpha_i$ sont dérivable, alors pour tout $i \in \entiers$ la limite
	\[
	\d{\alpha_i}(u_0) = \lim_{u \to u_0} \frac{\alpha_i(u) - \alpha_i(u_0)}{u - u_0}
	\]
	existe. De plus on remarque que
	\[
	\frac{\alpha(u) - \alpha(u_0)}{u - u_0} = \left(\frac{\alpha_1(u) - \alpha_1(u_0)}{u - u_0}, \ldots, \frac{\alpha_n(u) - \alpha_n(u_0)}{u - u_0} \right)
	\]
	et donc par passage à la limite, on a bien
	\[
	\dot \alpha (u_0) = \Bigg( \d{\alpha_1}(u_0), \ldots, \d{\alpha_n}(u_0)\Bigg).
	\]
	Ainsi $\alpha$ est bien différentiable. La réciproque est analogue.
}

\begin{definition} On a plusieurs définition importantes
	\begin{enumerate}
		\item Soit $\alpha$ une courbe de classe $C^1$ et $u_0$ une valeur de paramètre, alors on dira $p = \alpha(u_0)$ est \textbf{singulier} si
		\[
		\dot \alpha(u_0) = \bm 0 
		\]
		Si le point $p$ n'est pas singulier on dira qu'il est \textbf{régulier}. Ainsi une courbe est dite \textbf{régulière} si elle est de classe $C^1$ et si tous ses points sont réguliers.
		
		\item Le point $p = \alpha(u_0)$ sur une courbe de classe $C^2$ est \textbf{birégulier} si $\dot \alpha(u_0)$ et $\ddot \alpha(u_0)$ sont linéairement indépendant. Ainsi dans ce cas on peut définir le \textbf{plan osculateur} à la courbe $\alpha$ au point $p$ comme le plan passant par $p$ et de vecteur normal $\bm n = \dot \alpha(u_0) \times \ddot \alpha(u_0)$.
		
		\item Un point $p$ sur une courbe $\alpha : I \to \Rn$ est un \textbf{point double} s'il existe deux valeurs distinctes du paramètres telles que
		\[
		p = \alpha(u_1) = \alpha(u_2)
		\]
		On dira alors que la courbe est \textbf{simple} s'il ne contient pas de points doubles.
		
		\item Et finalement si $J \subset I$ est un intervalle, alors on dira que la restriction $\alpha_{\mid J}$ est un \textbf{arc de la courbe} $\alpha$.
		
	\end{enumerate}
	
\end{definition}

\section{Champ de vecteurs le long d'une courbe}

\begin{definition}
	Un \textbf{champ de vecteur} le long d'un courbe $\gamma : I \to \Rn$ est la donnée d'un vecteur
	\[
	\bm W(u) = w_1(u) \bm e_i + \ldots + w_n(u) \bm e_n
	\]
	pour toute valeur du paramètre $u \in I$.
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.5\textwidth]{"images/champ de vecteurs.png"}
		\caption*{Champ de vecteurs le long d'une courbe}
	\end{figure}
	Le champ de vecteur $\bm W$ est dit de classe $C^k$ si les dérivées de $w_1, \ldots, w_n$ sont de classe $C^k$.
\end{definition}

\begin{exemple}
	Un champ important est le \textbf{vecteur tangent} d'une courbe régulière $\gamma : I \to \Rn$ de classe $C^1$ est le champ de vecteur le long de la courbe obtenu en normalisant le vecteur vitesse
	\[
	\bm T_\gamma(u) := \frac{\dot \gamma(u)}{\norme{\dot \gamma(u)}}
	\]
\end{exemple}

\begin{lemme}[Règle de Leibnitz] \label{lem: Règle de Leibnitz}
	Soit $\bm W$ et $\bm Z$ deux champs de vecteurs de classe $C^1$ le long d'une courbe $\gamma : I \to \Rn$, alors
	\[
	\d{} \Big\langle \bm W, \bm Z \Big \rangle = \Big \langle \dot{\bm W}, \bm Z \Big \rangle + \Big \langle \bm W, \dot{\bm Z} \Big \rangle
	\]
	On a la même chose en dimension $2$ et $3$ avec, respectivement, les opérateurs $\land$ et $\times$.
\end{lemme}

\prf{Pour simplifier on note
	\[
	\bm W \cdot \bm Z := \Big \langle \bm W, \bm Z \Big \rangle
	\]
	On a alors par bilinéarité
	\[
		\bm W(u + \varepsilon) \cdot \bm Z(u + \varepsilon) - \bm W(u) \cdot \bm Z(u) = \Bigg( \bm W(u + \varepsilon) - \bm W(u)\Bigg) \cdot \bm Z(u + \varepsilon) + \bm W(u) \cdot \Bigg(\bm Z(u+ \varepsilon) - \bm Z(u)\Bigg).
	\]
	Il suffit de diviser cette identité par $\varepsilon$ et faire tendre $\varepsilon$ vers $0$ pour obtenir le lemme. Les autres formules sont analogue.
}

\begin{corollaire} \label{cor: invariance permuatation de la dérivée à signe près}
	Si $\bm W_1$ et $\bm W_2$ sont deux champs de vecteurs de classe $C^1$ le long de $\gamma$ tel que $\bm W_1 \cdot \bm W_2$ est constant, alors on a
	\[
	\bm W_1 \cdot \dot{\bm W}_2 = - \dot{\bm W}_1 \cdot \bm W_2
	\]
	De plus si $\bm W$ est de norme $\norme{\bm W}$ constante, alors $\dot{\bm W}$ est orthogonal à $\bm W$.
\end{corollaire}
\prf{La première affirmation est une conséquence immédiate du \autoref{lem: Règle de Leibnitz} et la deuxième découle directement de la première.
}

\section{Longueur et abscisse curviligne}

\begin{definition}
	La \textbf{longueur} d'un arc de courbe $\gamma : [a,b] \to \Rn$ de classe $C^1$ par morceaux est l'intégrale de sa vitesse
	\[
	l(\gamma) = \int_a^b V_\gamma(t) \, \mathrm dt
	\]
	où $V_\gamma(t) = \norme{\dot \gamma(t)}$.
\end{definition}

\begin{proposition}
	Si $g : \Rn \to \Rn$ est une similitude de rapport $\gamma > 0$ et $\gamma : [a,b] \to \Rn$ est un arc de courbe de classe $C^1$, alors $\tilde \gamma : = g \circ \gamma : [a,b] \to \Rn$ est aussi de classe $C^1$ et
	\[
	l(\tilde \gamma) = \lambda l(\gamma)
	\]
	En particulier la longueur d'une courbe est invariant par isométrie.
\end{proposition}

\prf{On que tout similitude $g$ est de la forme $g(x) = \lambda Ax + b$ où $A$ est une matrice orthogonale. On a donc $\tilde \gamma(u) = \lambda A \gamma(u) + b$ et, par la \nameref{lem: Règle de Leibnitz},
	\[
	\dot{\tilde \gamma}(u) = \lambda \dot A \gamma(u) + \lambda A \dot \gamma(u) + \dot b = \lambda A \dot \gamma(u)
	\]
	puique $A$ et $b$ sont des constantes. Par orthogonalité de $A$ on a
	\[
	V_{\tilde \gamma}(u) = \norme{\dot{\tilde \gamma}(u)} = \norme{\lambda A \dot \gamma(u)} = \lambda \norme{\dot \gamma(u)} = \lambda V_\gamma(u)
	\]
	et donc
	\[
	l(\tilde \gamma) = \int_a^b V_{\tilde \gamma}(u) \, \mathrm du = \int_a^b V_\gamma(u) \, \mathrm du = \lambda l(\gamma)
	\]
}
\begin{proposition}[additivité de la longueur]
	Soit $\gamma : [a,b] \to \Rn$ une courbe de classe $C^1$ et $c \in [a,b]$. On note $\alpha := \gamma_{\mid_{[a,c]}} : [a,c] \to \Rn$ et $\beta : \gamma_{\mid_{[c,b]}} : [c,b] \to \Rn$ les restrictions de $\gamma$ aux intervalles $[a,c]$ et $[c,b]$. Alors
	\[
	l(\gamma) = l (\alpha) + l(\beta)
	\]
\end{proposition}

\prf{On a par définition
	\begin{align*}
	l(\gamma) = \int_a^b V_\gamma(u) \, \mathrm du &= \int_a^c V_\gamma(u) \, \mathrm du + \int_c^b V_\gamma(u) \, \mathrm du  \\ &= \int_a^c V_\alpha(u) \, \mathrm du + \int_c^b V_\beta(u) \, \mathrm du = l(\alpha) + l(\beta)
	\end{align*}
	d'où le résultat.
}

\begin{proposition}
	Pour tout arc de courbe $\gamma : [a,b] \to \Rn$ de classe $C^1$ on a 
	\[
	d(\gamma(a), \gamma(b)) \leqslant l(\gamma).
	\]
	Autrement dit le plus court chemin reliant deux points est toujours le segment reliant les deux points.
\end{proposition}

\prf{Quitte à composer $\gamma$ par des isométrie, on peut supposer que $\gamma(a) = 0$ et $\gamma(b) = c e_1$. On écrit alors $\gamma(t)$ dans un système orthonormé. Alors
	\begin{align*}
		l(\gamma) = \int_a^b \norme{\dot \gamma(t)} \, \mathrm dt &= \int_a^b \Bigg(\sum_{i=1}^n \dot \gamma_i^2(t) \Bigg)^\frac12 \mathrm dt \\ & \geqslant \int_a^b |\dot \gamma_1(t)| \, \mathrm dt = \int_a^b \dot \gamma_1(t) \, \mathrm dt = \gamma_1(b) - \gamma_1(a) = d(\gamma(a), \gamma(b))
	\end{align*}
}
	
\begin{definition}
	Soit $\gamma : I \to \Rn$ une courbe paramétrée de classe $C^1$ et $u_0 \in I$ une valeur du paramètre. L'\textbf{abscisse curviligne} sur $\gamma$ correspondant au \textbf{point initial} $p_0 = \gamma(u_0)$	est la fonction $s_\alpha : I \to \Rn$ définie par
	\[
	S_\alpha(u) = \int_{u_0}^u V_\alpha(t) \, \mathrm dt
	\]
\end{definition}


\section{Changement de paramétrisation d'une courbe}
\begin{definition}
	Soit $\gamma : I \to \Rn$ une courbe paramétrée. On dit qu'une courbe $\alpha : J \to \Rn$ est une \textbf{reparamétrisation directe (resp. indirecte)} de $\gamma$ s'il existe un difféomorphisme de classe $C^1$
	\[
	h : I \to J
	\]
	avec $h' > 0$ (resp. $h' < 0$) de sorte que $\gamma = \alpha \circ h$.
\end{definition}
\begin{figure}[H]
	\centering
	\begin{tikzcd}
		I & J \\
		& {\mathbb R^n}
		\arrow["h", from=1-1, to=1-2]
		\arrow["\gamma"', from=1-1, to=2-2]
		\arrow["\alpha", from=1-2, to=2-2]
	\end{tikzcd}
\end{figure}

\subsection{Quantités géométriques et quantités cinématiques}
\begin{definition}
	Une quantité ou une notion attaché à une courbe est dite \textbf{géométrique} si elle est invariante par reparamétrisation, et elle est dite \textbf{cinématique} dans le cas contraire.
\end{definition}

\begin{lemme} \label{lem: vecteur tangent est invariant par reparamétrisation}
	Le vecteur tangent $\bm T_\gamma(t)$ est une quantité géométrique.
\end{lemme}

\prf{Soit $\alpha : J \to \Rn$ une reparamétrisation directe de la courbe $\gamma$. Il existe alors une fonction $h  : I \to J$ telle que $h' > 0$ et $\gamma = \alpha \circ h$. On sait que $V_\gamma(t) = V_\alpha(h(t)) h'(t)$, par conséquent
	\[
	\bm T_\gamma(t) = \frac1{V_\gamma(t)}\dt{\gamma(t)} = \frac1{V_\gamma(t)} \dt{\alpha(h(t))} =\frac{h'(t)}{V_\gamma(t)} \d{\alpha(u)} = \frac1{V_\alpha(u)} \d{\alpha(u)} =\bm T_\alpha(u)
	\]
}

\begin{proposition}
	Soient $\gamma : I \to \Rn$ une courbe de classe $C^1$, alors la longueur de la courbe est une quantité géométrique.
\end{proposition}

\prf{Soit $\alpha : J \to \Rn$ une reparamétrisation directe de la courbe $\gamma$. Alors en utilisant la formule de changement de variable dans les intégrales, on a
	\[
	l(\alpha) =\int_{a'}^{b'} V_\alpha(u) \, \mathrm du = \int_a^b V_\alpha(u) \dt u \, \mathrm dt = \int_a^b V_\gamma(t) \, \mathrm dt = l(\gamma)
	\]
De même dans le cas où $\alpha : J \to \Rn$ est une reparamétrisation indirecte de la courbe $\gamma$,
\[
l(\alpha) = \int_{a'}^{b'} V_\alpha(u) \, \mathrm du = \int_a^b V_\alpha \dt u \, \mathrm dt = - \int_b^aV_\gamma(t) \, \mathrm dt = \int_a^b V_\gamma(t) \, \mathrm dt = l(\gamma)
\]
}


\section{Paramétrisation naturelle d'une courbe naturelle}

\begin{theoreme} \label{thm: paramétrisation naturelle d'une courbe régulière}
	Soit $\gamma : I \to \Rn$ une courbe régulière de classe $C^1$ et $t_0 \in I$ une valeur de paramètre. Alors il existe une unique reparamétrisation direct $h : I \to J$, telle que $0 \in J$, $h(t_0) = 0$ et $\alpha := \gamma \circ h^{-1} : J \to \Rn$ est de vitesse 1, c'est-à-dire
	\[
	V_\alpha(s) = 1
	\]
	pour tout $s \in J$.
\end{theoreme}

\prf{
	On commence par l'unicité, on sait que
	\[
	V_\gamma(t) = h'(t) V_\alpha(s)
	\]
	Comme $V_\alpha(s) = 1$ et $h' > 0$, on a donc $h'(t) = V_\gamma(t)$ et comme $h(t_0) = 0$, on doit avoir
	\[
	s = h(t) = \int_{t_0}^t V_\gamma(\tau) \, \mathrm d \tau
	\]
	Ainsi la fonction $h$ coïncide avec l'abscisse curviligne $s(t)$. On définit dont naturellement $h$ par
	\[
	h(t) = s(t) = \int_{t_0}^t V_\gamma(\tau) \, \mathrm d \tau
	\]
	et l'intervalle $J = h(I)$. Alors on remarque que $h(t_0) = 0$ et $h'(t) = V_\gamma(t)$. Et donc en posant $\alpha := \gamma \circ h^{-1} : J \to \Rn$, on a bien
	\[
	V_\alpha(s)  = V_\gamma(t) \frac{1}{h'(t)} = 1
	\]
}

\begin{definition}
	On dit qu'une courbe régulière est \textbf{paramétrée naturellement} si $V_\gamma = 1$. Or le \autoref{thm: paramétrisation naturelle d'une courbe régulière} nous dit que tout courbe régulière de classe $C^1$ peut être reparamétrée naturellement.
\end{definition}

\section{Courbure d'une courbe de $\Rn$}

\begin{definition}
	Le \textbf{vecteur de courbure} d'une courbe régulière $\gamma : I \to \Rn$ de classe $C^2$ est le champ de vecteur le long de cette courbe défini par
	\[
	\bm K_\gamma(t) := \frac1{V_\gamma(t)} \dot{\bm T}_\alpha(t)
	\]
	A partir de là, on définit la \textbf{courbure} de la courbe $\gamma$ comme
	\[
	\kappa_\gamma(u) := \norme{\bm K_\gamma (u)}
	\]
\end{definition}

\begin{proposition}
	Le vecteur de courbure $\bm K_\gamma(t)$ et la courbure $\kappa_\gamma(t) = \norme{\bm K_\gamma(t)}$ sont des quantités géométriques.
\end{proposition}

\prf{Cela découle naturellement du fait que le vecteur tangent $\bm T_\gamma(t)$ soit une quantité géométrique par le \autoref{lem: vecteur tangent est invariant par reparamétrisation}.
}


\begin{lemme} \label{lem: le vecteur accéleration selon la courbure}
	Le vecteur accélération d'une courbe régulière $\gamma : I \to \Rn$ vérifie
	\[
	\ddot \gamma(t) = V_\gamma^2(t) \bm K_\gamma(t) + \dot V_\gamma(t) \bm T_\gamma(t)
	\]
\end{lemme}

\prf{On sait que $\dot \gamma(t) = V_\gamma(t) \bm T_\gamma(t)$ et donc
	\[
	\ddot \gamma(t) = \d{} (V_\gamma(t) \bm T_\gamma(t)) = V_\gamma(t) \dot{\bm T}_\gamma(t) + \dot V_\gamma(t)\bm T_\gamma(t) = V_\gamma^2(t) \bm K_\gamma(t) + \dot V_\gamma(t) \bm T_\gamma(t)
	\]
}

\begin{theoreme} \label{thm: vecteur de courbure d'une courbe régulière}
	Si $\gamma$ est paramétrée naturellement, alors
	\[
	\ddot \gamma(t) = \bm K_\gamma(t)
	\]
\end{theoreme}

\prf{Puisque $V_\gamma(t) = 1$ pour tout $t$, alors $\dot V_\gamma(t) = 0$ pour tout $t$, et donc par le \autoref{lem: le vecteur accéleration selon la courbure} on a bien
	\[
	\ddot \gamma (t) = \bm K_\gamma(t)
	\]
	d'où le théorème.
}


\begin{corollaire}
	Une courbe $\gamma : [a,b] \to \Rn$ est de courbure nulle si et seulement si c'est une droite ou un segment de droite.
\end{corollaire}

\prf{Par le \autoref{thm: paramétrisation naturelle d'une courbe régulière}, on peut supposer que $\gamma$ est paramétrisée naturellement, et donc le \autoref{thm: vecteur de courbure d'une courbe régulière} on a $\ddot \gamma(s) = \bm K_\gamma(s)$ et comme
	\[
	\kappa_\gamma(s) = \norme{\bm K_\gamma(s)} = 0
	\]
	on a donc $\ddot \gamma (s) = \bm 0$. Le vecteur $\bm v = \dot \gamma$ est donc constant, ainsi en intégrant on obtient
	\[
	\gamma(s) = \gamma(0) + s \bm v
	\]
	d'où le résultat.
}

\begin{lemme} \label{lem: le vecteur de courbure est colinéaire au vecteur normal}
	Le vecteur ed courbure d'une courbe de classe $C^2$	en un point est un multiple du vecteur normal principal en ce point ; 
	\[
	\bm K_\gamma(u) = \kappa_\gamma(u) \bm N_\gamma(u)
	\]
\end{lemme}


\begin{theoreme}
	La courbure d'une courbe de classe $C^2$ est la variation angulaire de la direction cette courbe par rapport au paramètre naturel.
\end{theoreme}

\prf{Soit $\gamma : I \to \Rn$ une courbe de classe $C^2$ paramétrée naturellement et notons 
	\[
	\varphi(s_0, s) := \angle(\bm T(s_0), \bm T(s))
	\]
	l'angle entre $\bm T(s_0)$ et $\bm T(s)$. Comme $\norme{\bm T(s_0)} = \norme{\bm T(s)} =1$, alors
	\[
	\norme{\bm T(s_0) - \bm T(s)} = 2 \sin \Big(\frac{\varphi(s_0, s)}2\Big)
	\]
	Et donc 
	\begin{align*}
	\lim_{s \to s_0^+} \frac{\varphi(s_0, s)}{s - s_0} &= \Bigg(\lim_{s \to s_0^+} \frac{\varphi(s_0,s)}{2 \sin\big(\frac{\varphi(s_0, s)}2 \big)} \Bigg) \cdot  \Bigg( \lim_{s \to s_0^+} \frac{2 \sin \big(\frac{\varphi(s_0, s)}2 \big)}{s - s_0}\Bigg)  \\&= \lim_{s \to s_0^+} \norme{\frac{\bm T(s_0) - \bm T(s)}{s -s_0}} = \norme{\dot{\bm T}(s_0)}
	\end{align*}

}

\section{Contact entre deux courbes}

\begin{definition}
	On dit que deux courbes de classe $C^k$
	\[
	\gamma_1, \gamma_2 : I \to \Rn
	\]
	ayant le même paramètre $t \in I$ ont un \textbf{contact d'ordre} $k$ en $t_0$ si $\gamma_1(t_0) = \gamma_2(t_0)$ et si leurs dérivées en $t_0$ coïncident jusqu'à l'ordre $k$ ; 
	\[
	\frac{\mathrm d^i \gamma_1}{\mathrm d t^i} (t_0) = \frac{\mathrm d^i \gamma_2}{\mathrm d t^i} (t_0)
	\]
	pour $i = 1,2, \ldots, k$.
\end{definition}


\begin{proposition}\label{prop: condition suffisante de point de contact d'ordre 2}
	Deux courbes $\gamma_1, \gamma_2 : I \to \Rn$ de classe $C^2$ ont un contact d'ordre $2$ en $t_0$ si et seulement si
	\[
	\gamma_1(t_0) = \gamma_2(t_0), \quad \dot \gamma_1(t_0) = \dot \gamma_2(t_0), \quad \dot V_{\gamma_1}(t_0) = \dot V_{\gamma_2}(t_0) \et \bm K_{\gamma_1}(t_0) = \bm K_{\gamma_2}(t_0)
	\]
\end{proposition}

\prf{C'est une conséquence directe du \autoref{lem: le vecteur accéleration selon la courbure}.}

\begin{theoreme}
	Soit $\gamma : I \to \R^2$ une courbe de classe $C^2$ qui est birégulière en $t_0 \in I$. Alors il existe un cercle $c : I \to \R^2$ ayant un contact d'ordre $2$ avec $\gamma$ en $t_0$. Ce cercle est unique, son rayon est l'inverse de $|\kappa_\gamma(t_0)|$ et son centre est donné par
	\[
	p = \gamma(t_0) + \frac1{\kappa_\gamma^2(t_0)} \bm K_\gamma(t_0)
	\]
	Ce cercle s'appelle le \textbf{cercle osculateur} ou le \textbf{cercle de courbure} à $\gamma$ en $t_0$ et, son centre est naturellement appelé le \textbf{centre de courbure} et son rayon est le \textbf{rayon de courbure} de $\gamma$ en $t_0$.
\end{theoreme}

\prf{Supposons que la courbe est paramétrée naturellement et notons $s$ le paramètre naturel. On considère également que le point considérer correspond à la valeur $s = 0$ du paramètre.
	
On pose $\rho = \frac1{\kappa_\gamma(0)}$, $\bm T := \bm T_\gamma(0) = \dot \gamma(0)$ et $\bm N := \rho \bm K_\gamma(0)$, puis posons
\[
p := \gamma(0) + \rho \bm N
\]
et considérons le cercle de centre $p$ et de rayon $\rho$ qu'on paramétrise par
\[
c(s) = p - \rho \cos \left(\frac{s}\rho \right)\bm N + \rho \sin \left(\frac{s}\rho \right) \bm T
\]
De plus il est claire que $\gamma(0) = c(0)$ et que $\dot c(0) = \bm T = \dot \gamma(0)$. On sait d'autre part que la courbure du cercle de rayon $\rho$ est constante et égale à $\frac1\rho = \kappa_\gamma(0)$. Et donc par la \autoref{prop: condition suffisante de point de contact d'ordre 2} on conclut bien que le cercle $c$ et la courbe $\gamma$ ont un contact d'ordre $2$ en $s=0$.
}


\section{Le repère de Frenet d'une courbe dans $\R^3$}

\begin{definition}
	Soit $\gamma : I \to \R^3$ de classe $C^2$, alors on définit le \textbf{vecteur binormal} de $\gamma$ comme le produit vectoriel unitaire tangent et du vecteur normal principal ;
	\[
	\bm B_\gamma(u) = \bm T_\gamma(u) \times \bm N_\gamma(u)
	\]
	Ainsi le \textbf{repère de Frenet} de $\gamma$ en $u$ est le repère défini par les trois champs de vecteurs
	\[
	\{\bm T_\gamma(u), \bm N_\gamma(u), \bm B_\gamma(u)\}
	\]
\end{definition}

\begin{figure}[H]
	\centering
	\includegraphics[width=0.5\textwidth]{"images/repère de Frenet.png"}
	\caption{Illustration du repère de Frenet (volé dans le poly)}
\end{figure}

\begin{lemme}
	Le vecteur binormal à la courbe $\gamma$ peut s'écrire 
	\[
	\bm B_\gamma(u) = \frac{\dot \gamma(u) \times \ddot \gamma(u)}{\norme{\dot \gamma(u) \times \ddot \gamma(u)}}
	\]
\end{lemme}

\prf{On a 
	\[
	\dot \gamma (\ddot \gamma - \langle \ddot \gamma, \bm T \rangle \, \bm T ) = \dot \gamma \times \ddot \gamma
	\]
	Et donc
	\[
	\bm B = \bm T \times \bm N = \frac{\dot \gamma}{V_\gamma} \times \frac{\ddot \gamma - \langle \ddot \gamma, \bm T \rangle \, \bm T }{\norme{\ddot \gamma - \langle \ddot \gamma, \bm T \rangle \, \bm T }} = \frac{\dot \gamma \times \ddot \gamma}{V_\gamma\norme{\ddot \gamma - \langle \ddot \gamma, \bm T \rangle \, \bm T } } = \frac{\dot \gamma \times \ddot \gamma}{\norme{\dot \gamma \times \ddot \gamma}}
	\]
}

\begin{definition}
	On dit qu'une courbe $\gamma : I \to \R^3$ est \textbf{régulière au sens de Frenet} si elle est de classe $C^2$, et que le vecteur principal est un champ $u \mapsto \bm N_\gamma(u)$ de classe $C^1$. On peut alors définir la \textbf{torsion} d'une courbe $\gamma : I \to \R^3$ régulière au sens de Frenet comme étant la fonction $\tau: I \to \R$ définie par
	\[
	\tau(u) := \frac{\langle \dot{\bm N}(u), \bm B(u) \rangle}{V_\gamma(u)}
	\]
\end{definition}

\begin{theoreme}[formule de Serret-Frenet] \label{thm: formules de Serret-Frenet}
	Soit $\gamma : I \to \R^3$ une courbe régulière au sens de Frenet. Alors le repère de Frenet est de classe $C^1$ et ses dérivées sont données par les formules
	\[
	\frac1{V_\gamma} \dot{\bm T} =\kappa \bm N, \quad \frac1{V_\gamma}\dot{\bm N} = -\kappa \bm T + \tau \bm B \et\frac1{V_\gamma} \dot{\bm B} = -\tau \bm N
	\]
\end{theoreme}

\prf{Le vecteur $\bm T$ et $\bm N$ sont de classe $C^1$ par hypothèse, et donc il en va de même pour le vecteur binormal car $\bm B = \bm T \times \bm N$. Donc le repère de Frenet est bien de classe $C^1$.
	
	La première équation est une conséquence directe des égalités
	\[
	\dot{\bm T} = V_\gamma \bm K = V_\gamma \kappa \bm N
	\]
	dû au \autoref{lem: le vecteur de courbure est colinéaire au vecteur normal}. Pour la deuxième égalité, on remarque d'abord que
	\[
	\dot{\bm N} = \langle \dot{\bm N}, \bm T \rangle \bm T + \langle \dot{\bm N}, \bm N \rangle \bm N + \langle \dot{\bm N}, \bm B \rangle \bm B
	\]
	car le repère de Frenet est orthonormé. D'autre part, on a $\langle \dot{\bm N}, \bm N \rangle = 0$ et par le \autoref{cor: invariance permuatation de la dérivée à signe près}
	\[
	\langle \dot{\bm N}, \bm  \rangle \bm T = -\langle \bm N, \dot{\bm T} \rangle \bm T = - V_\gamma \kappa \bm T
	\]
	Et donc
	\[
	\dot{\bm N} = - V_\gamma \kappa \bm T + V_\gamma \tau \bm B
	\]
	car $\langle \dot{\bm N}, \bm B \rangle = V_\gamma \tau$ par définition de la torsion. Et finalement pour le troisième formules, on remarque que 
	\[
	\dot{\bm B} = \langle \dot{\bm B}, \bm T \rangle \bm T + \langle \dot{\bm B}, \bm N \rangle \bm N + \langle \dot{\bm B}, \bm B \rangle \bm B
	\]
	Or $\langle \dot{\bm B}, \bm B \rangle = 0$ car $\norme{\bm B}$ est constante et 
	\[
	\langle \dot{\bm B}, \bm T \rangle = - \langle \bm B, \dot{\bm T} \rangle = - V_\gamma \kappa \langle \bm B, \bm N \rangle = 0
	\]
	et donc
	\[
	\dot{\bm B} = \langle \dot{\bm B}, \bm N \rangle \bm N = - \langle \bm B, \dot{\bm N} \rangle \bm N = - V_\gamma \tau \bm N
	\]
	Le théorème est donc démontrer.
}

\begin{proposition}
	Une courbe $\gamma : I \to \R^3$ régulière au sens de Frenet est située dans un plan si et seulement si sa torsion est identiquement nulle.
\end{proposition}

\prf{Supposons que la courbe $\gamma$ soit située dans un plan $\Pi \subset \R^3$, alors le vecteur binormal est constant. Et donc la troisième formules du \autoref{thm: formules de Serret-Frenet} on a bien $\tau_\gamma = 0$.
	
Réciproquement, supposons que $\tau = 0$. Alors la troisième formule du \autoref{thm: formules de Serret-Frenet}, implique que le vecteur binormal est constant. On note ce vecteur par $\bm B$ et on pose
\[
h(u) = \langle \gamma(u) - \gamma(u_0), \bm B \rangle
\]
or on remarque que
\[
\d h = \langle \dot \gamma(u), \bm B \rangle = V_\gamma \langle \bm T(u), \bm B \rangle = 0
\]
Par conséquent $h$ est constant, et comme $h(u_0) = 0$, la fonction $h$ est identiquement nulle, ce qui montre que la courbe $\gamma$ est contenu dans le plan d'équation
\[
\langle \bm x - \gamma(u_0), \bm B \rangle = 0
\]
qui n'est rien d'autre que le plan orthogonal à $\bm B$	passant par $\gamma(u_0)$.
}

\subsection{Variation augulaire du plan osculateur}

\begin{proposition}
	La torsion du courbe régulière au sens de Frenet mesure la variation de son plan osculateur.
\end{proposition}

\prf{Soit $\gamma : I \to \R^3$ une courbe au sens de Frenet, qu'on suppose paramétrée naturellement. On note $\theta(s_0, s)$ l'angle entres les plans osculateurs à $\gamma$ en $s_0$ et en $s$, on remarque alors que
	\[
	\theta(s_0, s) = \angle (\bm B(s_), \bm B(s))
	\]
	car le vecteur normal au plan osculateur en un point de $\gamma$ est le vecteur binormal de celui-ci. Et donc
	\[
	|\tau(s_0)| = \norme{\dot{\bm B}(s_0)} = \lim_{s \to s_0^+} \frac{\theta(s_0, s)}{s - s_0} = \dot \theta(s_0)
	\]
	d'où la proposition.
}

\subsection{Courbe de pente constante}

\begin{definition}
	On dit qu'une courbe $\gamma : I \to \R^3$ de classe $C^1$ est de \textbf{pente constante} si elle est régulière et son vecteur tangent fait un angle constant avec une direction fixe. Une telle courbe s'appelle aussi un \textbf{hélice généralisée}.
\end{definition}

\begin{theoreme}
	Une courbe $\gamma : I \to \R^3$ régulière au sens de Frenet est de pente constante si et seulement si le rapport
	\[
	\frac{\tau_\gamma(u)}{\kappa_\gamma(u)}
	\]
	est constant.
\end{theoreme}


\prf{On peut supposer sans perte de généralité que $\gamma$	est paramétrée naturellement. Supposons qu'il existe un vecteur non-nul $\bm A \in \R^3$ tel que le produit scalaire $\langle \bm T_\gamma(s), \bm A \rangle = a$ est constant. Ainsi en dérivant et par le première équation des \nameref{thm: formules de Serret-Frenet}, on a
	\[
	0 = \langle \dot{\bm T}(s), \bm A \rangle = \kappa_\gamma(s) \langle \bm N(s), \bm A \rangle
	\]
	or on a supposé que la courbe était birégulière, et donc sa courbure est non nulle, ainsi $\langle \bm N(s), \bm A \rangle = 0$ pour tout $s$. Or par la troisième équation des \nameref{thm: formules de Serret-Frenet} on a
	\[
	\frac{\mathrm d}{\mathrm ds}\langle \bm B(s), A \rangle = \langle \dot{\bm B}(s), A \rangle = - \tau_\gamma(s) \langle \bm N(s), \bm A \rangle = 0
	\]
	Donc $b = \langle \bm B(s), \bm A \rangle$ est également constant. Et finalement la deuxième équation des \nameref{thm: formules de Serret-Frenet} nous dit que
	\[
	0 = \frac{\mathrm d}{\mathrm ds} \langle \bm N(s), \bm A \rangle = \langle \dot{\bm N}(s), \bm A \rangle = -\kappa_\gamma(s) \langle \bm T(s), A \rangle + \tau_\gamma(s)\langle \bm B(s), \bm A \rangle
	\]
	et donc
	\[
	\frac{\tau_\gamma(u)}{\kappa_\gamma(u)} = \frac{\langle \bm T(s), \bm A \rangle}{\langle \bm B(s), \bm A \rangle} =\frac{a}{b}
	\]
	Réciproquement, supposons que $\lambda := \frac{\tau_\gamma(u)}{\kappa_\gamma(u)}$ est constant, on considère le champ de vecteurs
	\[
	A(s) := \lambda \bm T(s) + \bm B(s)
	\]
	Il est claire que l'angle entre $\bm T$ et $\bm A$ est constant car $\langle \bm A, \bm T \rangle = \lambda$. De plus
	\[
	\frac{\mathrm d}{\mathrm ds} \bm A(s) =\lambda \dot{\bm T} + \dot{\bm B} = \lambda\kappa_\gamma(s) \bm N - \tau_\gamma(s)\bm N = 0
	\]
	Et donc le théorème suit.
}

\subsection{Le théorème fondamental de la théorie des courbes de $\R^3$}

\begin{theoreme}
	Soient $\kappa, \tau : I \to \R$ deux fonctions continues. Si $\kappa > 0$ pour tout $s \in I$ alors il existe une courbe $\gamma : I \to \R^3$, régulière au sens de Frenet, paramétrée naturellement et telle que
	\[
	\kappa_\gamma(s) = \kappa(s) \et \tau_\gamma(s) = \tau(s)
	\]
	pour tout $s$. Cette courbe est unique à déplacement près.
\end{theoreme}

\prf{On commence par montrer l'unicité. Supposons que $\gamma_1, \gamma_2 : I \to \R^3$ soient deux courbes régulières au sens de Frenet, paramétrés naturellement et donc la courbure et la torsion valent respectivement $\kappa(s)$ et $\tau(s)$. On note $\{ \bm T_1(s), \bm N_1(s), \bm B_1(s)\}$ et $\{ \bm T_2(s), \bm N_2(s), \bm B_2(s)\}$ leurs repère de Frenet respectifs.
	
Sans perte de généralité, on peut supposer que l'intervalle $I$ contient $0$ et quitte à composer l'une ou l'autre courbes par un déplacement, on peut supposer que $\gamma_1(0) = \gamma_2(0) = 0$ et qu'en $s = 0$ les deux repères de Frenet coïncident avec la base canonique $\{e_1, e_2, e_3\}$ de $\R^3$. On note alors $\bm F_i(s) \in SO(3)$ la matrice orthogonal dont les colonnes sont les composantes des vecteurs $\bm T_1(s), \bm N_i(s), \bm B_i(s)$ pour $i=1,2$. Les \nameref{thm: formules de Serret-Frenet} s'écrivent alors
\[
\frac{\mathrm d}{\mathrm ds} \bm F_i(s) =\bm F_i(s) \Omega(s), \quad \text{où} \quad \Omega = \begin{pmatrix} 0 & - \kappa(s) & 0 \\ \kappa(s) & 0 & - \tau(s) \\ 0 & \tau(s) & 0 \end{pmatrix}
\]
On a alors
\begin{align*}
	\frac{\mathrm d}{\mathrm ds} (\bm F_1 \bm F_2^{-1}) &=  \frac{\mathrm d}{\mathrm ds} (\bm F_1 \bm F_2^\top) = \dot{\bm F_1} \bm F_2^\top + \bm F_1 \dot{\bm F}^\top_2 \\ &= (\bm F_1 \Omega) \bm F_2^\top + \bm F_1(\bm F_2 \Omega)^\top = \bm F_1 \Omega \bm F_2^\top + \bm F_1 \Omega^\top \bm F_2^\top = 0
\end{align*}
puisque $\Omega$ est antisymétrique. Par conséquent la matrice $\bm F_1 \bm F_2^{-1}$ est constante. Mais on a supposé que $\bm F_1(0) = \bm F_2(0) = \bm I_3$. Donc $\bm F_1(s) \bm F_2(s)^{-1} = \bm I_3$ pour tout $s$, c'est-à-dire $\bm F_1(s) = \bm F_2(s)$. En particulier $\bm T_1(s) = \bm T_2(s)$ pour tout $s$ et donc
\[
\gamma(s) = \int_0^s \bm T_1(u) \, \mathrm du = \int_0^s \bm T_2(u) \, \mathrm du = \gamma_2(s)
\]

Il reste à montrer l'existence. Pour cela on se donne deux fonctions $\kappa, \tau : I \to \R$ et on considère le problème de Cauchy linéaire
\begin{equation} \label{eq: équation différentielle de Frenet}
	\frac{\mathrm d}{\mathrm ds} \bm F_i(s) =\bm F_i(s) \Omega(s), \quad \bm F(0) = \bm I_3
\end{equation}
où $\Omega(s)$ est la matrice définie plus haut. Le théorème de Cauchy-Lipschitz global nous dit qu'il existe une solution globale $\bm F: I \to \mathrm M_3(\R)$ de classe $C^1$ de ce problème. Or on remarque que
\[
\frac{\mathrm d}{\mathrm ds} = \bm F \bm F^\top = \dot{\bm F} \bm F^\top + \bm F \dot{\bm F}^\top = \bm F \Omega \bm F^\top + \bm F \Omega^\top \bm F^\top = \bm F(\Omega + \Omega^\top) \bm F^\top = 0
\]
puisque $\Omega$ est antisymétrique. Or $\bm F(0) \bm F^\top(0) = \bm I_3$ et donc $\bm F(s) \bm F^\top(s) = \bm I_3$ pour tout $s \in I$, ce qui signifie que $\bm F(s) \in SO(3)$.

On note respectivement $\bm T(s), \bm N(s), \bm B(s)$ les trois colonnes de $\bm F(s)$ et on définit $\gamma : I \to \R^3$ par
\[
\gamma(s) = \int_0^s \bm T(u) \, \mathrm du
\]
Alors $\gamma$ est clairement une courbe de classe $C^2$ car $s \mapsto \bm T(s)$ est de classe $C^1$. De plus cette courbe est paramétrée naturellement puisque $\dot \gamma(s) = \bm T(s)$ est un vecteur unitaire. L'équation différentielle \eqref{eq: équation différentielle de Frenet} est équivalente aux \nameref{thm: formules de Serret-Frenet}. Cela implique que $s \mapsto \bm N(s)$ est aussi de classe $C^1$ et, que la courbure et la torsion sont données par les fonction $\kappa$ et $\tau$, ce qui complète la démonstration.
}

\section{Courbe dans un plan orienté}
\begin{definition}
	\begin{enumerate}
		\item Le \textbf{repère de Frenet orienté} d'une courbe régulière $\gamma : I \to \R^2$ de classe $C^2$ dans le plan orienté est le repère mobile d'origine $\gamma(u)$ qui est formé par les deux vecteurs
		\[
		\bm T_\gamma(u) = \frac{\dot \gamma(u)}{\norme{\dot \gamma(u)}} \et \bm N_\gamma^{\ori}(u) = \bm J \cdot \bm T_\gamma(u)
		\]
		où $\bm J$ désigne la matrice de rotation d'un quart de tour.
		\item La \textbf{courbure orienté} de $\gamma$ en $u$ est définie par
		\[
		\kappa_\gamma^{\ori} (u)= \frac{1}{V_\gamma(u)} \langle \dot{\bm T}_\gamma(u), \bm N_\gamma^{\ori}(u) \rangle.
		\]
	\end{enumerate}
\end{definition}

\begin{remarque}
	Le repère $\{\bm T_\gamma(u), \bm N_\gamma^{\ori}(u)\}$ est un repère orthonormé direct. La courbure non orientée de $\gamma$ est égale à la valeur absolue de $\kappa_\gamma^{\ori}(u)$. La courbure orientée peut aussi s'écrire comme
	\[
	\kappa_\gamma^{\ori}(u) = \frac{\bm T_\gamma(u) \land \dot{\bm T}_\gamma(u)}{V_\gamma(u)}
	\]
	Finalement, si la courbe $\gamma$ est birégulière, alors
	\[
	\kappa_\gamma^{\ori}(u) \bm N_\gamma^{\ori}(u) = \kappa_\gamma(u) \bm N_\gamma(u) = \bm K_\gamma(u)
	\]
	Cette égalité vient du fait que si on change l'orientation du plan, alors $\kappa_\gamma^{\ori}(u)$ et $\bm N_\gamma^{\ori}(u)$ changent tous les deux de signes.
\end{remarque}

\begin{proposition}
	Les \nameref{thm: formules de Serret-Frenet} pour une courbe plane de classe $C^2$ s'écrivent
	\[
	\frac1{V_\gamma} \dot{\bm T} = \kappa_\gamma^{\ori} \bm N_\gamma^{\ori} \et \frac{1}{V_\gamma} \dot{\bm N}_\gamma^{\ori} = - \kappa_\gamma^{\ori} \bm T_\gamma
	\]
\end{proposition}

\prf{On a d'une part
	\[
	\dot{\bm T} = \langle \dot{\bm T}, \bm T \rangle \bm T + \langle \dot{\bm T}, \bm N_\gamma^{\ori} \rangle \bm N_\gamma^{\ori} = \langle \dot{\bm T}, \bm N_\gamma^{\ori} \rangle \bm N_\gamma^{\ori} = V_\gamma \kappa_\gamma^{\ori} \bm N_\gamma^{\ori}
	\]
	par définition de la courbure orientée. D'autre part
	\[
	\dot{\bm N}_\gamma^{\ori} = \langle \dot{\bm N}_\gamma^{\ori}, \bm T \rangle \bm T  + \langle \dot{\bm N}_\gamma^{\ori}, \bm N_\gamma^{\ori} \rangle \bm N_\gamma^{\ori} = \langle \dot{\bm N}_\gamma^{\ori}, \bm T \rangle \bm T = - \langle \dot{\bm T}, \bm N_\gamma^{\ori} \rangle \bm T = - V_\gamma \kappa_\gamma^{\ori} \bm T
	\]
}

\begin{proposition}
	La courbure orienté d'une courbe plane $\gamma : I \to \R^2$ de classe $C^2$ est donnée par
	\[
	\kappa_\gamma^{\ori}(u) = \frac{\dot \gamma(u) \land \ddot \gamma(u)}{V_\gamma^3(u)}
	\]
\end{proposition}

\prf{On a
	\[
	\dot \gamma \land \ddot \gamma = (V_\gamma \bm T) \land (\dot V_\gamma \bm T + V_\gamma^2 \kappa_\gamma^{\ori} \bm N_\gamma^{\ori}) = V_\gamma^3 \kappa_\gamma^{\ori}
	\]
	car $\bm T \land \bm T = 0$ et $\bm T\land \bm N = 1$.
}

\begin{definition}
	On dit qu'un arc de courbe $\gamma : ]a,b[ \to \R^2$ est \textbf{convexe} si la courbure orientée $\kappa_\gamma^{\ori}$ est positive sur cet arc. L'arc est \textbf{concave} si la courbure orientée est négative. Un \text{point d'inflexion} est un point séparant un arc convexe d'un arc concave, et donc en particulier la courbure est nulle en un point d'inflexion.
	
	On dit qu'un arc est une \textbf{spirale} si la courbure est strictement monotone sur cet arc. Un point de la courbe est un \textbf{sommet} si c'est un maximum local ou minimum local de la courbure.
\end{definition}

\subsection{La fonction angulaire}

\begin{definition}
	Soit $\gamma : I \to \R^2$ une courbe régulière de classe $C^1$. La \textbf{fonction angulaire} de la courbe $\gamma$ de point initial $p = \gamma(u_0)$ est la fonction	$\varphi : I \to \R$ telle que
	\begin{enumerate}
		\item $\varphi(u_0)$ est l'angle orienté entre $\dot \gamma(u)$ et le vecteur $\bm e_1$,
		\item $\varphi$ est continue,
		\item L'angle orienté entre $\dot \gamma(u)$ et $\bm e_1$ est égale à $\varphi(u) \mod{2\pi}$ pour tout $u \in I$.
	\end{enumerate}
\end{definition}

\begin{lemme} \label{lem: repère de Frenet exprimé via la fontion angulaire}
	Le repère de Frenet orienté orienté d'une courbe régulière $\gamma : I \to \R^2$ de classe $C^2$ peut s'écrire
	\[
	\bm T_\gamma(u) = (\cos(\varphi(u)), \sin(\varphi(u))) \et \bm N_\gamma(u)^{\ori} = (- \sin(\varphi(u)), \cos(\varphi(u)))
	\]
	où $\varphi : [a,b] \to \R$ est la fonction angulaire de $\gamma$.
\end{lemme}

\prf{La formule pour $\bm T$ est évidente, puisque $\varphi \mod 2 \pi$ mesure l'angle du vecteur $\bm T$ avec $\bm e_1$. La formule pour $\bm N$ se déduit alors de la définition $\bm N = \bm J \cdot \bm T$.
}

\begin{theoreme}
	La courbure orientée d'une courbe $\gamma : I \to \R^2$ de classe $C^2$ vérifie
	\[
	\kappa_\gamma^{\ori}(u) = \frac{1}{V_\gamma (u)} \dot \varphi (u)
	\]
\end{theoreme}

\prf{Par le \autoref{lem: repère de Frenet exprimé via la fontion angulaire}, on a
	\[
	\dot{\bm T} = (- \sin(\varphi(u)), \cos(\varphi(u))) \dot \varphi(u) = \bm N_\gamma^{\ori} \dot \varphi(u)
	\]
	Et donc
	\[
	\kappa_\gamma^{\ori}(u) = \frac{1}{V_\gamma(u)} \langle \dot{\bm T}_\gamma(u), \bm N_\gamma^{\ori}(u) \rangle = \frac{1}{V_\gamma(u)} \langle \dot \varphi(u) \bm N_\gamma^{\ori}, \bm N_\gamma^{\ori} \rangle = \frac{1}{V_\gamma} \dot \varphi(u)
	\]
}

\begin{remarque}
	Lorsque la courbe est paramétrée naturellement, on a $\kappa_\gamma^{\ori}(s) = \dot \varphi(s)$. On écrit souvent cette relation sous la forme différentielle
	\[
	\mathrm d \varphi = \kappa_\gamma^{\ori} \mathrm ds
	\]
\end{remarque}

\begin{theoreme}
	Toute fonction continue $\kappa : I \to \R$ est la courbure orientée d'une courbe plane de classe $C^2$ paramétrée naturellement. Cette courbe est unique à déplacement près.
\end{theoreme}

\prf{On commence par montrer l'unicité. Supposons que $\gamma : I \to \R^2$ soit une courbe de classe $C^2$ paramétrée naturellement donc la courbure orientée est $\kappa(s)$. Le vecteur tangent est donné par
	\[
	\bm T_\gamma(s) = \dot\gamma(s) = (\dot x(s), \dot y(s)) = (\cos(\varphi(s)),\sin(\varphi(s)))
	\]
	où $\varphi : I \to \R$ est la fonction angulaire. Les trois fonctions $(x(s), y(s), \varphi(s))$ forment alors une solution du système d'équations différentielles
	\[
	\dot x = \cos(\varphi), \quad \dot y = \sin(\varphi) \et \dot \varphi = \kappa(s)
	\]
	La courbe $\gamma$ est donc déterminée à partir de la fonction $\kappa(s)$ en résolvant ces équations. Pour résoudre ce système, on calcule $\varphi$ par intégration : 
	\[
	\varphi(s) = \varphi_0 + \int_0^s \kappa(\sigma) \, \mathrm d\sigma
	\]
	Et donc
	\[
	x(s) = x_0 + \int_0^s\cos(\varphi(\sigma)) \, \mathrm d \sigma \et y(s) = y_0 + \int_0^s \sin(\varphi(\sigma)) \, \mathrm d\sigma
	\]
	En changeant les valeurs de $x_0$ et $y_0$, on modifie la courbe par translation, si l'on change $\varphi_0$, alors la courbe $\gamma$ subit une rotation. L'argument montre à la fois l'existence et l'unicité de la courbe $\gamma$ à un déplacement près.
}

\begin{remarque}
	La relation $\kappa_\gamma^{\ori} = \kappa(s)$ entre l'abscisse curviligne et la courbure orientée s'appelle l'\textbf{équation intrinsèque} de la courbe. Elle contient la même information que le diagramme de courbure.
\end{remarque}

\section{Le théorème des quatres sommets}

\begin{definition}
	On dit que $\gamma : [a,b] \to \R^n$ est une \textbf{courbe fermée de classe} $C^m$ si $\gamma(a) = \gamma(b)$ et
	\[
	\gamma^{(k)} (a) =  \gamma^{(k)}(b)
	\]
	pour tout $ 1 \leqslant k \leqslant m$.
\end{definition}

\begin{lemme}
	Soit $R>0$ et $\gamma_R : \R \to \R^2$ une paramétrisation, parcourue dans le
	sens trigonométrique, du cercle de rayon $R$. Soit $\sigma : I \to \R^2$ une
	courbe régulière de classe $C^2$ et $t_0 \in I$ tels que
	\[
	\sigma(t_0) = p = \gamma_R(t_0) \et \bm T_\sigma(p) = \bm T_{\gamma_R}(p).
	\]
	On suppose de plus que $\sigma$ est contenue, au voisinage de $t_0$, dans le
	disque fermé $D_R$ de bord $\gamma_R$. Alors
	\[
	\kappa_\sigma(p) \;\geqslant\; \frac{1}{R} \;=\; \kappa_{\gamma_R}(p),
	\]
	où $\kappa$ désigne la courbure signée.
\end{lemme}

\prf{Pour le graphe d'une fonction $f$ de classe $C^2$
	parcouru dans le sens des $x$ croissants, elle vaut
	\[
	\kappa(x)=\frac{f''(x)}{\bigl(1+f'(x)^2\bigr)^{3/2}}.
	\]
	
	La courbure signée est invariante par les isométries directes
	(translations et rotations) du plan, ainsi que par reparamétrisation
	préservant l'orientation. On peut donc supposer que $p=(0,0)$ et que
	$\bm T_\sigma(p)=\bm T_{\gamma_R}(p)=(1,0)$. Comme $\gamma_R$ est parcouru dans
	le sens trigonométrique, le centre du cercle se trouve du côté de
	$\bm N=(0,1)$ : c'est le point $(0,R)$. Ainsi
	\[
	D_R=\{(x,y)\in\R^2 : x^2+(y-R)^2\leqslant R^2\}.
	\]
	

	Pour $|x|<R$ et $y$ proche de $0$, on a
	\[
	(x,y)\in D_R \iff y \geqslant f_1(x), \qquad
	f_1(x) := R-\sqrt{R^2-x^2}.
	\]
	La fonction $f_1$ est de classe $C^\infty$ sur $]-R,R[$ et vérifie
	\[
	f_1(0)=0,\qquad f_1'(0)=0,\qquad f_1''(0)=\frac1R,
	\]
	donc $\kappa_{\gamma_R}(p)=f_1''(0)=\dfrac1R$.
	
	Écrivons $\sigma(t)=(x(t),y(t))$. Comme $\bm T_\sigma(p)=(1,0)$, on a
	$x'(t_0)=\|\sigma'(t_0)\|>0$. Par le théorème d'inversion locale, $x$ est un
	$C^2$-difféomorphisme d'un voisinage de $t_0$ sur un intervalle ouvert
	$J\ni 0$, et $\sigma$ coïncide près de $p$ avec le graphe de
	$f_2 := y\circ x^{-1}\in C^2(J)$. On a
	\[
	f_2(0)=0,\qquad f_2'(0)=\frac{y'(t_0)}{x'(t_0)}=0,
	\]
	et donc
	\[
	\kappa_\sigma(p)=f_2''(0).
	\]
	Comme $\sigma\subset D_R$ près de $p$, l'étape 2 donne $f_2(x)\geqslant f_1(x)$
	pour $x$ proche de $0$. Posons $h:=f_2-f_1$, de classe $C^2$ au voisinage de
	$0$. Alors
	\[
	h\geqslant 0,\qquad h(0)=0,\qquad h'(0)=0.
	\]
	Si l'on avait $h''(0)<0$, la formule de Taylor--Young
	$h(x)=\tfrac12 h''(0)x^2+o(x^2)$ donnerait $h(x)<0$ pour $x\neq 0$ petit, ce
	qui est absurde. Donc $h''(0)\geqslant 0$, c'est-à-dire
	\[
	\kappa_\sigma(p)=f_2''(0)\;\geqslant\; f_1''(0)=\frac1R=\kappa_{\gamma_R}(p).
	\]
}


\begin{lemme}
	Soit $R>0$ et $\gamma_R$ une paramétrisation, parcourue dans le sens
	trigonométrique, du cercle de rayon $R$. Soit $\sigma : I \to \R^2$ une courbe
	régulière de classe $C^2$, parcourue dans le sens trigonométrique, et soient
	$t_1<t_2$ dans $I$ tels que
	\[
	\sigma(t_1)=p_1=\gamma_R(t_1), \quad \sigma(t_2)=p_2=\gamma_R(t_2),
	\qquad
	\bm T_\sigma(p_i)=\bm T_{\gamma_R}(p_i)\quad (i=1,2).
	\]
	Soit $\theta\in\,]0,\pi]$ l'angle de l'arc de $\gamma_R$ allant de $p_1$ à
	$p_2$ dans le sens trigonométrique. On suppose que :
	\begin{itemize}
		\item l'angle total dont tourne $\bm T_\sigma$ entre $t_1$ et $t_2$ vaut
		exactement $\theta$ ;
		\item $\sigma_{|[t_1,t_2]}$ n'est pas entièrement contenue dans $\gamma_R$.
	\end{itemize}
	Alors il existe $p_3\in\sigma(]t_1,t_2[)$ tel que
	\[
	\kappa_\sigma(p_3)<\frac1R,
	\]
	où $\kappa$ désigne la courbure orientée.
\end{lemme}

\prf{On raisonne par l'absurde : supposons que $\kappa_\sigma\geqslant \frac1R$ en
	tout point de $\sigma([t_1,t_2])$. Notons $e(u)=(\cos u,\sin u)$.
	
	Quitte à reparamétrer par la longueur d'arc, on écrit
	$\bm T_\sigma = e(\varphi)$ avec $\varphi$ de classe $C^1$ (relèvement de
	l'angle, possible car $\bm T_\sigma$ est $C^1$), et
	$\varphi'=\kappa_\sigma\geqslant \frac1R>0$. Ainsi $\varphi$ est strictement
	croissante. Notons $\varphi_0=\varphi(t_1)$ ; par hypothèse
	$\varphi(t_2)=\varphi_0+\theta$. Comme $\varphi$ est un $C^1$-difféomorphisme
	de $[t_1,t_2]$ sur $[\varphi_0,\varphi_0+\theta]$, on peut reparamétrer
	$\sigma$ par $u=\varphi-\varphi_0\in[0,\theta]$. Le rayon de courbure
	$\rho(u)=1/\kappa_\sigma$ est alors continu et vérifie $0<\rho\leqslant R$, et
	puisque $\frac{ds}{du}=\rho(u)$,
	\[
	\sigma(u)=p_1+\int_0^u \rho(v)\,e(\varphi_0+v)\,dv .
	\]
	
	Comme $\bm T_{\gamma_R}(p_1)=\bm T_\sigma(p_1)=e(\varphi_0)$ et que $\gamma_R$
	est parcouru dans le sens trigonométrique, l'arc de cercle de $p_1$ à $p_2$ est
	paramétré par
	\[
	\gamma(u)=p_1+\int_0^u R\,e(\varphi_0+v)\,dv,\qquad u\in[0,\theta],
	\]
	et $\gamma(\theta)=p_2$.
	
	Comme $\sigma(\theta)=p_2=\gamma(\theta)$, on obtient
	\[
	\int_0^\theta \bigl(R-\rho(u)\bigr)\,e(\varphi_0+u)\,du=0.
	\]
	On projette sur le vecteur unitaire $e\!\left(\varphi_0+\frac\theta2\right)$ :
	\[
	\int_0^\theta \bigl(R-\rho(u)\bigr)\cos\!\Bigl(u-\frac\theta2\Bigr)\,du=0.
	\]
	
	Pour $u\in\,]0,\theta[$, on a $|u-\frac\theta2|<\frac\theta2\leqslant\frac\pi2$,
	donc $\cos(u-\frac\theta2)>0$. La fonction
	$u\mapsto (R-\rho(u))\cos(u-\frac\theta2)$ est continue et positive ou nulle,
	d'intégrale nulle : elle est donc identiquement nulle. Ainsi $\rho\equiv R$ sur
	$]0,\theta[$, puis sur $[0,\theta]$ par continuité. Alors $\sigma(u)=\gamma(u)$
	pour tout $u$, c'est-à-dire que $\sigma_{|[t_1,t_2]}$ est contenue dans
	$\gamma_R$, ce qui contredit l'hypothèse. Il existe donc un point $p_3$ de
	$\sigma([t_1,t_2])$ où $\kappa_\sigma(p_3)<\frac1R$. Ce point n'est pas une
	extrémité, car $\kappa_\sigma(p_i)\geqslant\frac1R$ en $p_1,p_2$ d'après le
	lemme 2.50, et la conclusion de ce lemme joue le rôle de $p_3$ dans
	$\sigma(]t_1,t_2[)$ par continuité de $\kappa_\sigma$.
}

\begin{theoreme}[Des 4 sommets] \label{thm: théorème des 4 sommets}
	Toutes courbes fermées simples de classe $C^2$ dans le plan orienté possède au moins quatre sommets.
\end{theoreme}



\prf{
	On peut exclure le cas du cercle, puisque que chaque point en un sommet. On peut donc supposer que la courbe n'est pas un cercle. De plus comme puisque la courbure est invariant par reparamétrisation directe, on peut aussi supposer que la courbe est paramétrisée naturellement. Soit $\gamma : [a, b] \to \R^2$ de classe $C^2$ une paramétrisée naturelle. Soit $C$ le plus petit cercle contenant la courbe, on va appeler $R$ le rayon de celui-ci. 
	
	On note qu'il y a au moins deux points d'intersection entre $C$ est la courbe. On va distinguer deux cas : 
	\begin{enumerate}
		\item (Deux points d'intersection) On nomme $p_1$ et $p_2$ les deux points d'intersection, nécessairement situé diamétralement opposé. Par le lemme $1$, 
		\[
		\kappa_\gamma \geqslant \frac1R
		\]
		entre $p_1$ et $p_2$. Par le lemme $2$, il existe $p_3$ sur la courbe entre $p_1$ et $p_2$, et $p_4$ sur la courbe entre $p_2$ et $p_3$ tel que $\kappa_\gamma < \frac1R$ entre $p_3$ et $p_2$. Par conséquent, $\kappa_\gamma$ admet un minimum local entre $p_1$ et $p_2$, de même entre $p_2$ et $p_1$. Il va de même que $\kappa_\gamma$ admet un maximum local entre $p_3$ et $p_4$ et un autre entre $p_4$ et $p_3$. On a donc bien quatre sommets.
		
		\item (Plus de deux points d'intersection) Il existe d'office $3$ points d'intersection, que l'on va noté $p_1, p_2$ et $p_3$ tels que l'arc de cercle $p_1$ à $p_2$ est contenu dans un demi cercle fermé et aussi l'arc de cercle de $p_2$ à $p_3$ est contenu dans un demi cercle fermé et finalement
		\[
		\gamma_{\mid_{]p_1, p_2[}} \cap C = \emptyset \et \gamma_{\mid_{]p_2, p_3[}} \cap C = \emptyset
		\]
		Par le lemme 1., la courbe orienté $\kappa_\gamma$ est plus grande que $\frac1R$ entre $p_3$ et $p_2$. Par le lemme 2. il existe $q_1$ sur la courbe entre $p_1$ et $p_2$ et $q_2$ sur la coure entre $p_2$ et $p_3$ tels que $\kappa_\gamma < \frac1R¨$ entre $q_1$ et $q_2$. Et donc par le même argument que dans le point 1., on a bien $4$ sommet.
	\end{enumerate}
}
