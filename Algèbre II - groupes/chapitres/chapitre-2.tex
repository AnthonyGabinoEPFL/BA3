\chapter{Groupes}

\section{définition de base}
\begin{definition}[Groupe] \label{def: Groupe}
	Un \textbf{groupe} est un ensemble $G$ muni d'une \emph{loi de composition interne} tels que : 
	\begin{enumerate}
		\item Il existe $e \in G, \, \forall g \in G, \quad g e = e g = g$
		\item $\forall g \in G, \, \exists ! g^{-1} \in G, \quad g g^{-1} = g^{-1} g = e$
		\item $\forall g_1, g_2, g_3 \in G, \quad g_1(g_2 g_3) = (g_1 g_2) g_3$
	\end{enumerate}	
	De plus on dira que $G$ est \textbf{abélien} si la loi de composition interne est \emph{commutative}.
\end{definition}

\begin{exemple}
	Un groupe important est le \textbf{groupe diédral} que l'on définit par
	\[
	D_{2n} = \{ \text{rotations et réflexions qui préservent un $n$-gone régulier}\}
	\]
	il en va de même avec le \textbf{groupe symétrique} que l'on définit par
	\[
	S_n = \S(\entiers)
	\]
\end{exemple}

\begin{definition}[Action de groupe] \label{def: Action de groupe}
	Une \textbf{action de groupe} $G$ sur un ensemble $X$ noté $G \curvearrowright X$, est une attribution qui à tout élément $g \in G$ associé une bijection 
	\[
	\Phi_g : X \to X
	\]
	qui respecte
	\begin{enumerate}
		\item L'élément neutre,
		\[
		\Phi_e = \mathrm{Id}_X
		\]
		\item L'inverse
		\[
		\forall g \in G, \quad \Phi_{g^{-1}} = (\Phi_g)^{-1}
		\]
		\item Le produit
		\[
		\forall g_1, g_2 \in G, \quad \Phi_{g_1 g_2} = \Phi_{g_1} \circ \Phi_{g_2}
		\]
	\end{enumerate}
	il est facile de voir que la propriétés $2.$ découle des deux autres.
\end{definition}

\begin{remarque}
	Bien que moins précis, on note usuellement les bijections $\Phi_g$ par le symbole
	\[
	\Phi_g(x) = g \cdot x
	\]
	pour tout $g \in G$ et $x \in X$. Dans cette esprit, les propriétés d'une action de groupe devient
	\[
	\forall x \in X, \quad e \cdot x = x, \et \forall g_1, g_2 \in G, \, \forall x \in X, \quad (g_1 g_2) \cdot x = g_1 \cdot (g_2 \cdot x)
	\]
\end{remarque}

\begin{exemple}
	Tout groupe $G$ agit sur le même de 3 façon différentes
	\begin{enumerate}
		\item Son \textbf{action à gauche} $G \curvearrowright G$ est l'attribution
		\[
		h \cdot g = hg, \quad \forall g,h \in G
		\]
		\item Son \textbf{action à droite} $G \curvearrowright G$ est l'attribution
		\[
		h \cdot g = g h^{-1}, \quad \forall g,h \in G
		\]	
		\item Son \textbf{action action adjointe} $G \curvearrowright G$ est l'attribution
		\[
		h \cdot g = h g h^{-1}, \quad \forall h,g \in G
		\]
	\end{enumerate}
\end{exemple}
\prf{Il est facile de voir que les actions à gauche et à droite définissent effectivement des actions de groupes, il reste donc à montrer l'action adjointe. On a 
	\[
	\forall g \in G, \quad e g e^{-1} = g
	\]
	Et finalement on a pour tout $g_1, g_2, h \in G$ alors 
	\[
	g_1 \cdot (g_2 \cdot h) = g_1 \cdot (g_2 h g_2^{-1}) = g_1 g_2 h g_2^{-1} g_1^{-1} = (g_1 g_2) h (g_1 g_2)^{-1} = (g_1 g_2) \cdot h
	\]
	Ainsi l'action adjointe définit bel et bien une action de groupe.
}

\begin{definition}[Morphisme de groupe] \label{def: Morphisme de groupe}
	Soit $G$ et $G'$ deux groupes, une fonction
	\[
	\varphi : G \to G'
	\]
	s'appelle \textbf{morphisme de groupe} si elle préserve les structures de groupe de $G$ et $G'$, c'est-à-dire
	\[
	\varphi(e) = e' \et \forall g_1, g_2 \in G, \quad \varphi(g_1 g_2) = \varphi(g_1) \varphi(g_2)
	\]
	De plus si $\varphi$ est également bijectif, alors on parlera d'isomorphisme et on pourra noté
	\[
	G \simeq G'
	\]
\end{definition}

\begin{proposition}
	Si $\varphi : G \to G'$ est un isomorphisme alors $\varphi^{-1} : G' \to G$ est également un isomorphisme.
\end{proposition}

\prf{Soit $g, h \in G$, on a
\[
gh = (\varphi^{-1} \circ \varphi) (gh) = \varphi^{-1} (\varphi(gh)) = \varphi^{-1}(\varphi(g) \varphi(h))
\]
de plus comme $\varphi$ est un isomorphisme alors il existe $g', h' \in G'$ tels que $\varphi(g) = g'$ et $\varphi(h) = h'$ et donc on a
\[
\varphi^{-1}(g') \varphi^{-1}(h') = \varphi^{-1} (g' h')
\]
Ainsi $\varphi^{-1}$ définit bien un isomorphisme.
}

\subsection{Sous-groupe}
\begin{definition}[Sous-groupe] \label{def: Sous-groupe}
	Soit $G$ un groupe, un sous-ensemble $H \subset G$ est appelé \textbf{sous-groupe} s'il est fermé par rapport à
	\begin{enumerate}
		\item $e \in H$
		\item $\forall h \in H, \quad h^{-1} \in H$
		\item $\forall h_1, h_2 \in H, \quad h_1 h_2 \in H$
	\end{enumerate}
\end{definition}

\begin{lemme}\label{lem: Noyau trivial}
	Un morphisme $f : G \to G'$ est injectif si et seulement si son noyau
	\[
	\ker f = \{ g \in G \mid f(g) = e'\}
	\]
	est le \emph{sous-groupe trivial} $\{e\} \subset G$.
\end{lemme}

\begin{theoreme}[Théorème de Cayley] \label{thm: Théorème de Cayley}
	Tout groupe $G$ est un groupe de permutation, c'est-à-dire un sous-groupe de $S_X$ pour un certain ensemble $X$.
\end{theoreme}

\prf{Il faut construire un morphisme injectif $\Psi : G \to S_X$ et donc par le \autoref{lem: Noyau trivial} on doit avoir $\ker \Psi = \{e\}$. De plus on a vu que $\Psi$ peut être induit par une action $G \curvearrowright X$.
	
Or on a
\[
\ker \Psi = \{g \in G \mid \Psi_g = \mathrm{Id}_X\} = \{ g \in G \mid g \cdot x = x, \forall x \in X\}
\]
Et donc il suffit de trouver une action fidèle\footnote{une action est dit \textbf{fidèle} si son noyau est juste le sous-groupe trivial.}. Or il se trouve que l'action à gauche est fidèle, en effet on a 
\[
\ker = \{ g \in G \mid g \cdot h = h, \forall h \in G\} = \{ g \in G \mid gh = h, \forall h \in G\} = \{ g \in G \mid g = e\} = \{e\}
\]
}

\subsection{Relation d'équivalence induit par une action de groupe}
\begin{definition}
	Soit $G \curvearrowright X$, alors cela induit une relation d'équivalence sur $X$ via
	\[
	x \sim y \iff \exists g \in G, \quad g \cdot x = y
	\]
	Et donc on peut alors parler de classe d'équivalence appelé \textbf{orbite}, et elle peut s'écrire
	\[
	G \cdot x = \{g \cdot x \mid g \in G\}
	\]
	et elle sera appelé l'orbite de $x$. De plus s'il n'y a qu'une seule orbite, alors l'action $G \curvearrowright X$ on parlera d'action \textbf{transitive}. Et finalement si $G \curvearrowright X$ est une action de groupe, alors pour tout $x \in X$ on définit son \textbf{stabilisateur} comme 
	\[
	\stab_G(x) = \{g \in G \mid g \cdot x = x\}
	\]
	Il est facile de voir que $\stab_G(x)$ est un sous-groupe de $G$. Une action de groupe $G \curvearrowright X$ est dit \textbf{libre} si
	\[
	\forall x \in X, \quad \stab_G(x) = \{e\}
	\]
\end{definition}

\begin{theoreme}[théorème de l'orbite-stabilisateur] \label{thm: théorème de l'orbite-stabilisateur}
	Soit $G \curvearrowright X$ une action de groupe fini, alors
	\[
	\forall x \in X, \quad |G \cdot x| = \frac{|G|}{|\stab_G(x)|}
	\]
\end{theoreme}

\prf{On considère l'application,
	\[
	\begin{array}{cccc}
		f : & G & \to & G\cdot x \\
			& g & \mapsto & g\cdot x
	\end{array}
	\]
	alors il est facile de voir que $f$ est surjective. De plus on a
	\[
	f^{-1} (h \cdot x) = \{ g \in G \mid g \cdot x = h \cdot x\} = \{ g \in G \mid h^{-1} g \cdot x = x\} = h \stab_G(x)
	\]
	Et donc on 
	\[
	\forall x \in X, \quad |f^{-1}(h \cdot x)| = |\stab_G(x)|
	\]
	Mais comme 
	\[
	|G| = |f(G)| |f^{-1}(x)| = |G \cdot x| |f^{-1}(x)|
	\]
	ce qui conclut la preuve.
	
}

\begin{remarque} \label{rmq: théorème de l'orbite-stabilisateur}
	De plus si $X$ est finit, alors on a
	\[
	|X| = \sum_{G \cdot x} |G \cdot x| = \sum_{G \cdot x} \frac{|G|}{|\stab_G(x)|}
	\]
\end{remarque}

\subsection{Action à droite et Action à gauche}
\begin{definition}
	Lorsque $H$ est un sous-groupe de $G$, alors les orbites de l'action à gauche
	\[
	H \curvearrowright G, \quad h \cdot g = hg
	\]
	sont les \textbf{classe à droite}. Cela vient du fait que les orbites s'écrivent
	\[
	Hg = \{hg \mid h \in H\}
	\]
	où $g$ est à droite. Il en va de même pour les orbites de l'action à droite
	\[
	H \curvearrowright G, \quad h \cdot g = g h^{-1}
	\]
	sont appelées \textbf{classe à gauche}
	\[
	g H = \{gh^{-1} \mid h \in H\} = \{ g h \mid h \in H\}
	\]De plus il est facile de voir que les actions à gauche et à droite sont libre. 	
	On note
	\[
	G / H \et H \backslash G
	\]
	respectivement l'ensemble des classes à gauches de $G$ par rapport à $H$ et l'ensemble des classes à droite droite de $G$ par rapport à $H$.
\end{definition}

\begin{proposition}
	Supposons que $G$ soit fini, et que $H \leq G$ est un sous-groupe quelconque alors
	\[
	|G / H| =  \frac{|G|}{|H|} = |H \backslash G|
	\]
	On désigne alors ce nombre par $[G : H]$ et on l'appelle l'\textbf{indice} du sous-groupe $H$ de $G$.
\end{proposition}

\prf{En effet on remarque que les actions à gauche et à droite sont libres, car
	\[
	hg = g \ou  g h^{-1} = g \implies h = e
	\]
	ainsi $\stab_H(g) = \{e\}$ pour tout $g \in G$. Et donc par le \autoref{thm: théorème de l'orbite-stabilisateur} alors on a 
	\[
	\forall g \in G, \quad |H \cdot g | = |H|
	\]
	Mais puisque $G$ est partitionné en classes à gauche ou à droite, on conclus que le nombre de ces classes est exactement
	\[
	|G/H| = |H \backslash G| = \frac{|G|}{|H|}
	\]
}
\subsection{Action adjointe}
\begin{definition}
	On considère désormais l'action adjointe $G \curvearrowright G$, alors les orbites sont appelés \textbf{classe de conjugaison}. Plus précisément, la classe de conjugaison de $g \in G$ est l'ensemble
	\[
	\{h g h^{-1} \mid h \in G\}
	\]
	Les éléments de la même orbites sont appelés \textbf{conjugués}. En parallèle, le stabilisateur de $g$ par rapport à l'action adjointe est appelé son \textbf{centralisateur}
	\[
	C_G(g) = \{h \in G \mid h g h^{-1} = g\} = \{	h \in G \mid hg = gh\}
	\]
	Autrement dit le centralisateur correspond à l'ensemble des éléments de $G$ qui commute avec $g$.
\end{definition}

\begin{proposition}
	Si $g_1$ et $g_2$ sont conjugués dans un groupe $G$, alors leurs centralisateurs sont isomorphes.
\end{proposition}

\prf{Si $g_1 = h g_2 h^{-1}$, alors l'application $x \mapsto h^{-1} x h$ donne un isomorphisme $C_G(g_1) \to C_G(g_2)$.
}

\begin{proposition} \label{prop: Équation de classe}
	Soit $G$ un groupe, alors
	\[
	1 = \sum_{\text{classe de conjugaison } \tilde g} \frac1{|C_G(\tilde g)|}
	\]
\end{proposition}

\prf{Par la \autoref{rmq: théorème de l'orbite-stabilisateur} on a
	\[
	|G| = \sum_{\text{classe de conjugaison } \tilde g} |\tilde g|
	\]
	Or par le \autoref{thm: théorème de l'orbite-stabilisateur}, on sait que
	\[
	|\tilde g| = \frac{|G|}{|\stab_G(\tilde g)|}
	\]
	Mais comme les stabilisateurs ne sont autres que les centralisateurs, on a
	\[
	|\tilde g| = \frac{|G|}{|C_G(\tilde g)|}
	\]
	Ce qui conclut bien que
	\[
		1 = \sum_{\text{classe de conjugaison } \tilde g} \frac1{|C_G(\tilde g)|}
	\]
	d'où la proposition.	
}

\subsection{Groupe diédral}
On se rappelle que l'on peut définir
\[
D_{2n} = \{e, \sigma, \sigma^2, \ldots, \sigma^{n-1}, \tau, \sigma \tau, \sigma^2 \tau, \ldots, \sigma^{n-1} \tau\}
\]
où $\sigma$ désigne un rotation d'angle $\frac{2\pi}{n}$ et $\tau$ est un réflexion d'axe principale. On va regarder le cas $n=3$, on a donc
\[
D_6 = \{e, \sigma, \sigma^2, \tau, \sigma \tau, \sigma^2 \tau\}
\]
où $\sigma^3 = \tau^2 = e$ et $\sigma \tau = \tau \sigma^{-1}$. On remarque alors que l'on a trois classe de conjugaison
\[
\{e\}, \quad \{\sigma, \sigma^2\} \et \{\tau, \sigma \tau, \sigma^2 \tau\}
\]
De plus il est facile de voir que le centralisateur est respectivement le groupe entier, les réflexions et les symétries. Ainsi par la \autoref{prop: Équation de classe} on a
\[
1 = \frac16 + \frac13 + \frac12
\]
ce qui est assurément une affirmation véridique.

\subsection{Groupe symétrique}
On se rappelle que tout permutation $\sigma \in \S_n$ peut s'écrire comme un produit de cycles disjoints. La définition suivante est alors naturelle
\begin{definition}
	Soit $\sigma \in \S_n$ une permutation, alors le \textbf{type cyclique} de $\sigma$ sera une \emph{partition} de $n$, c'est-à-dire une collection d'entiers positifs,
	\[
	\lambda = (\lambda_1 \geqslant \lambda_2 \geqslant \ldots \geqslant \lambda_k)
	\]
	avec une somme totale $|\lambda| = \sum_i |\lambda_i| = n$.
\end{definition}

\begin{proposition}
	Deux permutations $\sigma, \tau \in \S_n$ sont conjugués si et seulement s'ils ont le même type cyclique.
\end{proposition}

\prf{On remarque que si on considère $\sigma, \tau \in \S_n$, alors
	\[
	\tau \sigma \tau^{-1} =
	\begin{pmatrix}
		\cdots & \tau(i) & \cdots \\
		\cdots & \tau(\sigma(i))& \cdots 
	\end{pmatrix}
	\]
	Ainsi, il y a une correspondance biunivoque entre les cycles $i_1 \to i_1 \to \ldots \to i_k \to i_1$ de la permutation $\sigma$ et les cycles $\tau(i_1) \to \tau(i_2) \to \ldots \to \tau(i_k) \to \tau(i_1)$ de la permutation $\tau \sigma \tau^{-1}$. Cela montre immédiatement que toutes permutations conjuguées ont les types cycliques, mais cela montre aussi la réciproque : car toute permutation de type cyclique $\lambda$ peut être écrite comme $\tau \sigma_\lambda \tau^{-1}$ pour $\sigma_\lambda$ et $\tau$ bien choisit
}

Ainsi il reste à déterminer l'ordre des centralisateurs d'un éléments donné. Mais il suffit de le faire pour le représentant $\sigma_\lambda$. On a $\tau \in C_{\S_n}(\sigma_\lambda)$ si et seulement si
\[
\tau \sigma_\lambda \tan^{-1} = \sigma_\lambda
\] 
Cela signifie que $\tau$ doit permuter les cycles de $\sigma_\lambda$ de longueur $i$ entre eux. Ainsi si on note $\#^i_\lambda$ le nombre de tel cycles, on peut se convaincre que
\[
|C_{\S_n}(\sigma_\lambda)| = \prod_{i \geqslant 1} i^{\#^i_\lambda} \#^i_\lambda!
\]
Et donc on peut conclure par la \autoref{prop: Équation de classe} que
\[
1 = \sum_{\lambda \text{ une partition de $n$}} \frac1{\prod_{i \geqslant 1} i^{\#^i_\lambda} \#^i_\lambda!}
\]
Ce qui pour $n=4$ nous donne
\[
1 = \frac1{24} + \frac14 + \frac18 + \frac13 + \frac14
\]

\begin{definition}[Sous-groupe normal] \label{def: Sous-groupe normal}
	Soit $H \leq G$ un sous-groupe de $G$, on dira que $H$ est un sous-groupe \textbf{normal} de $G$, si
	\[
	\forall g \in G, \quad g H g^{-1} = H
	\]
	et on le note $H \trianglelefteq G$.
\end{definition}

\begin{lemme}
	Si $f : G \to G'$ est un morphisme de groupe quelconque, alors $\ker f$ est un sous-groupe normal
\end{lemme}

\prf{Soit $g \in G$, et $h \in \ker f$, alors
	\[
	f(g h g^{-1}) = f(g) f(h) f(g^{-1}) = f(g) f(g)^{-1} = e'
	\]
	ainsi $g h g^{-1} \in \ker f$.
}

\begin{proposition}
	On  a deux propriétés importantes des sous-groupes normaux : 
	\begin{enumerate}
		\item Soit $H_1, H_2 \trianglelefteq G$, alors $H_1 \cap H_2 \trianglelefteq G$,
		
		\item Si $H \trianglelefteq G $, alors pour tout $F \leq G$ tel que $H \subset F$ alors $H \trianglelefteq F$
	\end{enumerate}	
\end{proposition}

\prf{
	\begin{enumerate}
	\item Soit $g \in G$ et $h \in H_1 \cap H_2$, alors
	\[
	g h g^{-1} \in H_1 \et g h g^{-1} \in H_2
	\]
	car $H_1$ et $H_2$ sont normaux, et donc on a bien $g h g^{-1} \in H_1 \cap H_2$. Et donc $H_1 \cap H_2 \trianglelefteq G$.
	\item Soit $f \in F$ et $h \in H$, alors en particulier $f \in G$ et donc par normalité de $H$
	\[
	f h f^{-1} \in H
	\]
	De plus comme $H$ est un sous-groupe de $G$ contenu dans $F$, c'est un sous-groupe de $F$. Ainsi on a bien $H \trianglelefteq F$.
	\end{enumerate}
}


\begin{theoreme}[Premier théorème d'isomorphisme] \label{thm: Premier théorème d'isomorphisme}
	Pour tout morphisme $f : G \to G'$, on a un isomorphisme
	\[
	G / \ker f \simeq\Ima f
	\]
	induit par $[g] \mapsto f(g)$, pour tout $g \in G$.
\end{theoreme}

\begin{definition}
	Pour tout sous-ensemble $X$ d'un groupe $G$, on définit son \textbf{centralisateur} comme
	\[
	C_G(X) = \big\{ g \in G \mid  gx = xg, \forall x \in X\}
	\]
	ainsi son \textbf{normalisateur} comme
	\[
	N_G(X) = \{g \in G \mid g X = Xg\}
	\]
	Il est clair que $C_G(X) \leq N_G(X)$ pour tout ensemble $X \subset G$.
\end{definition}

\begin{proposition}
	Pour tout ensemble $X \subset G$, son centralisateur est un sous-groupe normal de sons normalisateur
	\[
	C_G(X) \trianglelefteq N_G(X)
	\]
\end{proposition}

\prf{On remarque que pour tout sous-ensemble $X$ d'un groupe, on a une action naturelle
	\[
	N_G(X) \curvearrowright X, \quad g \cdot x = g x g^{-1}
	\]
	Le noyau de cette action est, par définition, le sous-groupe centralisateur $C_G(X)$. Étant donné que le noyau de tout action de groupe est normal, ainsi la proposition suit. De plus, on obtient une action fidèle
	\[
	N_G(X) / C_G(X) \curvearrowright X
	\]
}

\begin{definition}
	Un \textbf{automorphisme} d'un groupe $G$ est un isomorphisme $G \to G$, et on écrit
	\[
	\Aut G
	\]
	pour le groupe des automorphismes de $G$, par rapport à la composition. De plus on dit qu'un groupe $L$ agit sur un groupe $K$ \emph{par automorphisme}, encore noté
	\[
	L \curvearrowright K
	\]
	si les bijections $\Phi_j(k) = l \cdot k$ sont des automorphismes pour tout $l \in L$.
\end{definition}

\begin{lemme}
	Si $H$ est un sous-groupe de $G$, alors l'action
	\[
		N_G(H) / C_G(H) \curvearrowright H
	\]
	est par automorphisme.
\end{lemme}

\prf{C'est un conséquence immédiate de la formule
	\[
	g hh' g^{-1} = g h g^{-1} g h' g^{-1}
	\]
	pour tout $h, h' \in H$ et $g \in G$.
}

\begin{proposition}
	Soit $H$ et $K$ des sous-groupes de $G$, alors
	\begin{enumerate}
		\item On a $HK = KH$ si et seulement si $HK$ est un sous-groupe de $G$.
		\item Si $H$ et $K$ sont des sous-groupes normaux de $G$, alors $HK$ est un sous-groupe normal de $G$.
	\end{enumerate}
\end{proposition}

\begin{theoreme}
	Si $K$ et $H$ sont des sous-groupes de $G$ tels que $K \leq N_G(H)$, alors
	\[
	K /K \cap H \simeq HK/ H
	\]
\end{theoreme}
